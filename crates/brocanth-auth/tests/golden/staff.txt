//! My Blog -- everything this binary contains, in the order it happens.
//!
//! This file is yours. `brocanth add` does not rewrite it: what it would
//! have to say is in `src/plugins/mod.rs`, which is generated, and this one
//! only asks that file four questions.
//!
//! A plugin that is not named in `plugins::register!` is not registered, and
//! one that is not in `Cargo.toml` is not compiled at all. "The framework
//! decided for you and you cannot see where" is the thing this layout exists
//! to make impossible.
//!
//! `serve` and every other command read this, so there is one answer to what
//! this application is.

use std::sync::Arc;
use std::time::{Duration, Instant};

use brocanth::plugin_app::AppBuilder;
use brocanth::plugin_hooks::{self as hooks, Begin, RequestState};
use brocanth::plugin_routes::BoxFuture;
use brocanth::plugin_shutdown::ShutdownReason;
use brocanth::routing::{AsyncMiddlewareStack, Dispatch, Router, Verb};

use crate::pages::{self, Ctx, Handler, IntoResponse, Response};
use crate::plugins;

/// The application name. Every environment variable this binary reads is
/// `MY_BLOG_<SUFFIX>`: one convention, derived once, whatever spelling the
/// name arrived in.
pub const APP: &str = "my-blog";

/// Where `serve` listens when nothing says else: `$MY_BLOG_ADDR`, or
/// `127.0.0.1:3000`. The container image sets `0.0.0.0:3000`.
pub fn default_addr() -> String {
    std::env::var("MY_BLOG_ADDR").unwrap_or_else(|_| "127.0.0.1:3000".to_string())
}

/// What a request needs, built once and shared by every connection.
pub struct App {
    pub router: Router<Handler>,
    pub stack: AsyncMiddlewareStack<Ctx, Response>,
    pub plugins: Vec<String>,
    /// The application the plugins built.
    ///
    /// Held rather than dropped, because three things need it
    /// after `build` returns: a per-request context, the routes a
    /// plugin contributed, and an HTTP test client. It used to be
    /// a boxed closure over `context()` alone, which cost an
    /// indirect call per request and reached none of the rest.
    ///
    /// Nameable at last: `plugins::Installed` and `plugins::Hooks`
    /// are the two type parameters, and the generator writes both.
    pub core: Core,
}

/// The built application, with its provider list and hook record.
type Core = brocanth::plugin_app::App<plugins::Installed, plugins::Hooks>;

/// Builds it. The only place that decides what this binary is.
pub fn build() -> Result<App, String> {
    // 1. The application, and every plugin it contains. The chain is in
    //    `src/plugins/mod.rs`, one line per plugin and each one a line a
    //    reader can delete. It is there rather than here because the
    //    generator rewrites it whenever the module list changes.
    let app = plugins::register!(AppBuilder::new())
        .setup(|setup| {
            // A plugin *offers* middleware and cannot mount it: a
            // `MiddlewareCatalog` has no `mount`. Mounting takes a pattern, the
            // pattern is a decision, and the decision is the application's --
            // which is why these lines are here, in the file you own.
            // Nothing in this binary offers middleware, so nothing is mounted.
            let _ = setup.router();
            Ok(())
        })
        .build()
        .map_err(|e| e.to_string())?;

    // 2. The route table: the shape of src/pages and nothing else. A file is
    //    a route, and `pages::router` is the only place that says so.
    let router = pages::router().map_err(|e| e.to_string())?;

    // 3. The chain, from exactly what the block above mounted. A name this
    //    binary has no layer for stops the application rather than silently
    //    not running.
    let mut stack: AsyncMiddlewareStack<Ctx, Response> = AsyncMiddlewareStack::new();
    for mount in app.router_plan().mounts() {
        let layer = plugins::layer(&mount.middleware).ok_or_else(|| {
            format!(
                "`{}` is mounted, but this binary has no layer for it",
                mount.middleware
            )
        })?;
        stack
            .use_at(&mount.pattern, layer)
            .map_err(|e| e.to_string())?;
    }

    // 4. What the plugins put on the per-request context, said once.
    plugins::announce!(&app);

    let plugins = app.instance_names().map(str::to_string).collect();

    Ok(App {
        router,
        stack,
        plugins,
        core: app,
    })
}

/// Turns a page's answer into the response the hooks see.
///
/// `text/html` unless the page names its own `content-type`, which then
/// replaces it rather than joining it. Every other header is appended,
/// because `set-cookie` has to repeat. A page that answered with a stream
/// keeps it: the hooks see its headers, and the server polls its body.
fn sent(answered: Response) -> hooks::Response {
    match answered.stream {
        Some(stream) => {
            brocanth_server::bridge::streaming(answered.status, answered.headers, stream)
        }
        None => brocanth_server::bridge::buffered(
            answered.status,
            answered.headers,
            answered.body.into_bytes(),
        ),
    }
}

/// The request headers as text, for the page.
///
/// A value that is not UTF-8 is dropped: a page reads a header as `&str`,
/// and a lossy copy would say something the client did not send.
fn headers(request: &hooks::Request) -> Vec<(String, String)> {
    request
        .headers()
        .iter()
        .filter_map(|(name, value)| Some((name.to_string(), value.to_str().ok()?.to_string())))
        .collect()
}

/// How long a stop waits for open requests and then for the plugins.
///
/// Kubernetes sends SIGTERM, waits `terminationGracePeriodSeconds` (30 by
/// default) and then sends SIGKILL. 25 seconds leaves 5 for the exit.
/// `[server] shutdown_grace_secs` overrides it: `--server.shutdown_grace_secs`,
/// `$<APP>_SERVER__SHUTDOWN_GRACE_SECS`, or with the config plugin the file.
/// See `plugins::shutdown_grace`.
pub const SHUTDOWN_GRACE: Duration = Duration::from_secs(25);

/// Serves until SIGTERM or SIGINT, then stops in two phases (PLAN.md ADR 2).
///
/// 1. The listener closes. Each open request has the grace period to
///    finish, and each open response stream is told to end.
/// 2. Each plugin stops, in reverse registration order, inside what is left
///    of the grace period.
///
/// An error is a non-zero exit, and an orchestrator reads it: the drain ran
/// out of time, the listener failed, or a plugin did not stop cleanly.
///
/// Synchronous on the outside because the command that calls it is, and
/// the runtime is built here rather than wrapped around `main`: a command
/// that never serves should not start a reactor to run.
pub fn serve(addr: &str) -> Result<(), String> {
    let grace = plugins::shutdown_grace()?;
    // Before the socket, so a signal during startup stops the server
    // through the drain and does not kill the process.
    let stop = watch(grace).map_err(|e| e.to_string())?;
    let runtime = tokio::runtime::Builder::new_multi_thread()
        .enable_all()
        .build()
        .map_err(|e| e.to_string())?;

    // The runtime first, and the application inside it. A plugin
    // that opens a connection pool builds it in `before_start`
    // with no connection, and the pool spawns its upkeep task on
    // the current runtime. Building the application first would
    // leave it with no runtime to spawn on.
    let app = {
        let _guard = runtime.enter();
        build()?
    };
    let names = app.plugins.join(", ");
    plugins::say!(info, "plugins: [{names}]"; plugins = %names);

    let result = runtime.block_on(run(addr, app, stop, grace));
    // A request that outlived the drain can still be on a worker. Waiting
    // for it here would undo the grace period, so it is dropped.
    runtime.shutdown_background();
    result
}

/// The async half of [`serve`].
async fn run(
    addr: &str,
    mut app: App,
    mut stop: tokio::sync::mpsc::UnboundedReceiver<Stop>,
    grace: Duration,
) -> Result<(), String> {
    // Every readiness check once, before the socket: the same checks
    // `/readyz` runs. The build waits for no I/O, so a wrong password or
    // an unreachable database stops the start here, with the plugin's
    // message and its `help:` line, and not at the first request. A
    // check can wait for a pool's acquire timeout, so a signal ends the
    // wait.
    tokio::select! {
        checked = app.core.readiness().check_all() => {
            checked.map_err(|not_ready| not_ready.to_string())?;
        }
        _ = stop.recv() => {
            let report = app.core.shutdown(ShutdownReason::Signal, Some(grace));
            plugins::say!(info, "{report}");
            return Ok(());
        }
    }

    // One gate: a stop at the listener and a stop at `App::shutdown`
    // close the same set of response streams.
    let server = brocanth_server::Server::bind(addr)
        .await
        .map_err(|e| e.to_string())?
        .with_streams(app.core.streams().clone());
    let bound = server.local_addr().map_err(|e| e.to_string())?;
    let stopper = server.stopper();

    // After the bind, so a plugin does not start work for a port that
    // turned out to be taken.
    if let Err(e) = app.core.start() {
        let report = app.core.shutdown(ShutdownReason::Failed, None);
        plugins::say!(error, "{report}");
        return Err(e.to_string());
    }

    // The probes and `/metrics` on a port of their own, when the
    // configuration names one.
    plugins::side_listeners(&app).await?;

    let app = Arc::new(app);
    // `plugins::wrap` puts whatever observes a request around it.
    let handler = plugins::wrap(&app, {
        let app = Arc::clone(&app);
        move |request| {
            let app = Arc::clone(&app);
            async move { handle(&app, request).await }
        }
    });
    let mut serving = tokio::spawn(server.run(handler));
    plugins::say!(info, "listening on http://{bound}"; addr = %bound);

    // Phase 1: stop accepting, and give the open requests the grace period.
    let mut failed = tokio::select! {
        event = stop.recv() => {
            let signal = match event {
                Some(Stop::Signal(signal)) => signal,
                _ => "stop",
            };
            plugins::say!(
                info,
                "{signal}: stopping; open requests have {grace:?} to finish";
                signal,
                ?grace
            );
            None
        }
        // The listener stopped by itself. That is a failure, and the
        // plugins still stop.
        served = &mut serving => Some(stopped(served)),
    };
    let reason = match failed {
        None => ShutdownReason::Signal,
        Some(_) => ShutdownReason::Failed,
    };
    let started = Instant::now();
    // Not ready from the first instant of the stop (ADR 2): `/readyz`
    // answers 503, and a plugin that awaits `Readiness::draining` stops
    // taking work now -- the job worker stops claiming -- while the
    // listener still drains.
    app.core.readiness().drain();
    if failed.is_none() {
        // With `observe`, `/readyz` fails first, and the listener closes
        // after `[observability] drain_delay_secs`. Inside the grace period.
        plugins::stop_accepting(&app, stopper).await;
        failed = tokio::select! {
            served = &mut serving => match served {
                Ok(Ok(())) => None,
                other => Some(stopped(other)),
            },
            event = stop.recv() => Some(match event {
                // A second Ctrl-C means "now".
                Some(Stop::Signal(signal)) => {
                    format!("a second {signal} arrived, so open requests were cut off")
                }
                _ => format!("open requests did not finish in {grace:?} and were cut off"),
            }),
        };
    }
    // Nothing when the drain finished. After a timeout, this drops the
    // connections that are still open.
    serving.abort();

    // Phase 2: the plugins, inside what is left of the grace period.
    let remaining = grace.saturating_sub(started.elapsed());
    match owned(app).await {
        Some(mut app) => {
            let report = app.core.shutdown(reason, Some(remaining));
            plugins::say!(info, "{report}");
            if failed.is_none() && (!report.is_clean() || report.overran()) {
                failed = Some(format!("the plugins did not stop cleanly in {grace:?}"));
            }
        }
        None => plugins::say!(
            warn,
            "a request still holds the application, so no plugin stopped"
        ),
    }
    match failed {
        None => Ok(()),
        Some(why) => Err(why),
    }
}

/// Why the listener task ended before a clean drain.
fn stopped(
    served: Result<Result<(), brocanth_server::ServeError>, tokio::task::JoinError>,
) -> String {
    match served {
        Ok(Ok(())) => "the listener stopped with no signal".to_string(),
        Ok(Err(e)) => e.to_string(),
        Err(e) => e.to_string(),
    }
}

/// The application back from the connections that shared it, or `None` when
/// one of them still runs.
///
/// `App::shutdown` needs `&mut`. A finished connection task can be a few
/// instructions from dropping its clone, so this yields a few times first.
async fn owned(mut app: Arc<App>) -> Option<App> {
    for _ in 0..64 {
        match Arc::try_unwrap(app) {
            Ok(app) => return Some(app),
            Err(shared) => {
                app = shared;
                tokio::task::yield_now().await;
            }
        }
    }
    None
}

/// What the stop thread reports to [`run`].
enum Stop {
    /// SIGTERM or SIGINT, by name.
    Signal(&'static str),
    /// The grace period since the first signal is over.
    Deadline,
}

/// Watches for SIGTERM and SIGINT on a thread of its own. It reports each
/// signal, and the end of the grace period after the first one.
///
/// Not on the serving runtime. A handler is synchronous and holds its
/// worker, and while it runs that runtime can leave its signal and timer
/// drivers unpolled: a stop would wait for the slowest request, and the
/// grace period would not run out. This thread has a runtime of its own,
/// and the channel wakes `run` directly.
fn watch(grace: Duration) -> std::io::Result<tokio::sync::mpsc::UnboundedReceiver<Stop>> {
    let runtime = tokio::runtime::Builder::new_current_thread()
        .enable_all()
        .build()?;
    let mut signals = {
        let _guard = runtime.enter();
        Signals::listen()?
    };
    let (tx, rx) = tokio::sync::mpsc::unbounded_channel();
    std::thread::Builder::new()
        .name("stop".to_string())
        .spawn(move || {
            runtime.block_on(async move {
                let first = signals.next().await;
                if tx.send(Stop::Signal(first)).is_err() {
                    return;
                }
                let next = tokio::select! {
                    () = tokio::time::sleep(grace) => Stop::Deadline,
                    second = signals.next() => Stop::Signal(second),
                };
                let _ = tx.send(next);
            });
        })?;
    Ok(rx)
}

/// SIGTERM and SIGINT as one stream. An orchestrator sends SIGTERM, and
/// Ctrl-C is SIGINT.
#[cfg(unix)]
struct Signals {
    term: tokio::signal::unix::Signal,
    int: tokio::signal::unix::Signal,
}

#[cfg(unix)]
impl Signals {
    fn listen() -> std::io::Result<Self> {
        use tokio::signal::unix::{signal, SignalKind};
        Ok(Self {
            term: signal(SignalKind::terminate())?,
            int: signal(SignalKind::interrupt())?,
        })
    }

    /// The next signal, by name.
    async fn next(&mut self) -> &'static str {
        tokio::select! {
            _ = self.term.recv() => "SIGTERM",
            _ = self.int.recv() => "SIGINT",
        }
    }
}

/// Ctrl-C alone, where there is no SIGTERM.
#[cfg(not(unix))]
struct Signals;

#[cfg(not(unix))]
impl Signals {
    fn listen() -> std::io::Result<Self> {
        Ok(Self)
    }

    /// The next signal, by name.
    async fn next(&mut self) -> &'static str {
        let _ = tokio::signal::ctrl_c().await;
        "Ctrl-C"
    }
}

/// One request: the request hooks, the table, then the response hooks.
///
/// A request hook may answer instead of the page -- a CSRF refusal, a
/// 429 -- and then the page never runs. It may also rewrite the request
/// first, so the table reads the request as the hooks left it. The
/// response hooks see every answer, the 404 and the 405 included.
///
/// `begin_async` and not `begin`: it runs a hook that waits as well as
/// one that does not. With no hook that waits, it is ready on its first
/// poll.
pub async fn handle(app: &App, request: brocanth_server::Request) -> brocanth_server::Response {
    let Ok(mut request) = brocanth_server::bridge::to_core(request) else {
        // The path and the query do not make a URI. The table was not
        // asked, so this is not a 404.
        return brocanth_server::Response::text(400, "Bad Request");
    };
    let mut state = RequestState::new();
    let outcome = match app.core.begin_async(&mut request, &mut state).await {
        Begin::Dispatch => Ok(sent(dispatch(app, &request, &mut state).await)),
        Begin::Answered(outcome) => outcome,
    };
    let response = app.core.finish(&mut request, &mut state, outcome);
    brocanth_server::bridge::from_core(response)
}

/// The table, then the chain, then the handler.
///
/// The chain runs only for a route that matched. A 404 has no route and
/// therefore no context. `state` is what the request hooks wrote; the
/// page reads it through `Ctx::hooked`, and it goes back afterwards so
/// the response hooks read what the page left.
///
/// `async` because a page may be: the chain and the page are awaited
/// here, on the worker, and a page that waits on a database does not
/// hold a thread while it does.
pub async fn dispatch(app: &App, request: &hooks::Request, state: &mut RequestState) -> Response {
    let Some(verb) = Verb::from_http(request.method()) else {
        return Response::new(501, "Not Implemented");
    };
    let path = request.uri().path();
    match app.router.dispatch(verb, path) {
        Dispatch::Found { handler, matched } => {
            let mut ctx = Ctx::new(verb, path, matched.params())
                .with_plugins(app.core.context_for(request, state))
                .with_state(std::mem::take(state))
                .sent(
                    request.uri().query().unwrap_or(""),
                    request.body().clone(),
                    headers(request),
                );
            // The one place a failed handler becomes a response --
            // what Echo calls an HTTPErrorHandler. It sits inside
            // the chain's terminal, so a layer still sees what the
            // error turned into and can rewrite it.
            //
            // `handler(c)` is called before the future is built: the
            // future may borrow the context and nothing else.
            let response = app
                .stack
                .run(path, &mut ctx, |c: &mut Ctx| -> BoxFuture<'_, Response> {
                    let answer = handler(c);
                    Box::pin(
                        async move { answer.await.unwrap_or_else(IntoResponse::into_response) },
                    )
                })
                .await;
            *state = ctx.take_state();
            response
        }
        Dispatch::MethodNotAllowed { allow, .. } => Response {
            status: 405,
            headers: vec![("allow".to_string(), allow.to_string())],
            ..Default::default()
        },
        Dispatch::NotFound => Response::new(404, "Not Found"),
    }
}

//! The route table, which is this directory.
//!
//! One file is one route, and the path on disk is the URL: `index.rs` is `/`,
//! `about.rs` is `/about`, `users/[id].rs` is `/users/{id}`.
//!
//! Nothing below lists them. The macro reads this directory while the crate
//! is compiled and writes the `mod` lines and [`router`] from what is there,
//! so adding a page is adding a file and renaming one renames its route.
//! The verb comes off the handler's name: `handle` is GET, `handle_post` is
//! POST, and a `pub fn handle_*` that is not a verb is a build error rather
//! than a handler nothing reaches.
//!
//! `brocanth_pages::pages!` and `cargo expand` is how to read the result.

brocanth_pages::pages!("src/pages", mounts);

use brocanth::plugin_body::{BodyStream, Streaming};
use brocanth::plugin_context::{Context, ContextItem, Has};
use brocanth::plugin_hooks::RequestState;

/// What a handler produces.
///
/// The framework owns rendering: a handler returns a value and never writes a
/// response itself, so a half-written body on an error is not something that
/// can happen. Echo and Gin hand the handler a writer and take an error back,
/// and both carry the scar -- once a 200 has gone out, the central error
/// handler cannot take it back. Here there is nothing to take back.
///
/// The one exception is opt-in: [`Response::stream`] and [`Response::events`]
/// answer with a body that is written after the handler returns. A stream
/// that fails halfway cannot become a 500, because the status is already on
/// the wire. Use it for server-sent events and large downloads, and nowhere
/// else.
pub struct Response {
    pub status: u16,
    pub body: String,
    pub headers: Vec<(String, String)>,
    /// A body the server polls after the handler returns. `None` for every
    /// page that answers with `body`, which is nearly all of them; when it
    /// is `Some`, `body` is ignored. Set it with [`Response::stream`].
    pub stream: Option<Streaming>,
}

/// Written by hand rather than derived, and the difference matters.
///
/// A derived `Default` would give `status: 0`, which is not an HTTP status, so
/// `Response { body: .., ..Default::default() }` would compile and be wrong.
/// 200 is what "I did not say" means for a response.
impl Default for Response {
    fn default() -> Self {
        Self {
            status: 200,
            body: String::new(),
            headers: Vec::new(),
            stream: None,
        }
    }
}

impl Response {
    pub fn new(status: u16, body: impl Into<String>) -> Self {
        Self {
            status,
            body: body.into(),
            ..Default::default()
        }
    }

    pub fn ok(body: impl Into<String>) -> Self {
        Self::new(200, body)
    }

    pub fn empty(status: u16) -> Self {
        Self::new(status, "")
    }

    /// A body written as it is produced: the first bytes reach the client
    /// before the last exist.
    ///
    /// The response hooks see the headers and never the body. The server
    /// tells the stream to end when it stops (`BodyStream::finish`), so a
    /// long-lived stream should implement that, or a deploy waits for it.
    #[allow(dead_code)]
    pub fn stream(status: u16, content_type: &str, stream: impl BodyStream) -> Self {
        Self {
            status,
            headers: vec![("content-type".to_string(), content_type.to_string())],
            stream: Some(Streaming::new(stream)),
            ..Default::default()
        }
    }

    /// Server-sent events: `text/event-stream`, and the two headers that stop
    /// a cache or a proxy from holding the events back.
    ///
    /// `brocanth_server::sse::channel` makes the stream from async code:
    ///
    /// ```ignore
    /// let (events, stream) = brocanth_server::sse::channel(16);
    /// tokio::spawn(async move {
    ///     // `Err` when the client left or the server is stopping.
    ///     let _ = events.send(Event::new("hello").with_event("greeting")).await;
    /// });
    /// Ok(Response::events(stream.heartbeat(Duration::from_secs(15))))
    /// ```
    #[allow(dead_code)]
    pub fn events(stream: impl BodyStream) -> Self {
        Self::stream(200, "text/event-stream", stream)
            .with_header("cache-control", "no-cache")
            .with_header("x-accel-buffering", "no")
    }

    /// For what the struct literal does not cover: several headers, added one
    /// at a time. Rust has no variadic generics, so a tuple return is
    /// fixed-arity and this is the unbounded form.
    #[must_use]
    pub fn with_header(mut self, name: impl Into<String>, value: impl Into<String>) -> Self {
        self.headers.push((name.into(), value.into()));
        self
    }

    /// Drops every header with this name, ignoring case.
    ///
    /// `with_header` appends, because `set-cookie` has to repeat. A
    /// single-valued header does not: two `content-type` lines are two
    /// claims and the client picks. Replacing one drops it first, which
    /// is what `set_header` below does.
    #[allow(dead_code)]
    #[must_use]
    pub fn without_header(mut self, name: &str) -> Self {
        self.headers
            .retain(|(held, _)| !held.eq_ignore_ascii_case(name));
        self
    }

    /// The one value for this header, replacing any already set.
    ///
    /// What to compose with when overriding a type an earlier step chose:
    /// `Ok(render_json(tpl)?.set_header("content-type", "application/problem+json"))`.
    #[allow(dead_code)]
    #[must_use]
    pub fn set_header(self, name: &str, value: impl Into<String>) -> Self {
        self.without_header(name).with_header(name, value)
    }
}

/// What a handler may return instead of building a [`Response`] by hand.
///
/// The Crystal brocanth overloads `write_response` on the handler's return
/// type, so a page returns the simplest thing that says what it means. Rust
/// spells the same idea as a trait: each handler still returns exactly one
/// type, and this is what lets different handlers return different ones.
pub trait IntoResponse {
    fn into_response(self) -> Response;
}

impl IntoResponse for Response {
    fn into_response(self) -> Response {
        self
    }
}

/// A body, answered 200.
impl IntoResponse for String {
    fn into_response(self) -> Response {
        Response::ok(self)
    }
}

impl IntoResponse for &str {
    fn into_response(self) -> Response {
        Response::ok(self)
    }
}

/// Nothing: 200 with no body. Crystal's `Nil` return.
impl IntoResponse for () {
    fn into_response(self) -> Response {
        Response::empty(200)
    }
}

/// A status and nothing else -- `Ok(404)`.
impl IntoResponse for u16 {
    fn into_response(self) -> Response {
        Response::empty(self)
    }
}

/// A body and a status.
impl<T: IntoResponse> IntoResponse for (T, u16) {
    fn into_response(self) -> Response {
        Response {
            status: self.1,
            ..self.0.into_response()
        }
    }
}

/// A body and headers.
impl<T: IntoResponse> IntoResponse for (T, Vec<(String, String)>) {
    fn into_response(self) -> Response {
        let mut response = self.0.into_response();
        response.headers.extend(self.1);
        response
    }
}

/// A body, a status and headers.
impl<T: IntoResponse> IntoResponse for (T, u16, Vec<(String, String)>) {
    fn into_response(self) -> Response {
        let mut response = self.0.into_response();
        response.status = self.1;
        response.headers.extend(self.2);
        response
    }
}

/// What a handler may fail with.
///
/// Deliberately **not** an `impl std::error::Error`, which is what lets the
/// blanket `From` below exist without colliding with the reflexive
/// `impl<T> From<T> for T`. `anyhow::Error` is this shape for the same reason.
/// The effect is that `?` works on anything in a handler.
#[derive(Debug)]
pub struct Error {
    pub status: u16,
    pub message: String,
    /// Headers the failure needs. A `content-type` for a body that is not
    /// HTML, a `www-authenticate` for a 401, a `retry-after` for a 429.
    ///
    /// `Response` always had these and `Error` did not, which made a
    /// machine-readable error body impossible to send.
    pub headers: Vec<(String, String)>,
}

impl Error {
    pub fn new(status: u16, message: impl Into<String>) -> Self {
        Self {
            status,
            message: message.into(),
            headers: Vec::new(),
        }
    }

    /// The default for a failure nobody classified: it is a bug here, not a
    /// mistake by the caller.
    pub fn internal(message: impl Into<String>) -> Self {
        Self::new(500, message)
    }

    /// The same, carrying a header.
    ///
    /// `allow(dead_code)` for the same reason the request readers below
    /// have it: it is here for the first handler that needs it, and a
    /// brand-new project has none yet.
    #[allow(dead_code)]
    #[must_use]
    pub fn with_header(mut self, name: impl Into<String>, value: impl Into<String>) -> Self {
        self.headers.push((name.into(), value.into()));
        self
    }
}

impl<E: std::error::Error + Send + Sync + 'static> From<E> for Error {
    fn from(error: E) -> Self {
        Self::internal(error.to_string())
    }
}

impl IntoResponse for Error {
    fn into_response(self) -> Response {
        // Everything a client sees for a 5xx is the status. The message is for
        // whoever is reading the logs, and putting it in the body is how an
        // internal path ends up on a stranger's screen.
        let withheld = self.status >= 500;
        if withheld {
            // So the message does reach the logs: a database error that came
            // up through `?` says here what failed, and the client sees none
            // of it.
            let (status, message) = (self.status, &self.message);
            crate::plugins::say!(error, "error {status}: {message}"; status, %message);
        }
        let body = if withheld {
            String::from("Internal Server Error")
        } else {
            self.message
        };
        let mut headers = self.headers;
        if withheld {
            // The body is replaced, so a `content-type` describing the
            // original no longer describes anything. Keeping it labelled a
            // plain sentence `application/problem+json` -- harmless while
            // `sent` sent two types and the client could pick the other,
            // and wrong now that the replacement is exact.
            headers.retain(|(name, _)| !name.eq_ignore_ascii_case("content-type"));
        }
        Response {
            status: self.status,
            body,
            // The rest survive: the body is withheld, not the framing the
            // client needs to read what is left.
            headers,
            stream: None,
        }
    }
}

/// What a handler is given. Middleware sees it too, and may change it before
/// the handler runs or after it returns.
pub struct Ctx {
    /// Which verb matched. A layer reads it; a page usually does not,
    /// because the verb is already in the name of the function it is in.
    #[allow(dead_code)]
    pub verb: Verb,
    pub path: String,
    /// Everything after the first `?`, without it. Empty when there was none.
    #[allow(dead_code)]
    pub query: String,
    /// The body as it arrived. Empty for the verbs that carry none, and for
    /// a context built by a test that did not set one.
    #[allow(dead_code)]
    pub body: Vec<u8>,
    params: Vec<(String, String)>,
    headers: Vec<(String, String)>,
    /// What the plugins put here for this request, or `None` for a context
    /// built by hand.
    plugins: Option<Context<plugins::Installed>>,
    /// What the request hooks wrote for this request: a `CsrfToken`, a
    /// client address. Empty for a context built by hand.
    state: RequestState,
}

impl Ctx {
    pub fn new(verb: Verb, path: &str, params: Params<'_>) -> Self {
        Self {
            verb,
            path: path.to_string(),
            query: String::new(),
            body: Vec::new(),
            params: params
                .iter()
                .map(|(name, value)| (name.to_string(), value.to_string()))
                .collect(),
            headers: Vec::new(),
            plugins: None,
            state: RequestState::new(),
        }
    }

    /// The rest of the request: what the client sent, rather than what the
    /// router matched.
    ///
    /// Two calls and not one, because a test that exercises a handler
    /// usually wants the first and not the second. `src/app.rs` makes both.
    ///
    /// The three are copied out of the server's request, which lives for
    /// the connection and not for the handler. One copy per request, and
    /// the handler owns what it reads.
    #[must_use]
    pub fn sent(mut self, query: &str, body: Vec<u8>, headers: Vec<(String, String)>) -> Self {
        self.query = query.to_string();
        self.body = body;
        self.headers = headers;
        self
    }

    /// What the plugins put on this request's context.
    ///
    /// `src/app.rs` calls it once per request, with a context the
    /// application made from its providers. A `Ctx` built in a test has
    /// none, which is what the `Option` says.
    #[must_use]
    pub fn with_plugins(mut self, plugins: Context<plugins::Installed>) -> Self {
        self.plugins = Some(plugins);
        self
    }

    /// What the request hooks wrote. `src/app.rs` hands it over before the
    /// page runs and takes it back after, for the response hooks.
    #[must_use]
    pub fn with_state(mut self, state: RequestState) -> Self {
        self.state = state;
        self
    }

    /// The hook state back, leaving this context an empty one.
    pub fn take_state(&mut self) -> RequestState {
        std::mem::take(&mut self.state)
    }

    /// The value a plugin published for this request.
    ///
    /// **Compiles only when some registered plugin provides `T`.** Ask for
    /// a type nothing installs and the build stops and names it; there is
    /// no string key to misspell and no guard to forget. That check is the
    /// whole point, and it is the compiler's, not this method's.
    ///
    /// `Option` is the other half, and it is not about `T`'s type. It is
    /// `None` in two cases: a `Ctx` built by hand in a test, which has no
    /// application behind it and so no providers; and a provider that
    /// declared `T` and inserted nothing, which `Plugin::provide` allows
    /// because it is one method and `Provides` is a separate declaration.
    /// A dispatched request whose provider wrote its value gets `Some`.
    ///
    /// The `_` is the position of `T` in the provider list, which is
    /// always inferred. It is there so two impls do not overlap, and it is
    /// the same `_` `Context::get` takes.
    ///
    /// ```ignore
    /// let session = ctx.provided::<Session, _>().expect("dispatched");
    /// ```
    #[allow(dead_code)]
    pub fn provided<T, I>(&self) -> Option<&T>
    where
        T: ContextItem,
        plugins::Installed: Has<T, I>,
    {
        // `and_then` with `try_get`, not `map` with `get`: `get` panics
        // when the provider list claims `T` and the provider inserted
        // nothing, which the generated plugin templates make easy --
        // `Provides` is unconditional and `provide` is often
        // `if let Some(..)`. A method returning `Option` must not abort
        // the process to say no.
        self.plugins.as_ref().and_then(|cx| cx.try_get::<T>())
    }

    /// A route parameter, or `None`.
    ///
    /// Never `""`. The Crystal prototype returned an empty string for a name
    /// nobody declared, so a typo read as a present-but-blank value and went
    /// unnoticed; `Option` is that bug deleted rather than documented.
    pub fn param(&self, name: &str) -> Option<&str> {
        self.params
            .iter()
            .find(|(key, _)| key == name)
            .map(|(_, value)| value.as_str())
    }

    /// The same, for a parameter the route declares and so must be there.
    ///
    /// Returns `Result` rather than `Option` so it composes with `?`. A name
    /// the route does not declare is this file disagreeing with its own
    /// filename, which is a 500 and not a 400: nobody sent a bad request.
    pub fn require(&self, name: &str) -> Result<&str, Error> {
        self.param(name)
            .ok_or_else(|| Error::internal(format!("the route declares no `{name}` parameter")))
    }
}

/// What the client sent, as opposed to what the router matched.
///
/// A second block, and `allow(dead_code)`, for one reason: the request is
/// on the context whether or not this project reads it yet. The other way
/// round is a context whose shape depends on which modules were chosen,
/// and then a page that starts reading a header needs the generator rather
/// than a line.
#[allow(dead_code)]
impl Ctx {
    /// A value a request hook wrote for this request, by its type.
    ///
    /// `ctx.hooked::<brocanth_middleware::CsrfToken>()` is the token a form
    /// carries. `None` when no registered hook wrote one, and always for a
    /// context built by hand.
    pub fn hooked<T: Send + Sync + 'static>(&self) -> Option<&T> {
        self.state.get::<T>()
    }

    /// The same value, to change. `None` when no hook wrote one.
    ///
    /// The state goes back to the hooks when the page returns, so a change
    /// made here is what `after_dispatch` and the response hooks read: a
    /// request hook puts a per-request log record on the state, and the
    /// page fills in the fields only it knows.
    pub fn hooked_mut<T: Send + Sync + 'static>(&mut self) -> Option<&mut T> {
        self.state.get_mut::<T>()
    }

    /// Puts a value on the request state for the hooks that run after the
    /// page, and returns the value it replaced.
    ///
    /// For a value that no request hook wrote first. A hook that reads it
    /// must also accept `None`: a 404, a 405 and a request that a request
    /// hook answered run no page.
    pub fn hook_insert<T: Send + Sync + 'static>(&mut self, value: T) -> Option<T> {
        self.state.insert(value)
    }

    /// A request header by name, case-insensitively, or `None`.
    pub fn header(&self, name: &str) -> Option<&str> {
        self.headers
            .iter()
            .find(|(key, _)| key.eq_ignore_ascii_case(name))
            .map(|(_, value)| value.as_str())
    }

    /// Every request header, in the order they arrived.
    pub fn headers(&self) -> impl Iterator<Item = (&str, &str)> {
        self.headers
            .iter()
            .map(|(key, value)| (key.as_str(), value.as_str()))
    }

    /// What the body is written in, or `None` when the client did not say.
    pub fn content_type(&self) -> Option<&str> {
        self.header("content-type")
    }

    /// The body as text, or `None` when it is not UTF-8.
    ///
    /// `Option` and not a lossy conversion: a body that is not text is a
    /// request to answer, not a string to repair.
    pub fn body_text(&self) -> Option<&str> {
        core::str::from_utf8(&self.body).ok()
    }
}

/// A handler, as the table stores it. One per verb a page answers.
///
/// Boxed rather than a bare `fn` pointer, because every handler returns its
/// own type and a `fn` pointer can only hold one of them. [`page`] and
/// [`page_async`] are what unify them, and they run once per route at
/// startup rather than per request.
///
/// It returns a future, so a page may wait -- on a database, on another
/// service -- without holding a thread. A page that waits on nothing is
/// still a plain `fn`, and [`page`] hands back its answer already finished.
pub type Handler =
    Box<dyn for<'a> Fn(&'a mut Ctx) -> BoxFuture<'a, Result<Response, Error>> + Send + Sync>;

/// Puts a `pub fn handle` in the table whatever it returns.
///
/// Generic over the return type, so `-> Result<impl IntoResponse, Error>` on
/// the handler and one concrete `Handler` in the table are the same thing.
/// The answer is ready when the handler returns; the future around it is
/// only there so a sync page and an async page share one table.
pub fn page<H, T>(handler: H) -> Handler
where
    H: Fn(&mut Ctx) -> Result<T, Error> + Send + Sync + 'static,
    T: IntoResponse,
{
    Box::new(move |ctx| {
        let answered = handler(ctx).map(IntoResponse::into_response);
        Box::pin(std::future::ready(answered))
    })
}

/// Puts a `pub async fn handle` in the table.
///
/// `pages!` writes the call, with a closure that awaits the page and turns
/// its answer into a `Response`. The closure is a `fn` pointer and not a
/// generic, because the page's future borrows the context and a generic
/// cannot name that borrow.
///
/// `allow(dead_code)`: nothing calls it until a page is `async`.
#[allow(dead_code)]
pub fn page_async(
    handler: for<'a> fn(&'a mut Ctx) -> BoxFuture<'a, Result<Response, Error>>,
) -> Handler {
    Box::new(handler)
}

/// Every route this project did not read off `src/pages`.
///
/// One table holds all of them. A module contributes routes by offering a
/// `mount` and the application decides the prefix, which is the same rule
/// the core states for a plugin: a path is a decision, so a module that
/// mounted itself would be making one on the application's behalf.
///
/// A prefix two sources both claim is a `BuildError` naming the pattern,
/// rather than whichever of them happened to run last.
fn mounted(
    routes: &mut ::brocanth::routing::RouterBuilder<Handler>,
) -> Result<(), ::brocanth::routing::BuildError> {
    crate::plugins::admin::mount(routes, crate::modules::admin::MODULE.mount_at)?;
    Ok(())
}

/// The database, on every request the application dispatches.
///
/// `allow(dead_code)`: it is here for the first page that queries, and a
/// new project has none.
#[allow(dead_code)]
impl Ctx {
    /// The pool the `db` plugin opened. Each statement takes a connection
    /// and gives it back, so a page holds none while it waits on anything
    /// else.
    ///
    /// ```ignore
    /// use brocanth_db::Exec;
    /// let rows: Vec<(i64, String)> = ctx.db().all(&query).await?;
    /// ```
    ///
    /// A `brocanth_db::DbError` goes through `?`: it is a 500, its message
    /// goes to the log and not to the client.
    ///
    /// # Panics
    ///
    /// On a `Ctx` built by hand, which has no application behind it.
    pub fn db(&self) -> &brocanth_db::Db {
        self.provided::<brocanth_db::Db, _>().expect(
            "this Ctx has no database: it was built by hand, not dispatched by the application\n\
             help: send the request through `app::dispatch`, or give the code under test a \
             `brocanth_db::Db` of its own",
        )
    }
}

//! `/` -- because the file is called `index.rs`.

use crate::pages::{Ctx, Error, IntoResponse};

/// GET. The framework reads the verb off the function name: `handle` is GET,
/// `handle_post` is POST, and `handle_json` would be a content-negotiation hook
/// rather than a verb.
///

//! `/about` -- one file, one route.

use crate::pages::{Ctx, Error, IntoResponse, Response};

pub fn handle(ctx: &mut Ctx) -> Result<impl IntoResponse, Error> {
    // The context is the request. The framework owns the response, so a
    // handler returns a value and never writes one itself -- which is what
    // makes a half-written body on an error impossible rather than unlikely.
    //
    // The struct literal is the form that scales: name the fields that differ
    // and `..Default::default()` fills the rest. Rust will not let you omit a
    // field without it, and there are no implicit zero values.
    Ok(Response {
        body: format!(
            "<!doctype html><title>About \u{b7} My Blog</title>\
             <h1>About</h1>\
             <p>You asked for <code>{} {}</code>. This page answers GET and \
             nothing else, so a DELETE to it is a 405 carrying an \
             <code>Allow</code> header rather than a 404.</p>",
            ctx.verb, ctx.path
        ),
        ..Default::default()
    })
}

//! `/users/{id}` -- the brackets in the file name are the parameter.
//!
//! There is no ceiling on how many a route may have. The Crystal prototype had
//! `MAX_PARAMS = 4` and dropped the fifth without saying so, because its
//! collector could not prove a captured parameter does not outlive the request.
//! Here the parameters borrow the path, so the question does not arise.

use crate::pages::{routes, Ctx, Error, IntoResponse, Response};

pub fn handle(ctx: &mut Ctx) -> Result<impl IntoResponse, Error> {
    // `?` is why a handler returns `Result`. Anything that fails converts
    // into `Error` through a blanket `From`, so a parse, a query or a file
    // read is one character rather than a match arm.
    let id: u64 = ctx.require("id")?.parse()?;

    Ok(Response {
        body: format!("<!doctype html><title>User {id}</title><h1>User {id}</h1>"),
        ..Default::default()
    })
}

/// POST to the same path. One file, one route, one handler per verb.
///
/// A redirect is a status, one header and no body. The struct literal above is
/// the other form; this one stays readable when there are several headers,
/// because Rust has no variadic generics and a tuple return is therefore
/// fixed-arity.
///
/// The location is built rather than written. `routes` is generated from the
/// directory alongside the table, so a link to a route that does not exist
/// does not compile, and neither does one missing a parameter.
pub fn handle_post(ctx: &mut Ctx) -> Result<impl IntoResponse, Error> {
    let id: u64 = ctx.require("id")?.parse()?;

    Ok(Response::empty(303).with_header("location", routes::users::id::path(id)))
}

/// Every plugin this application contains, in dependency order.
///
/// A macro and not a function, and the reason is the builder's type:
/// registering a plugin that provides a context item widens
/// `AppBuilder`'s type-level provider list, so the type this chain ends
/// on is one nested associated type per plugin. A function would have to
/// spell that type out and re-spell it on every change. The macro expands
/// at its one call site in `src/app.rs`, where the compiler works it out
/// and nobody has to read it.
///
/// Each registration is still one line, and a line a reader can delete.
macro_rules! register {
    ($builder:expr) => {{
        use brocanth_config::AppBuilderConfigExt;

        // The files, the environment and, where src/modules/ turns them
        // on, the flags: `brocanth-config` layers them and types each
        // section, and it is the only reason a TOML parser is in the tree.
        let config = crate::plugins::config()?;
        let builder = $builder.with_config(config);
        let builder = builder
            .plugin_from_config::<crate::plugins::admin::AdminSettings>("admin")
            .map_err(|e| e.to_string())?;
        let builder = builder
            .plugin_from_config::<crate::plugins::db::Database>("db")
            .map_err(|e| e.to_string())?;
        let builder = builder
            .plugin_from_config::<crate::plugins::auth::Auth>("auth")
            .map_err(|e| e.to_string())?;
        let builder = builder
            .plugin_from_config::<crate::plugins::cache::Cache>("cache")
            .map_err(|e| e.to_string())?;
        let builder = builder
            .plugin_from_config::<crate::plugins::superuser::Superuser>("superuser")
            .map_err(|e| e.to_string())?;
        builder
    }};
}
pub(crate) use register;

/// What this binary puts on the context, said once at startup.
///
/// The lines are a demonstration and they are load-bearing: the installed
/// provider set lives in the application's *type*, so an `extract` for a
/// plugin nobody registered does not compile. That is the modularity claim,
/// checked by the compiler rather than written in a README.
///
/// A handler reads the same values through `Ctx::provided`, which carries
/// the same check. This runs at startup so a misconfiguration is a failure
/// to start rather than a failure on the first request that needed it.
///
/// A macro for the reason `register!` is one: the context's type is the
/// builder's type, and neither is worth spelling.
macro_rules! announce {
    ($app:expr) => {{
        let cx = $app.context();
        let db: &brocanth_db::Db = cx.extract();
        let connections = db.pool().size();
        crate::plugins::say!(info, "db: {connections} connection(s) open"; connections);
        let auth: &crate::plugins::auth::Services = cx.extract();
        let cookie = &auth.sessions().config().cookie_name;
        let providers = auth.providers().len();
        crate::plugins::say!(
            info,
            "auth: session cookie {cookie}, {providers} provider(s)";
            %cookie,
            providers
        );
    }};
}
pub(crate) use announce;

/// A line about the application itself: startup, the listener, the stop.
///
/// A line on stderr, prefixed with the application name. The level and the
/// fields after `;` are what `observability-logs` would record; without it
/// there is no subscriber, so they are dropped and the message is printed.
///
/// ```ignore
/// plugins::say!(info, "listening on http://{bound}"; %bound);
/// ```
macro_rules! say {
    ($level:ident, $message:literal $(; $($field:tt)+)?) => {
        eprintln!("{}: {}", crate::app::APP, format_args!($message))
    };
}
pub(crate) use say;

///
/// A layer may wait, so it returns a future. The lifetime is named
/// because the future borrows both the context and the rest of the

/// One value this run was given, above the defaults.
pub struct Given {
    /// What it sets.
    pub setting: &'static Setting,
    /// Where it came from, as typed: `--cookie`, or `$MY_APP_SESSIONS__COOKIE`.
    pub from: String,
    /// The value, as typed. Empty is `None` for an `Option` setting.
    pub value: String,
}

/// The name each flag answers to after the renames in src/modules/, whether
/// the flags are read or not.
///
/// A name that is not spelled as a flag, and two flags under one name, each
/// stop the binary: each would otherwise be a flag that does nothing, or does
/// something else. A rename of a key no flag sets does not compile: `Key` has
/// no variant for it.
fn names() -> Result<Vec<(&'static Setting, &'static str)>, String> {
    let renames = renames();
    let mut names: Vec<(&'static Setting, &'static str)> = Vec::new();
    for setting in SETTINGS.iter().filter(|setting| setting.flag) {
        let name = renames
            .iter()
            .find(|(path, _)| *path == setting.path)
            .map_or(setting.path, |(_, flag)| flag.name());
        if name.is_empty() {
            continue;
        }
        let spelled = name
            .chars()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || "._-".contains(c));
        let taken = ["help", "version", "settings"].contains(&name);
        if !spelled || name.starts_with(['-', '.', '_']) || taken {
            return Err(format!(
                "{} names a flag `{name}`\nhelp: a flag is lower case, digits, `.`, `_` and `-`, starts with a letter or a digit, and is not `help`, `version` or `settings`",
                setting.file
            ));
        }
        if let Some((other, _)) = names.iter().find(|(_, taken)| *taken == name) {
            let files = if other.file == setting.file {
                other.file.to_string()
            } else {
                format!("{} or {}", other.file, setting.file)
            };
            return Err(format!(
                "`--{name}` would set both `{}` and `{}`\nhelp: rename one of them in {files}",
                other.path, setting.path
            ));
        }
        names.push((setting, name));
    }
    Ok(names)
}

/// Every rename in src/modules/, as the setting's path and its flag.
fn renames() -> Vec<(&'static str, Flag)> {
    let mut renames = Vec::new();
    for (key, flag) in crate::modules::brocanth::MODULE.rename {
        renames.push((key.path(), *flag));
    }
    for (key, flag) in crate::modules::db::MODULE.rename {
        renames.push((key.path(), *flag));
    }
    for (key, flag) in crate::modules::cache::MODULE.rename {
        renames.push((key.path(), *flag));
    }
    for (key, flag) in crate::modules::admin::MODULE.rename {
        renames.push((key.path(), *flag));
    }
    for (key, flag) in crate::modules::auth::MODULE.rename {
        renames.push((key.path(), *flag));
    }
    for (key, flag) in crate::modules::superuser::MODULE.rename {
        renames.push((key.path(), *flag));
    }
    renames
}

/// Whether a module's flags are read: its file says `Flags::On` or
/// `Flags::Off`, or follows the project rule. With a config file, the file is
/// the source until src/modules/brocanth.rs turns the flags on.
fn flags_on(rule: Flags) -> bool {
    match rule {
        Flags::On => true,
        Flags::Off => false,
        Flags::Project => crate::modules::brocanth::MODULE.flags_with_config,
    }
}

/// The flags this run reads: each module file decides, and
/// src/modules/brocanth.rs holds the project rule. Every name is checked
/// first, read or not.
pub fn flag_names() -> Result<Vec<(&'static Setting, &'static str)>, String> {
    Ok(names()?
        .into_iter()
        .filter(|(setting, _)| flags_on(setting.flags))
        .collect())
}

/// A flag given while the flags are off, refused by name with what to do
/// instead. Silently ignoring it would run with a value nobody chose.
pub fn refuse_off_flags() -> Result<(), String> {
    let names = names()?;
    for arg in std::env::args_os().skip(1) {
        let arg = arg.to_string_lossy();
        if arg == "--" {
            break;
        }
        let Some(written) = arg.strip_prefix("--") else {
            continue;
        };
        let name = written.split_once('=').map_or(written, |(name, _)| name);
        let Some((setting, _)) = names.iter().find(|(_, known)| *known == name) else {
            continue;
        };
        if flags_on(setting.flags) {
            continue;
        }
        let (section, key) = setting.path.rsplit_once('.').unwrap_or(("", setting.path));
        let fix = if setting.file.ends_with("/brocanth.rs") {
            "flags_with_config: true"
        } else {
            "flags: Flags::On"
        };
        return Err(format!(
            "`--{name}` is off: this project reads its settings from config/<env>.toml\nhelp: set `{fix}` in {}, or set `{key}` under [{section}] in config/<env>.toml",
            setting.file
        ));
    }
    Ok(())
}

/// The flags this run was given, in the order typed. The environment is not
/// here: `brocanth-config` reads it, under the file and over the flags.
///
/// `--name=VALUE` and `--name VALUE`. Every other argument is left alone: it
/// is a command's, and the command line's own parser has already refused what
/// nobody declared.
pub fn given() -> Result<Vec<Given>, String> {
    let mut given = Vec::new();
    let names = flag_names()?;
    let args: Vec<String> = std::env::args_os()
        .skip(1)
        .map(|arg| arg.to_string_lossy().into_owned())
        .collect();
    let mut rest = args.iter();
    while let Some(arg) = rest.next() {
        if arg == "--" {
            break;
        }
        let Some(written) = arg.strip_prefix("--") else {
            continue;
        };
        let (name, inline) = match written.split_once('=') {
            Some((name, value)) => (name, Some(value.to_string())),
            None => (written, None),
        };
        let Some((setting, name)) = names.iter().copied().find(|(_, known)| *known == name) else {
            continue;
        };
        let value = match inline {
            Some(value) => value,
            None => rest
                .next()
                .cloned()
                .ok_or_else(|| format!("--{name} needs a value\nhelp: write --{name}=VALUE"))?,
        };
        given.push(Given {
            setting,
            from: format!("--{name}"),
            value,
        });
    }
    Ok(given)
}

/// The configuration this run reads: the files, the environment, and the
/// flags on top where src/modules/ turns them on.
pub fn config() -> Result<brocanth_config::Config, String> {
    refuse_off_flags()?;
    let mut loader = brocanth_config::Loader::new(crate::app::APP);
    for setting in SETTINGS {
        if let Some(alias) = setting.none_as {
            loader = loader.none_as(setting.path, alias);
        }
    }
    for given in given()? {
        let flag = given.from.trim_start_matches('-');
        loader = loader.flag(given.setting.path, flag, &given.value, given.setting.text);
    }
    loader.load().map_err(|e| e.to_string())
}

/// The flags, as arguments every command takes.
///
/// A name a command already takes is refused here: `clap` panics on it in a
/// debug build and lets one of the two win in a release one.
pub fn flag_args(
    command: brocanth_command::clap::Command,
) -> brocanth_command::anyhow::Result<brocanth_command::clap::Command> {
    use brocanth_command::{anyhow, clap};

    refuse_off_flags().map_err(anyhow::Error::msg)?;
    let mut command = command;
    for (setting, name) in flag_names().map_err(anyhow::Error::msg)? {
        let taken = command
            .get_subcommands()
            .flat_map(|sub| sub.get_arguments())
            .any(|arg| arg.get_long() == Some(name));
        if taken {
            anyhow::bail!(
                "`--{name}` sets `{}` and is also an argument of a command\nhelp: rename it in {}",
                setting.path,
                setting.file
            );
        }
        command = command.arg(
            clap::Arg::new(name)
                .long(name)
                .value_name("VALUE")
                .help(setting.help)
                .global(true)
                .action(clap::ArgAction::Append),
        );
    }
    Ok(command)
}

/// Every settings layer on its own, lowest first, then the merged values.
/// A secret is masked in every layer.
pub fn settings_report() -> Result<String, String> {
    let config = config()?;
    let mut layers = config.layers().defaults(defaults());
    for setting in SETTINGS.iter().filter(|setting| !setting.flag) {
        layers = layers.secret(setting.path);
    }
    Ok(layers.to_string())
}

/// `[server]`: the settings the framework reads itself. It declares them as

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ServerSettings {

    pub shutdown_grace_secs: u64,
}

impl Default for ServerSettings {
    fn default() -> Self {
        Self {
            shutdown_grace_secs: crate::app::SHUTDOWN_GRACE.as_secs(),
        }
    }

/// `[server] shutdown_grace_secs` from the files, the environment and the
/// flags, or [`SHUTDOWN_GRACE`](crate::app::SHUTDOWN_GRACE) when none sets it.
pub fn shutdown_grace() -> Result<std::time::Duration, String> {
    let config = config()?;
    let Some(value) = config
        .raw_section("server")
        .and_then(|server| server.get("shutdown_grace_secs"))
    else {
        return Ok(crate::app::SHUTDOWN_GRACE);
    };
    // An integer in a file, a flag or a variable; a string when a layer
    // could not type it.
    value
        .as_integer()
        .and_then(|secs| u64::try_from(secs).ok())
        .or_else(|| value.as_str().and_then(|secs| secs.trim().parse().ok()))
        .map(std::time::Duration::from_secs)
        .ok_or_else(|| {
            format!(
                "`[server] shutdown_grace_secs` is {}, and it must be a whole number of seconds\nhelp: set it to a whole number of seconds, such as 25",
                value.display()
            )
        })
}

/// What the server is handed for one request.
pub type Answer = brocanth::plugin_routes::BoxFuture<'static, brocanth_server::Response>;

/// What `serve` answers a request with: `next`, as it is. `brocanth add
/// observe` logs and counts each request here, and answers the probes.
pub fn wrap<F, Fut>(
    app: &std::sync::Arc<crate::app::App>,
    next: F,
) -> impl Fn(brocanth_server::Request) -> Answer + Send + Sync + 'static
where
    F: Fn(brocanth_server::Request) -> Fut + Send + Sync + 'static,
    Fut: std::future::Future<Output = brocanth_server::Response> + Send + 'static,
{
    let _ = app;
    move |request| -> Answer { Box::pin(next(request)) }
}

    let _ = app;
    Ok(())
}

    stopper.stop();
}

//! `admin eject` and `admin diff` -- shipped admin screens, copied into
//! this project and compared with the shipped ones.
//!
//! The work is the `brocanth` CLI's. This command runs
//! `brocanth admin eject <screen> [--model <key>]` or
//! `brocanth admin diff [<screen>] [--model <key>]` in the project root and
//! exits with its status, so `cargo run -- admin eject list` and
//! `brocanth admin eject list` do the same thing. The application does not
//! link the code that writes source.
//!
//! This file is yours. See `docs/admin-customization.md`, level 8.

use std::ffi::OsStr;
use std::path::Path;
use std::process;

use brocanth_command::{anyhow, clap, cmd, Command, Context};

/// Work on the admin: `eject` copies a shipped screen into this project,
/// `diff` compares the copies with the shipped screens.
#[cmd]
#[derive(clap::Args)]
pub struct Admin {
    #[command(subcommand)]
    pub action: Action,
}

#[derive(clap::Subcommand)]
pub enum Action {
    /// Copy a shipped screen into this project and install it over the
    /// default. `brocanth admin eject --help` lists the screens.
    Eject {
        /// The screen to copy, such as `list` or `layout`.
        screen: String,
        /// Install the copy for this model only. list, detail, form and
        /// delete take one.
        #[arg(long, value_name = "KEY")]
        model: Option<String>,
    },
    /// Compare ejected screens with the shipped ones of the installed
    /// brocanth-admin. Run it after an upgrade.
    Diff {
        /// The screen to compare. Without one, every ejected copy.
        screen: Option<String>,
        /// Compare the copy for this model only.
        #[arg(long, value_name = "KEY", requires = "screen")]
        model: Option<String>,
    },
}

impl Command for Admin {
    fn run(self, ctx: &mut Context) -> anyhow::Result<()> {
        let status = brocanth(OsStr::new("brocanth"), ctx.root(), &self.action.args())?;
        if !status.success() {
            // `brocanth` printed the error and its help line.
            process::exit(status.code().unwrap_or(1));
        }
        Ok(())
    }
}

impl Action {
    /// The `brocanth` arguments this action stands for.
    fn args(&self) -> Vec<String> {
        match self {
            Action::Eject { screen, model } => {
                let mut args = vec!["admin".to_string(), "eject".to_string(), screen.clone()];
                if let Some(model) = model {
                    args.extend(["--model".to_string(), model.clone()]);
                }
                args
            }
            Action::Diff { screen, model } => {
                let mut args = vec!["admin".to_string(), "diff".to_string()];
                args.extend(screen.clone());
                if let Some(model) = model {
                    args.extend(["--model".to_string(), model.clone()]);
                }
                args
            }
        }
    }
}

/// Runs `program` with `args` in `root`. A program that is not installed
/// is an error that says how to install it.
fn brocanth(program: &OsStr, root: &Path, args: &[String]) -> anyhow::Result<process::ExitStatus> {
    let name = program.to_string_lossy();
    process::Command::new(program)
        .args(args)
        .current_dir(root)
        .status()
        .map_err(|e| match e.kind() {
            std::io::ErrorKind::NotFound => anyhow::anyhow!(
                "`{name}` is not installed or not on PATH\n\
                 help: install the brocanth CLI: `cargo install --git \
                 https://github.com/DHHwasRight/brocanth brocanth-cli`, or \
                 `cargo install --path crates/brocanth-cli` in a brocanth checkout"
            ),
            _ => anyhow::anyhow!("could not run `{name}`: {e}"),
        })
}

//! `createsuperuser` -- an operator account that passes every check of the
//! admin.
//!
//! Opens the accounts at `[auth] url` and makes the account with the
//! address a superuser. It creates the account first when there is none,
//! with the password in the variable that `[superuser] password_env` names.
//! An account that exists keeps its password. Run it twice and nothing
//! changes.
//!
//! Ordinary project code, like `serve`: rename it or delete it, and delete
//! its line in `src/commands/mod.rs`.

use std::sync::Arc;

use brocanth_auth::{
    ensure_superuser, Accounts, Passwords, PgStore, SuperuserOutcome, SystemClock,
};
use brocanth_command::{anyhow, clap, cmd, Command, Context};

/// Make the account with this address a superuser, creating it if needed
#[cmd]
#[derive(clap::Args)]
pub struct Createsuperuser {
    /// The account's address
    #[arg(long, value_name = "ADDRESS")]
    email: String,
}

impl Command for Createsuperuser {
    fn run(self, ctx: &mut Context) -> anyhow::Result<()> {
        std::env::set_current_dir(ctx.root())?;
        let (auth, superuser) = settings()?;
        let name = superuser.password_env;
        let password = std::env::var(&name).unwrap_or_default();
        let runtime = tokio::runtime::Builder::new_current_thread()
            .enable_all()
            .build()?;
        let outcome = runtime.block_on(async {
            let store = Arc::new(PgStore::connect(&auth.url).await?);
            store.migrate().await?;
            let accounts = Accounts::new(
                Arc::clone(&store),
                Arc::new(SystemClock),
                Arc::new(Passwords::recommended()?),
            )?;
            anyhow::Ok(ensure_superuser(&accounts, store.as_ref(), &self.email, &password).await)
        })?;
        match outcome {
            Ok(SuperuserOutcome::Created(_)) => println!("created superuser {}", self.email),
            Ok(SuperuserOutcome::Promoted(_)) => println!("{} is now a superuser", self.email),
            Ok(SuperuserOutcome::Unchanged(_)) => println!("{} is a superuser already", self.email),
            // A new account with no password: say where it goes.
            Err(e) if password.is_empty() => anyhow::bail!(
                "{e}\nhelp: a new account needs a password; export {name}=<a long passphrase>"
            ),
            Err(e) => return Err(e.into()),
        }
        Ok(())
    }
}

    Ok((loaded.section("auth")?, loaded.section("superuser")?))

===== src/commands/migrate.rs =====
//! `migrate` -- apply the SQL files in `migrations/` to `[db] url`.
//!
//! A migration is one file, `migrations/<UTC time>_<name>.sql`. The
//! statements before a `-- down` line apply it; the statements after that
//! line undo it. `migrate new <name>` writes an empty one. Each file runs in
//! one transaction, and `brocanth_migrations` records it.
//!
//! The files are read when this command runs, not built into the binary.
//! The container image copies `migrations/` next to the binary, so
//! `<binary> migrate up` works in a release job. See `brocanth_db::migrate`.

use std::path::Path;

use brocanth_command::{anyhow, clap, cmd, Command, Context};
use brocanth_db::migrate::{self, Migrator};
use brocanth_db::{Db, DbConfig};

/// Apply, roll back or list the SQL migrations in migrations/
#[cmd]
#[derive(clap::Args)]
pub struct Migrate {
    #[command(subcommand)]
    action: Action,
}

#[derive(clap::Subcommand)]
enum Action {
    /// Apply each pending migration, oldest first
    Up,
    /// Roll back the newest N applied migrations
    Down {
        /// How many migrations to roll back
        #[arg(value_name = "N", default_value_t = 1)]
        n: usize,
    },
    /// List each migration as applied or pending
    Status,
    /// Write an empty migration, migrations/<UTC time>_<NAME>.sql
    New {
        /// The name of the migration, in snake case: add_posts
        #[arg(value_name = "NAME")]
        name: String,
    },
}

impl Command for Migrate {
    fn run(self, ctx: &mut Context) -> anyhow::Result<()> {
        // `migrations/` and the config files resolve from the project
        // root, wherever the command started.
        std::env::set_current_dir(ctx.root())?;
        let dir = Path::new("migrations");
        if let Action::New { name } = &self.action {
            let path = migrate::new_file(dir, name)?;
            println!("wrote {}", path.display());
            return Ok(());
        }
        let files = migrate::read_dir(dir)?;
        let config = database()?;
        let runtime = tokio::runtime::Builder::new_current_thread()
            .enable_all()
            .build()?;
        runtime.block_on(async {
            let db = Db::connect(&config).await?;
            let migrator = Migrator::new(&db, files);
            let result = apply(&migrator, self.action).await;
            db.close().await;
            result
        })
    }
}

async fn apply(migrator: &Migrator<'_>, action: Action) -> anyhow::Result<()> {
    match action {
        Action::Up => {
            let applied = migrator.up().await?;
            if applied.is_empty() {
                println!("Nothing to apply: each migration is applied.");
            }
            for name in applied {
                println!("applied      {name}");
            }
        }
        Action::Down { n } => {
            let undone = migrator.down(n).await?;
            if undone.is_empty() {
                println!("Nothing to roll back: no migration is applied.");
            }
            for name in undone {
                println!("rolled back  {name}");
            }
        }
        Action::Status => {
            let states = migrator.status().await?;
            if states.is_empty() {
                println!("No migration in migrations/. Write one with `migrate new <name>`.");
            }
            for (name, state) in states {
                println!("{:<12} {name}", state.as_str());
            }
        }
        Action::New { .. } => {}
    }
    Ok(())
}

/// The pool's configuration: `[db]`, through the loader the server uses, so

    let config = brocanth_config::Loader::new(crate::app::APP).load()?;
    Ok(config
        .section::<crate::plugins::db::Settings>("db")?
        .config())
}

//! Every route this application answers.
//!
//! `brocanth routes` builds this project and runs this command, so the
//! listing is the binary's own answer rather than a second reading of the
//! directory. The rows are `crate::pages::handlers()`, read off `src/pages`
//! while the crate compiled: a route is a file, so there is nothing to
//! regenerate and nothing that can fall behind.
//!
//! ```text
//! $ brocanth routes
//! VERB    PATTERN        HANDLER
//! GET     /              pages::index::handle
//! GET     /about         pages::about::handle
//! POST    /signup        pages::signup::handle_post
//! ```
//!
//! An argument keeps the rows that contain it, in any column, so
//! `routes signup` is the one page and `routes POST` is every route that
//! writes.
//!
//! These are the pages. A plugin that mounts its own routes through a
//! `RouteSource` would not be in the list, and nothing this project has
//! mounts one -- so today the two are the same table. Say so here if that
//! stops being true.

use brocanth_command::{anyhow, clap, cmd, Command, Context};

/// List the routes this application answers.
#[cmd]
#[derive(clap::Args)]
pub struct Routes {
    /// Keep only the rows containing this text
    pub filter: Option<String>,
}

impl Command for Routes {
    fn run(self, _ctx: &mut Context) -> anyhow::Result<()> {
        let (header, mut rows) = table();
        if let Some(filter) = &self.filter {
            rows.retain(|row| row.iter().any(|cell| cell.contains(filter)));
        }
        if rows.is_empty() {
            // An application with no page and a filter that matched nothing
            // both print nothing, and only one of the two is the reader's
            // own doing.
            match &self.filter {
                Some(filter) => println!("no route matches `{filter}`"),
                None => println!("this application answers no route"),
            }
            return Ok(());
        }

        let mut widths: Vec<usize> = header.iter().map(|title| title.chars().count()).collect();
        for row in &rows {
            for (column, cell) in row.iter().enumerate() {
                widths[column] = widths[column].max(cell.chars().count());
            }
        }

        print_row(&widths, header.iter().copied());
        for row in &rows {
            print_row(&widths, row.iter().map(String::as_str));
        }
        Ok(())
    }
}

/// One line, every cell padded to its column but the last.
///
/// The last is left alone, because trailing spaces are invisible until
/// something pipes this into a diff.
fn print_row<'a>(widths: &[usize], cells: impl Iterator<Item = &'a str>) {
    let mut line = String::new();
    for (column, cell) in cells.enumerate() {
        if column > 0 {
            line.push_str("  ");
        }
        if column + 1 == widths.len() {
            line.push_str(cell);
        } else {
            line.push_str(&format!("{cell:<width$}", width = widths[column]));
        }
    }
    println!("{line}");
}

/// The header, and one row per route.
///
/// Three columns: a route is a verb, a pattern, and the function that
/// answers it. What each one takes and gives would be a fourth and a fifth,
/// and they arrive with the `input` module -- without it nothing in this
/// project has read a handler's signature.
fn table() -> (Vec<&'static str>, Vec<Vec<String>>) {
    let rows = crate::pages::handlers()
        .iter()
        .map(|(verb, pattern, handler)| {
            vec![
                verb.as_str().to_string(),
                (*pattern).to_string(),
                format!("pages::{handler}"),
            ]
        })
        .collect();
    (vec!["VERB", "PATTERN", "HANDLER"], rows)
}

//! The backoffice, at `mount_at` in `src/modules/admin.rs`.
//!
//! Written once by `brocanth create`. This file is yours.
//!
//! It ships locked, and two answers are yours to give:
//!
//! 1. **Who may open a screen.** The policy is `DenyAll` until
//!    `AdminConfig::with_policy` names another, so every screen answers 403.

//!    it reads your user off the request.
//!
//! It has no models yet. A model needs a data source that reads its rows and

//! resource answers 403 on every screen. `brocanth_admin::rows` has the
//! worked example, and `brocanth-admin-model` is the data source over
//! SeaORM.

/// The settings this run chose, kept for [`mount`].
///
/// `mount` runs when the route table is built, after the plugins are, and
/// takes no argument it could carry them in. A test that builds the table

static CHOSEN: OnceLock<Settings> = OnceLock::new();

/// Holds the admin's settings. It puts nothing on the context.
pub struct AdminSettings;

impl Plugin for AdminSettings {
    type Config = Settings;
    type Provides = NoContext;
    const NAME: &'static str = "admin";

    }

/// Every screen of the admin, under `prefix`.
///
/// Four routes: the index and everything under it, for GET and for POST.
/// The admin parses its own paths, so the table gives it the whole path and
/// does not split it into twelve routes.
pub fn mount(routes: &mut RouterBuilder<Handler>, prefix: &str) -> Result<(), BuildError> {
    let defaults = Settings::default();
    let settings = CHOSEN.get().unwrap_or(&defaults);
    let mut config = AdminConfig::new().titled(&settings.title).at(prefix);
    if let Some(header) = &settings.header {
        config = config.headed(header);
    }
    if let Some(heading) = &settings.index_title {
        config = config.index_titled(heading);
    }
    if let Some(href) = &settings.site_url {
        config = config.site_url(href);
    }
    if let Some(href) = &settings.stylesheet {
        config = config.stylesheet(href);
    }
    // The tabs other modules add, such as analytics. Each one is on or off in
    // the module file of the module that adds it: `integrate.admin`.
    config = crate::plugins::admin_panels(config);
    let admin = Arc::new(Admin::new(
        // An empty in-memory source: no models, so no rows to read. Put
        // your own `AdminData` here, or `brocanth_admin_model::admin_source!`
        // over your database connection.
        Arc::new(Fixture::new()),
        config,
    ));
    let rest = format!("{}/{{*rest}}", prefix.trim_end_matches('/'));
    for pattern in [prefix, rest.as_str()] {
        routes.route(pattern, Verb::Get, screen(Arc::clone(&admin)))?;
        routes.route(pattern, Verb::Post, screen(Arc::clone(&admin)))?;
    }
    Ok(())
}

/// One handler for every screen. The admin reads the path and decides.
///
/// The admin is async because a database is, and so is a handler in this
/// table: the answer is awaited and holds no thread while it waits.
fn screen(admin: Arc<Admin>) -> Handler {
    Box::new(move |ctx: &mut Ctx| {
        let admin = Arc::clone(&admin);
        Box::pin(async move {
            let ctx = &*ctx;
            // HEAD reaches the GET handler, and the admin knows two verbs.
            let method = match ctx.verb {
                Verb::Post => "POST",
                _ => "GET",
            };
            let form = submission(ctx);
            let staff = crate::plugins::superuser::staff(ctx).await?;
            let who = principal(staff.as_ref());
            // `?` on the admin's own failure: a data source that failed, or
            // a template that did not render. A refusal is not one: it is a
            // page.
            let answer = admin
                .dispatch(who, method, &ctx.path, &ctx.query, &form)
                .await?;
            Ok(match answer {
                brocanth_admin::Response::Page { status, html } => Response::new(status, html),
                brocanth_admin::Response::Redirect { status, location } => {
                    Response::empty(status).with_header("location", location)
                }
            })
        })
    })
}

/// Who is asking, as the admin's policy sees it: the signed-in member of
/// staff, or anonymous.
///
/// `StaffPolicy` reads it, behind `integrate.admin` in
/// `src/modules/superuser.rs`. A policy of your own can read it too:
/// `principal.downcast::<brocanth_auth::Staff>()`.
fn principal(staff: Option<&brocanth_auth::Staff>) -> Principal<'_> {
    match staff {
        Some(staff) => Principal::new(staff),
        None => Principal::anonymous(),
    }
}

/// The form a POST sent. Anything that is not urlencoded is an empty form,
/// so the screen shows its errors and writes nothing.
fn submission(ctx: &Ctx) -> Submission {
    let urlencoded = ctx
        .content_type()
        .is_some_and(|t| t.starts_with("application/x-www-form-urlencoded"));
    match ctx.body_text() {
        Some(body) if urlencoded => Submission::from_urlencoded(body),
        _ => Submission::new(),
    }
}

===== src/plugins/auth/mod.rs =====
//! Accounts and sessions, as this application registers them.
//!
//! One interface, no providers yet, and a local account path that does not
//! leak who has one: `register`, `login`, `request_password_reset` and
//! `verify_email` answer the same shape and take the same time whether or not
//! the address exists, because the expensive hash runs on both paths.
//!
//! `brocanth-auth` is in `Cargo.toml` because this file is here, and through
//! no other route. Delete the registration in `src/plugins/mod.rs` and
//! delete this file, and the application has no accounts and links nothing
//! to hash a password with.
//!
//! # Signing somebody in
//!
//! Everything is behind one context item, and a handler in an application
//! that never registered this plugin does not compile:
//!
//! ```ignore
//! use brocanth_auth::mfa::{Admission, Enrolled, FactorPolicy, FactorSet};
//!
//! let auth = ctx.provided::<Services, _>().expect("a dispatched request");
//!
//! // 1. The credential. `login` answers the same shape and takes the same
//! //    time whether or not the address has an account.
//! let who = auth.accounts().login(email, password).await?;
//!
//! // 2. What this account could prove. Nothing: `Mfa` needs a store trait
//! //    this application does not implement, so the enrolment is empty and
//! //    the constructor is named so that a reviewer stops on it. The day
//! //    there is an `MfaStore`, `Mfa::second_factors` answers this properly
//! //    and this line goes.
//! let enrolled =
//!     Enrolled::without_asking_the_store(who.account().id.clone(), FactorSet::empty());
//!
//! // 3. The policy, which consumes the proof. `Admitted` has no public
//! //    constructor, so this step cannot be skipped -- a sign-in that
//! //    forgot the second factor is a program that does not compile rather
//! //    than one that ships.
//! let Admission::Admit(admitted) =
//!     FactorPolicy::default().admit(&enrolled, who.into_proof())
//! else {
//!     return Err(Error::new(401, "sign in"));
//! };
//!
//! // 4. The session, which takes the admission by value.
//! let started = auth.sessions().begin_with_factors(admitted, None).await?;
//! // `started.cookie` is the `Set-Cookie` header to send back.
//! ```
//!
//! **A page cannot do that yet.** A handler here is
//! `Box<dyn Fn(&mut Ctx) -> Result<Response, Error>>`, which is synchronous,
//! and every method above is `async fn`. What runs today is a test, a
//! `manage` subcommand, and anything else of yours that already has a
//! runtime. The sign-in form arrives with the core's request hooks, which are
//! also what the two middleware offers below are waiting for.
//!
//! # Three decisions this file makes, each in one line
//!
//! They are type aliases so that changing one changes one line and nothing
//! else in the project changes type.
//!
//! 1. **`Store = PgStore`.** Accounts and sessions are rows in the
//!    `brocanth_auth_*` tables of `[auth] url`, so they survive a restart and
//!    a deploy. The pool is built in `before_start` with no connection, and
//!    the readiness check that `serve` awaits before it listens runs
//!    `migrate` once (it is idempotent and takes an advisory lock). To review
//!    the schema in a migration file instead, copy `PgStore::schema_sql()`
//!    into one and delete the `migrate` call.
//! 2. **`Http = StubHttp`.** There are no providers, so nothing in this
//!    binary ever performs a provider request, and a client that answers "no
//!    stubbed route" is a louder failure than one that opens a socket. It is
//!    also the line that keeps reqwest, rustls and their trees out of the lock
//!    file. Adding a provider changes it; see below.
//! 3. **`Passwords::recommended()`, not a setting.** The hash is argon2id and
//!    there is no choice to make. The one question a deployment gets to answer
//!    is cost, and it is a named constructor rather than three numbers in a
//!    file, because three numbers in a file is how an application ends up
//!    hashing at a cost nobody chose.
//!
//! # Adding a provider
//!
//! Three steps, and the flow code never learns the provider's name.
//!
//! 1. In `Cargo.toml`, `brocanth-auth = { ..., features = ["reqwest-client"] }`.
//! 2. Here, `pub type Http = brocanth_auth::http_client::ReqwestClient;`.
//! 3. Here, in `configure`, push one `ConfiguredProvider`:
//!
//! ```ignore
//! use brocanth_auth::oauth::ClientConfig;
//! use brocanth_auth::plugin::ConfiguredProvider;
//! use brocanth_auth::providers::catalog::GOOGLE;
//!
//! providers: vec![ConfiguredProvider {
//!     provider: GOOGLE,
//!     client: ClientConfig::new(
//!         std::env::var("GOOGLE_CLIENT_ID")?,
//!         std::env::var("GOOGLE_CLIENT_SECRET")?,
//!         "https://example.test/auth/google/callback",
//!     ),
//! }]
//! ```
//!
//! A provider is data -- endpoints, scopes, and a mapping from its profile
//! JSON to a common identity -- so the second one is another entry in that
//! `vec!` and no new code anywhere. `brocanth_auth::providers::catalog` has
//! the ones that are already written down.
//!
//! Two things change with the first provider. `same_site` below must become
//! `Lax`, because the callback is a cross-site top-level navigation and
//! `Strict` drops the cookie on the way back. And a matching email still does
//! not link an account to a provider: `Linking` returns a refusal the
//! application answers with a page that asks, because "this provider says
//! alice@example.com" is not Alice consenting, and several real products have
//! shipped that account takeover.
//!
//! # What is not here
//!
//! Second factors, one-time codes and login alerts. Each needs a store trait
//! of its own -- `MfaStore`, `OtpStore`, `SignInLocationStore` -- and
//! `brocanth-auth` deliberately keeps all three off `Auth`, so that an
//! application wanting a password form does not have to implement a TOTP
//! table. An application that wants them builds them here, from the same
//! `Arc<Store>`, and puts them on the context itself: one line each, in this
//! file, where the decision is visible.

use std::sync::Arc;

use brocanth::plugin_api::{Lifecycle, Plugin, PluginError};
use brocanth::plugin_context::{ContextBuilder, ProvidesContext};
use brocanth::plugin_shutdown::ShutdownCx;
use brocanth_auth::http_client::StubHttp;
use brocanth_auth::plugin::AuthConfig;
use brocanth_auth::session::{SameSite, SessionConfig, SigningKey};
use brocanth_auth::{Clock as _, Passwords, PgStore, SystemClock};

/// Where identities, sessions and single-use tokens live.
///
/// Postgres, at `[auth] url`. See the file header.
pub type Store = PgStore;

/// How this application reaches a provider.
///
/// It does not. There are no providers, so nothing calls this.
pub type Http = StubHttp;

/// Where the time comes from.
pub type Clock = SystemClock;

/// Everything a handler needs, behind one `Arc`: registration, login, the
/// token service, sessions and identity linking.
///
/// An alias rather than a wrapper, because `brocanth_auth::Auth` is already a
/// `ContextItem`. Cloning is cheap and the clone shares the provider key
/// cache, which is why there is one of these and not one per request.
pub type Services = brocanth_auth::Auth<Store, Http, Clock>;

/// What this plugin is built from.
///
/// Five values a file can carry. The store, the client, the clock and the
/// hasher are built below, because none of them is a thing a TOML file can
/// say.
///
/// It derives `Deserialize` because this application reads it from
/// `[auth]` in a TOML file. Drop the config plugin and the derive goes
/// with it: `Plugin::Config` carries no bounds the core imposes.
#[derive(Debug, serde::Deserialize)]
#[serde(default, deny_unknown_fields)]
pub struct Settings {
    /// The Postgres database the accounts and sessions live in.
    pub url: String,
    /// The *name* of the environment variable holding the session signing
    /// key, not the key. 32 bytes, base64url.
    pub signing_key_env: String,
    /// The session cookie's name. `__Host-` in production: it is the only
    /// prefix a browser enforces, and what it enforces is `Secure`, a path of
    /// `/` and no `Domain` -- so a subdomain takeover cannot set a session
    /// cookie for the parent.
    pub cookie_name: String,
    /// Whether the cookie is `Secure`. False only on plain-HTTP localhost,
    /// and then the name above must lose its `__Host-` prefix as well.
    pub secure: bool,
    /// Seconds of inactivity before the session is dead.
    pub idle_timeout: i64,
    /// Seconds from creation before it is dead however much it was used.
    pub absolute_timeout: i64,
}

impl Default for Settings {
    fn default() -> Self {
        Self {
            url: "postgres://localhost/development".to_string(),
            signing_key_env: "MY_BLOG_SESSION_KEY".to_string(),
            cookie_name: "__Host-session".to_string(),
            secure: true,
            idle_timeout: 8 * 60 * 60,
            absolute_timeout: 30 * 24 * 60 * 60,
        }
    }
}

pub struct Auth {
    settings: Settings,
    /// Built at `before_start`, so this is an `Option` for exactly the length
    /// of the build.
    services: Option<Services>,
    /// The store, kept for the sweep.
    store: Option<Arc<Store>>,
    /// The sweep, stopped at shutdown.
    sweeper: Option<tokio::task::JoinHandle<()>>,
}

impl Plugin for Auth {
    type Config = Settings;
    type Provides = ProvidesContext<Services>;
    const NAME: &'static str = "auth";

    fn new(config: Settings) -> Self {
        Self {
            settings: config,
            services: None,
            store: None,
            sweeper: None,
        }
    }

    fn configure(&mut self, _lc: &mut Lifecycle<'_>) -> Result<(), PluginError> {
        if self.settings.url.is_empty() {
            return Err(PluginError::new(
                "[auth] url is required; accounts live in a Postgres database\n\
                 help: set `url` under [auth] in the config file, or the variable for `auth.url`",
            ));
        }
        Ok(())
    }

    fn before_start(&mut self, lc: &mut Lifecycle<'_>) -> Result<(), PluginError> {
        if self.settings.cookie_name.is_empty() {
            return Err(PluginError::new(
                "[auth] cookie_name must not be empty\n\
                 help: set `cookie_name` under [auth] in the config file, such as \"__Host-session\"",
            ));
        }
        // A `__Host-` cookie is `Secure` by definition, so a configuration
        // where the two disagree produces a cookie no browser stores and a
        // login that silently never takes. Caught here, at startup, where it
        // can still be answered.
        if self.settings.cookie_name.starts_with("__Host-") && !self.settings.secure {
            return Err(PluginError::new(
                "[auth] a __Host- cookie is Secure by definition, and secure is false\n\
                 help: set secure = true under [auth], or drop the prefix for plain-HTTP development",
            ));
        }

        let name = self.settings.signing_key_env.clone();
        let signing_key = match std::env::var(&name) {
            // Rejected rather than stretched. Padding a four-character secret
            // to 32 bytes is how a deployment ends up with a forgeable cookie
            // and no warning anywhere.
            Ok(raw) => SigningKey::parse(&raw).map_err(|e| {
                PluginError::new(format!(
                    "[auth] ${name}: {e}\n\
                     help: set it to 32 random bytes, base64url:\n  \
                     export {name}=$(head -c 32 /dev/urandom | base64 | tr '+/' '-_' | tr -d '=')"
                ))
            })?,
            // This project's own tests build the application with no secret
            // in the environment. The key is fixed and public, and `cfg(test)`
            // keeps it out of every binary.
            Err(_) if cfg!(test) => SigningKey::from_bytes([7; 32]),
            Err(_) => {
                return Err(PluginError::new(format!(
                    "[auth] ${name} is not set. There is no default: a key invented at startup \
                     would log every session out on each restart and differ between the \
                     instances of a fleet.\n\
                     help: set it to 32 random bytes, base64url:\n  \
                     export {name}=$(head -c 32 /dev/urandom | base64 | tr '+/' '-_' | tr -d '=')"
                )));
            }
        };

        let passwords = Passwords::recommended().map_err(|e| {
            PluginError::new(format!(
                "[auth] argon2id: {e}\n\
                 help: the parameters are brocanth-auth's own, so this is a bug; report it with this message"
            ))
        })?;

        let session = SessionConfig {
            cookie_name: self.settings.cookie_name.clone(),
            secure: self.settings.secure,
            idle_timeout: self.settings.idle_timeout,
            absolute_timeout: self.settings.absolute_timeout,
            // Strict, and not the crate's own `Lax` default. `Lax` is there
            // for the cross-site top-level navigation an OAuth callback is,
            // and this application has no provider. Add one and this line
            // has to become `Lax` or the cookie is dropped on the way back.
            same_site: SameSite::Strict,
            // Path `/`, and thirty minutes for an impersonated session. Both
            // are the crate's defaults and both are arguments rather than
            // numbers; `SessionConfig` carries them.
            ..SessionConfig::default()
        };

        // The hooks are synchronous and opening a pool is not, so the pool
        // opens lazily and nothing here waits. Waiting would hang a
        // current-thread runtime, where this thread drives the I/O. The pool
        // spawns its upkeep task, so it needs the runtime entered.
        let runtime = tokio::runtime::Handle::try_current().map_err(|e| {
            PluginError::new(format!(
                "[auth] needs a Tokio runtime at startup to open its pool ({e})\n\
                 help: build the application inside the runtime that serves it"
            ))
        })?;
        let store = {
            let _entered = runtime.enter();
            PgStore::connect_lazy(&self.settings.url)
                .map_err(|e| PluginError::new(format!("[auth] {e}")))?
        };
        let store = Arc::new(store);
        self.store = Some(Arc::clone(&store));
        // Ready when the database answers. The first check also creates the
        // tables, and the first check is the one `serve` awaits before it
        // listens: an unreachable database stops the start there.
        let probe = Arc::clone(&store);
        let migrated = Arc::new(tokio::sync::OnceCell::new());
        lc.ready_when(move || {
            let probe = Arc::clone(&probe);
            let migrated = Arc::clone(&migrated);
            async move {
                migrated
                    .get_or_try_init(|| probe.migrate())
                    .await
                    .map_err(|e| format!("creating the auth tables: {e}"))?;
                probe.ping().await.map_err(|e| e.to_string())
            }
        });

        self.services = Some(
            brocanth_auth::Auth::build(AuthConfig {
                store,
                http: Arc::new(Http::new()),
                clock: Arc::new(SystemClock),
                signing_key,
                session,
                passwords: Arc::new(passwords),
                // Add one here. The file header has the three lines.
                providers: Vec::new(),
            })
            .map_err(|e| PluginError::new(format!("[auth] {e}")))?,
        );
        // In the services too, for a plugin that reads the accounts at
        // startup, such as `superuser`.
        if let Some(services) = &self.services {
            lc.services_mut().insert(services.clone());
        }
        Ok(())
    }

    fn start(&mut self, _lc: &mut Lifecycle<'_>) -> Result<(), PluginError> {
        // Dead sessions, spent tokens and expired one-time codes, every
        // fifteen minutes. Every read checks expiry already, so this only
        // keeps the tables small. Each instance runs it, and the deletes do
        // not mind.
        let Some(store) = self.store.clone() else {
            return Ok(());
        };
        let idle_timeout = self.settings.idle_timeout;
        self.sweeper = Some(tokio::spawn(async move {
            let mut every = tokio::time::interval(std::time::Duration::from_secs(15 * 60));
            loop {
                every.tick().await;
                if let Err(e) = store.sweep(SystemClock.now(), idle_timeout).await {
                    crate::plugins::say!(warn, "[auth] sweep: {e}"; error = %e);
                }
            }
        }));
        Ok(())
    }

    fn shutdown(&mut self, _sd: &mut ShutdownCx<'_>) -> Result<(), PluginError> {
        if let Some(sweeper) = self.sweeper.take() {
            sweeper.abort();
        }
        Ok(())
    }

    fn provide(&self, cx: &mut ContextBuilder) {
        // `before_start` runs before the server is handed the application, so
        // the services are here by the time any request builds a context.
        if let Some(services) = &self.services {
            cx.insert(services.clone());
        }
    }
}
===== src/plugins/cache/mod.rs =====
//! The cache, as this application registers it.
//!
//! The cache is a table, and the table is `UNLOGGED`: Postgres writes no
//! write-ahead log for it, ships it to no replica, and truncates it on crash
//! recovery. For any other table that is a catastrophe. For a cache it is the
//! only state you can reason about, and it is what makes a cache in a
//! relational database defensible at all. A deployment that already runs
//! Postgres therefore gets a cache without running Redis.
//!
//! It lives in the database the `db` plugin opens, on the same pool: one set
//! of connections to size, and a cache read is visible as a connection a
//! request is not using for its real query.
//!
//! `brocanth-cache` is in `Cargo.toml` because this file is here, and through
//! no other route. Delete the registration in `src/plugins/mod.rs` and delete
//! this file, and the application caches nothing and links nothing to do it
//! with.
//!
//! The crate also ships `CachePlugin`, and this is not that. `CachePlugin` is
//! handed a cache that is already open, because a pool is made asynchronously
//! and `Plugin::new` is not async. An application whose cache is one setting
//! wants the shorter road, so this plugin opens the cache at startup and
//! publishes the same `CacheHandle`. Hand-built caches -- a second namespace
//! policy, a backend of your own -- are what `CachePlugin` is for, and
//! swapping to it is a change in this file alone.

use std::sync::Arc;

use brocanth::plugin_api::{Lifecycle, Plugin, PluginError};
use brocanth::plugin_context::{ContextBuilder, ProvidesContext};
use brocanth::plugin_shutdown::ShutdownCx;
use brocanth_cache::{CacheHandle, PgCache, Sweeper};

/// What this plugin is built from.
///
/// It derives `Deserialize` because this application reads it from
/// `[cache]` in a TOML file. Drop the config plugin and the derive goes
/// with it: `Plugin::Config` carries no bounds the core imposes.
#[derive(Debug, serde::Deserialize)]
#[serde(default, deny_unknown_fields)]
pub struct Settings {
    /// Create the unlogged cache table at startup if it is absent.
    pub migrate: bool,
    /// Seconds between two sweeps of expired entries; 0 turns the sweep off.
    pub sweep_every: u64,
}

impl Default for Settings {
    fn default() -> Self {
        Self {
            migrate: true,
            sweep_every: 300,
        }
    }
}

pub struct Cache {
    settings: Settings,
    /// Open at `before_start` and not before, so the field is an `Option` for
    /// exactly the length of the build.
    handle: Option<CacheHandle>,
    /// The sweep, from `start` to `shutdown`.
    sweeper: Option<Sweeper>,
}

impl Plugin for Cache {
    type Config = Settings;
    type Provides = ProvidesContext<CacheHandle>;
    const NAME: &'static str = "cache";

    fn new(settings: Settings) -> Self {
        Self {
            settings,
            handle: None,
            sweeper: None,
        }
    }

    fn before_start(&mut self, lc: &mut Lifecycle<'_>) -> Result<(), PluginError> {
        // The pool the `db` plugin opened in its own `before_start`, which
        // runs first because that plugin is registered first.
        let db = lc.services().get::<brocanth_db::Db>().ok_or_else(|| {
            PluginError::new(
                "[cache] found no database pool\n\
                 help: register the `db` plugin before the cache in src/plugins/mod.rs",
            )
        })?;
        let cache = PgCache::from_pool(db.pool().clone());
        let cache: Arc<dyn brocanth_cache::Cache> = Arc::new(cache);
        // Ready when one read of the table answers, as `CachePlugin` does. A
        // miss is a pass: the question is whether the store answers. With
        // `migrate`, the first check creates the table, as the jobs module
        // does, so a new project starts; a project with reviewers turns it
        // off and keeps `PgCache::schema_sql` in a migration. `serve` awaits
        // this once before it listens, so an unreachable store stops the
        // start there.
        let probe = Arc::clone(&cache);
        let migrate = self.settings.migrate;
        let migrated = Arc::new(tokio::sync::OnceCell::new());
        lc.ready_when(move || {
            let probe = Arc::clone(&probe);
            let migrated = Arc::clone(&migrated);
            async move {
                if migrate {
                    migrated
                        .get_or_try_init(|| probe.migrate())
                        .await
                        .map_err(|e| format!("creating the cache table: {e}"))?;
                }
                let key = brocanth_cache::Namespace::new("brocanth.readyz")
                    .and_then(|ns| brocanth_cache::Key::new(ns, "probe"))
                    .map_err(|e| e.to_string())?;
                probe.get(&key).await.map(drop).map_err(|e| {
                    format!(
                        "{e}\nhelp: check `[db] url`; with `migrate = false`, the \
                         `brocanth_cache` table must exist: `PgCache::schema_sql()` has the \
                         statements for a migration"
                    )
                })
            }
        });
        let handle = CacheHandle::new(cache);
        // The first cache this process opens is the one the admin's cache
        // tab reads: the admin is built before any plugin starts.
        handle.install();
        self.handle = Some(handle);
        Ok(())
    }

    fn start(&mut self, _lc: &mut Lifecycle<'_>) -> Result<(), PluginError> {
        // Nothing else deletes expired entries. Every read checks expiry
        // already, so this only frees their space. Each instance runs it,
        // and the deletes do not mind.
        let every = self.settings.sweep_every;
        let Some(handle) = self.handle.as_ref().filter(|_| every > 0) else {
            return Ok(());
        };
        let runtime = tokio::runtime::Handle::try_current().map_err(|e| {
            PluginError::new(format!(
                "[cache] needs a Tokio runtime to sweep ({e})\n\
                 help: build the application inside the runtime that serves it"
            ))
        })?;
        let on_error = |e: &brocanth_cache::CacheError| {
            crate::plugins::say!(warn, "[cache] sweep: {e}"; error = %e);
        };
        self.sweeper = Some(Sweeper::spawn(
            &runtime,
            Arc::clone(handle.inner()),
            std::time::Duration::from_secs(every),
            Arc::new(on_error),
        ));
        Ok(())
    }

    fn shutdown(&mut self, _sd: &mut ShutdownCx<'_>) -> Result<(), PluginError> {
        if let Some(sweeper) = self.sweeper.take() {
            sweeper.stop();
        }
        Ok(())
    }

    fn provide(&self, cx: &mut ContextBuilder) {
        // `before_start` runs before the server is handed the application, so
        // the handle is here by the time any request builds a context.
        if let Some(handle) = &self.handle {
            cx.insert(handle.clone());
        }
    }
}

//! The database, as this application registers it.
//!
//! One Postgres pool, built at startup and shared by every request. It opens
//! no connection in the build; `serve` checks it once before it listens. A page
//! reads it with `ctx.db()`. Another plugin reads it from the application's
//! services: `lc.require_service::<brocanth_db::Db>()`, from its own
//! `before_start`, registered after this one.
//!
//! `brocanth-db` is in `Cargo.toml` because this file is here, and through no
//! other route. Delete the registration in `src/plugins/mod.rs` and this
//! file, and the application links no SQL driver.

use std::time::Duration;

use brocanth::plugin_api::{Lifecycle, Plugin, PluginError};
use brocanth::plugin_context::{ContextBuilder, ProvidesContext};
use brocanth::plugin_shutdown::ShutdownCx;
use brocanth_db::{Db, DbConfig, DbError};

/// What this plugin is built from: `[db]`.
///
/// A duration is a whole number of seconds, and 0 turns an optional one
/// off.
///
/// It derives `Deserialize` because this application reads it from
/// `[db]` in a TOML file. Drop the config plugin and the derive goes
/// with it: `Plugin::Config` carries no bounds the core imposes.
#[derive(Debug, serde::Deserialize)]
#[serde(default, deny_unknown_fields)]
pub struct Settings {
    /// `postgres://user:password@host:5432/database`.
    pub url: String,
    /// The most connections the pool opens.
    pub max_connections: u32,
    /// The connections the pool keeps open when idle.
    pub min_connections: u32,
    /// Seconds a statement waits for a connection, opening one included.
    pub acquire_timeout: u64,
    /// Seconds before an idle connection closes. 0 keeps it open.
    pub idle_timeout: u64,
    /// Seconds before the server cancels a statement. 0 is no limit.
    pub statement_timeout: u64,
    /// The schemas for a table name with no schema. Empty is the server's
    /// default.
    pub search_path: Vec<String>,
    /// Every transaction is read-only.
    pub read_only: bool,
    /// The name `pg_stat_activity` shows. Empty sends none.
    pub application_name: String,
    /// A PEM bundle to trust, with `verify-full`. `None` uses the system's
    /// roots.
    pub ca_file: Option<String>,
}

impl Default for Settings {
    fn default() -> Self {
        Self {
            url: "postgres://localhost/development".to_string(),
            max_connections: 10,
            min_connections: 0,
            acquire_timeout: 30,
            idle_timeout: 600,
            statement_timeout: 0,
            search_path: Vec::new(),
            read_only: false,
            application_name: "my-blog".to_string(),
            ca_file: None,
        }
    }
}

impl Settings {
    /// The pool's configuration, from these settings.
    pub fn config(&self) -> DbConfig {
        let seconds = |secs: u64| (secs > 0).then(|| Duration::from_secs(secs));
        let mut config = DbConfig::new(&self.url);
        config.max_connections = self.max_connections;
        config.min_connections = self.min_connections;
        config.acquire_timeout = Duration::from_secs(self.acquire_timeout);
        config.idle_timeout = seconds(self.idle_timeout);
        config.statement_timeout = seconds(self.statement_timeout);
        config.search_path = self.search_path.clone();
        config.read_only = self.read_only;
        config.application_name =
            Some(self.application_name.clone()).filter(|name| !name.is_empty());
        config.ca_file = self.ca_file.clone().map(Into::into);
        config
    }
}

pub struct Database {
    config: DbConfig,
    /// Open from `before_start` until `shutdown`.
    db: Option<Db>,
}

impl Plugin for Database {
    type Config = Settings;
    type Provides = ProvidesContext<Db>;
    const NAME: &'static str = "db";

    fn new(settings: Settings) -> Self {
        Self {
            config: settings.config(),
            db: None,
        }
    }

    fn configure(&mut self, _lc: &mut Lifecycle<'_>) -> Result<(), PluginError> {
        if self.config.url.is_empty() {
            return Err(PluginError::new(
                "[db] url is required; there is no default database\n\
                 help: set `url` under [db] in the config file, or the variable for `db.url`",
            ));
        }
        Ok(())
    }

    fn before_start(&mut self, lc: &mut Lifecycle<'_>) -> Result<(), PluginError> {
        // The hook is synchronous and connecting is not, so the pool opens
        // lazily and nothing here waits. Waiting would hang a current-thread
        // runtime, where this thread is the one that drives the I/O. The
        // pool spawns its upkeep task, so it needs the runtime entered.
        let runtime = tokio::runtime::Handle::try_current().map_err(|e| {
            PluginError::new(format!(
                "[db] needs a Tokio runtime at startup to open its pool ({e})\n\
                 help: build the application inside the runtime that serves it"
            ))
        })?;
        let db = {
            let _entered = runtime.enter();
            Db::connect_lazy(&self.config).map_err(|e| PluginError::new(format!("[db] {e}")))?
        };
        // Ready while `SELECT 1` answers on a pooled connection. `serve`
        // awaits this once before it listens, so a wrong URL, password or
        // certificate stops the start there and not on the first request.
        let probe = db.clone();
        let url = self.config.redacted_url();
        lc.ready_when(move || {
            let probe = probe.clone();
            let url = url.clone();
            async move {
                probe.ping().await.map_err(|e| match e {
                    DbError::Connect { .. } => format!("connecting to {url}: {e}"),
                    other => other.to_string(),
                })
            }
        });
        // For a plugin registered after this one: the model layer wraps the
        // same pool, so there is one set of connections to size.
        lc.services_mut().insert(db.clone());
        self.db = Some(db);
        Ok(())
    }

    fn shutdown(&mut self, sd: &mut ShutdownCx<'_>) -> Result<(), PluginError> {
        let Some(db) = self.db.take() else {
            return Ok(());
        };
        // Without a runtime there is nothing to wait on: the connections
        // close when the pool drops.
        let Ok(runtime) = tokio::runtime::Handle::try_current() else {
            return Ok(());
        };
        let remaining = sd.remaining();
        // `close` waits for the connections in use to come back, so it
        // waits inside what is left of the grace period and no longer.
        let close = async move {
            match remaining {
                Some(left) => tokio::time::timeout(left, db.close()).await.is_ok(),
                None => {
                    db.close().await;
                    true
                }
            }
        };
        // On a current-thread runtime this thread drives the I/O, so a wait
        // here never ends. The close runs as a task, under the same limit.
        if runtime.runtime_flavor() == tokio::runtime::RuntimeFlavor::CurrentThread {
            runtime.spawn(close);
            return Ok(());
        }
        // On a multi-thread runtime the workers drive it, and a scoped
        // thread waits without blocking one of them.
        std::thread::scope(|scope| {
            scope
                .spawn(|| runtime.block_on(close))
                .join()
                .map_err(|_| {
                    PluginError::new(
                        "[db] the closing thread panicked\n\
                         help: read the panic message above; it names the cause",
                    )
                })
        })?
        .then_some(())
        .ok_or_else(|| {
            PluginError::new(
                "[db] connections were still in use at the deadline\n\
                 help: raise `shutdown_grace_secs` under [server], or end the work that holds a connection",
            )
        })
    }

    fn provide(&self, cx: &mut ContextBuilder) {
        // `before_start` runs before the server is handed the application,
        // so the pool is here by the time a request builds a context.
        if let Some(db) = &self.db {
            cx.insert(db.clone());
        }
    }
}

//! Roles: named sets of permissions, given to accounts.
//!
//! Written once by `brocanth create`. This file is yours.
//!
//! A permission is a string. The admin asks for `<model>.view`, `.add`,
//! `.change` and `.delete` for a model, and `<panel>.view` and `.act` for a
//! tab. A page of yours may ask for any other. A superuser holds every
//! permission. Only a superuser changes roles, in the admin's Roles tab.
//!
//! Who may see a page is a role question:
//!
//! ```ignore
//! let staff = crate::plugins::roles::require(ctx, "report.view").await?;
//! ```
//!
//! Which rows is a row question, and it goes into the query through
//! `brocanth-policy`. `Staff::gate` turns a permission into a predicate, so
//! both travel in one `condition()`:
//!
//! ```ignore
//! fn update(&self, who: &Visitor) -> EntityPred<Post> {
//!     // Editors change every post; everybody else changes their own.
//!     who.staff.gate("post.change").or(Pred::eq(PostColumn::AuthorId, who.id))
//! }
//! ```

use brocanth_auth::Staff;

use crate::pages::{Ctx, Error};

/// The signed-in member of staff, when it holds `permission`.
///
/// # Errors
/// 401 when nobody on the staff is signed in, 403 when the account lacks
/// the permission.
#[allow(dead_code)] // for your pages
pub async fn require(ctx: &Ctx, permission: &str) -> Result<Staff, Error> {
    let Some(staff) = crate::plugins::superuser::staff(ctx).await? else {
        return Err(Error::new(401, "sign in with a staff account"));
    };
    if !staff.allows(permission) {
        return Err(Error::new(
            403,
            format!("this needs the permission {permission}"),
        ));
    }
    Ok(staff)
}

//! Superusers: operator accounts that pass every check of the admin.
//!
//! Written once by `brocanth create`. This file is yours.
//!
//! A superuser is a flag on an account. An operator sets it, never a form a
//! visitor can reach:
//!
//! * at start, from `[superuser] email` and the password in the variable
//!   that `password_env` names, when `email` is set;
//! * with `createsuperuser --email <address>`, when the accounts are in
//!   Postgres.
//!
//! Both are idempotent. An account that exists is promoted and keeps its
//! password, so a second run does not reset a password somebody changed.
//!
//! [`staff`] reads the signed-in member of staff off a request: a
//! superuser, or an account that holds a role. The admin uses it as its
//! principal. A page of yours can too:
//!
//! ```ignore
//! let Some(staff) = crate::plugins::superuser::staff(ctx).await? else {
//!     return Err(Error::new(403, "staff only"));
//! };
//! ```

use std::sync::{Arc, LazyLock, OnceLock};

use brocanth::plugin_api::{Lifecycle, Plugin, PluginError};
use brocanth::plugin_context::NoContext;
use brocanth_auth::{ensure_superuser, Staff, SuperuserOutcome};

use crate::pages::{Ctx, Error};
use crate::plugins::auth::{Services, Store};

/// The superuser the server makes at start: `[superuser]` in the config
/// file, or `$<APP>_SUPERUSER__<KEY>`.
///
/// It derives `Deserialize` because this application reads it from
/// `[superuser]` in a TOML file. Drop the config plugin and the derive goes
/// with it: `Plugin::Config` carries no bounds the core imposes.
#[derive(Debug, serde::Deserialize)]
#[serde(default, deny_unknown_fields)]
pub struct Settings {
    /// The superuser's address. `None` makes no superuser at start.
    pub email: Option<String>,
    /// The *name* of the variable that holds its password, not the
    /// password.
    pub password_env: String,
}

impl Default for Settings {
    fn default() -> Self {
        Self {
            email: None,
            password_env: "MY_BLOG_SUPERUSER_PASSWORD".to_string(),
        }
    }
}

/// The accounts store, once the auth plugin has built it. The admin's
/// roles tab reads it: the admin is built with the route table, before any
/// plugin starts.
static STORE: LazyLock<Arc<OnceLock<Arc<Store>>>> = LazyLock::new(Default::default);

/// The cell [`STORE`] fills at startup.
#[allow(dead_code)] // read by the roles tab, with `roles`
pub fn store() -> Arc<OnceLock<Arc<Store>>> {
    Arc::clone(&STORE)
}

/// The signed-in member of staff, or `None`: no session, a pre-login one,
/// or an account that is neither a superuser nor holds a role.
///
/// # Errors
/// A store that fails is a 500. A missing or forged cookie is `None`.
#[allow(dead_code)] // read by the admin, with `admin`
pub async fn staff(ctx: &Ctx) -> Result<Option<Staff>, Error> {
    let (Some(auth), Some(cookies)) = (ctx.provided::<Services, _>(), ctx.header("cookie")) else {
        return Ok(None);
    };
    Staff::from_cookie_header(auth, cookies)
        .await
        .map_err(|e| Error::internal(format!("superuser: {e}")))
}

pub struct Superuser {
    settings: Settings,
    /// Read in `configure`, used once in `start`.
    password: Option<String>,
}

impl Plugin for Superuser {
    type Config = Settings;
    type Provides = NoContext;
    const NAME: &'static str = "superuser";

    fn new(config: Settings) -> Self {
        Self {
            settings: config,
            password: None,
        }
    }

    fn configure(&mut self, _lc: &mut Lifecycle<'_>) -> Result<(), PluginError> {
        if self.settings.email.is_none() {
            return Ok(());
        }
        let name = &self.settings.password_env;
        match std::env::var(name) {
            Ok(password) if !password.is_empty() => self.password = Some(password),
            // This project's own tests build the application with no secret
            // in the environment, and make no superuser.
            _ if cfg!(test) => {}
            _ => {
                return Err(PluginError::new(format!(
                    "[superuser] email is set and ${name} is not\n\
                     help: export {name}=<a long passphrase>, or unset [superuser] email"
                )));
            }
        }
        Ok(())
    }

    fn before_start(&mut self, lc: &mut Lifecycle<'_>) -> Result<(), PluginError> {
        let services = lc.require_service::<Services>()?;
        let _ = STORE.set(Arc::clone(services.store()));
        Ok(())
    }

    fn start(&mut self, lc: &mut Lifecycle<'_>) -> Result<(), PluginError> {
        let (Some(email), Some(password)) = (self.settings.email.clone(), self.password.take())
        else {
            return Ok(());
        };
        let services = lc.require_service::<Services>()?.clone();
        // `start` runs after the startup gate, so the auth tables exist.
        tokio::spawn(async move {
            let outcome = ensure_superuser(
                services.accounts(),
                services.store().as_ref(),
                &email,
                &password,
            )
            .await;
            match outcome {
                Ok(SuperuserOutcome::Created(_)) => {
                    crate::plugins::say!(info, "superuser: created {email}"; %email);
                }
                Ok(SuperuserOutcome::Promoted(_)) => {
                    crate::plugins::say!(info, "superuser: promoted {email}"; %email);
                }
                Ok(SuperuserOutcome::Unchanged(_)) => {}
                Err(e) => {
                    crate::plugins::say!(warn, "superuser: {email}: {e}"; %email, error = %e);
                }
            }
        });
        Ok(())
    }
}

Run these after every change, in this order. They are ordered by how

- `config`  config/<env>.toml, merged; without it nothing links a TOML parser
- `commands`  this project's own command line, `serve` plus whatever you add
- `db`  one Postgres pool and sea-query statements, read as ctx.db(); without it no SQL driver
- `cache`  a cache on an unlogged Postgres table, not Redis; without it no cache at all
- `admin`  a Django-shaped backoffice at /admin, locked until you say who may in
- `auth`  argon2id accounts and signed sessions; no provider, so no HTTP client
- `superuser`  operator accounts that pass every admin check, and a createsuperuser command
- `roles`  roles that grant permissions, stored with the accounts, for staff and the admin

**Absent:** `logging`, `observability`, `observability-logs`, `observability-health`, `observability-metrics`, `observability-tracing`, `analytics`, `beacon`, `sessions`, `model`, `cache-memory`, `cache-redis`, `jobs`, `jobs-memory`, `flags`, `audit`, `mail`, `auth-oauth`, `auth-screens`, `auth-lockout`, `policy`, `storage`, `storage-routes`, `input`, `client`, `middleware`, `middleware-csrf`, `middleware-security-headers`, `middleware-rate-limit`, `middleware-conditional-get`, `middleware-compression`, `flash`, `gtag`, `openapi`, `validation`, `resource`, `admin-model`, `admin-actions`, `webhooks`, `webhooks-in`, `webhooks-out`, `oauth`, `oauth-client`, `oauth-client-memory`, `realtime`, `realtime-sse`, `realtime-websocket`, `rsx`, `jt`, `rsx-tailwind`, `rsx-htmx`, `sanitize`, `static`, `robots`, `sitemap`, `pagination`, `filters`, `search`, `i18n`, `i18n-routing`, `meta`, `errors`, `docker`, `docker-kubernetes`, `ci`, `ci-github` — not compiled, not linked, absent from `Cargo.lock`. Do not write code that uses them; add the dependency and name the plugin in `main` first.

That is the framework's central rule: a capability nobody asked for is
not compiled and not linked. If you need one, add it to `Cargo.toml`
*and* name its plugin in `src/main.rs`. Nothing registers itself.

## Adding a route

A route is a file. `src/pages/about.rs` is `/about`;
`src/pages/users/[id].rs` is `/users/{id}`. **Write the file. That is
the whole of it** -- no `mod` line, no row to register, because there is
no list. `brocanth_pages::pages!` in `src/pages/mod.rs` reads this
directory while the crate compiles and writes both.

The verb comes off the handler's name: `handle` is GET, then
`handle_head`, `handle_post`, `handle_put`, `handle_patch`,
`handle_delete`, `handle_options`. A `pub fn handle_*` outside that
list is a build error naming the list, rather than a handler nothing
reaches; a page with no handler is a build error too. A private
`fn handle_*` is a helper and is left alone.

A handler may be `pub async fn` when it waits on something, such as a
database: the table awaits it and the wait holds no thread. A page that
waits on nothing stays a plain `fn`. An async handler is dropped at an
`.await` when the client goes away, so do not leave a write half done
across one.

**Adding a file does not rebuild on its own** -- nothing referenced it
yet. Run `touch src/pages/mod.rs` after creating a page or a layout.
Editing or deleting one is noticed.

A handler returns a value and never writes a response itself. That is
what makes a half-written body on an error impossible rather than
unlikely.

## The admin

`src/plugins/admin/mod.rs` mounts the backoffice at `mount_at` in
`src/modules/admin.rs` (`/admin` by default). It ships
locked: the policy is `DenyAll` and `principal` returns an anonymous
principal, so every screen answers 403. Do not open it with
`AllowAllUnsafeForDevelopment` outside a test.

A model reaches the admin through `AdminConfig::resource(..)`: a
`brocanth_admin::AdminResource` with a `brocanth_policy::Policy` over
the model's rows. A model with no resource answers 403. Never filter
rows after a read; the row policy goes into the query.

## Migrations

A schema change is a SQL file in `migrations/`. Write one with
`brocanth migrate new add_posts` (or `cargo run -- migrate new add_posts`),
then fill it:

```sql
CREATE TABLE posts (id bigserial PRIMARY KEY, title text NOT NULL);

-- down
DROP TABLE posts;
```

`migrate up` applies each pending file in name order, each in one
transaction. `migrate down [N]` runs the `-- down` part of the newest N.
`migrate status` lists each file. Do not edit a file that is applied: write
a new one.

## The database

`src/plugins/db/mod.rs` opens one Postgres pool at startup from `[db] url`.
A page reads it with `ctx.db()`, a `&brocanth_db::Db`. Build a statement
with `brocanth_db::sea_query`. Do not format values into SQL text: sea-query
sends each value as a bound parameter.

```rust
use brocanth_db::sea_query::{Expr, ExprTrait, Order, Query};
use brocanth_db::Exec;

use crate::pages::{Ctx, Error, IntoResponse};

pub async fn handle(ctx: &mut Ctx) -> Result<impl IntoResponse, Error> {
    let q = Query::select()
        .columns(["id", "title"])
        .from("posts")
        .and_where(Expr::col("published").eq(true))
        .order_by("id", Order::Desc)
        .limit(10)
        .to_owned();
    let posts: Vec<(i64, String)> = ctx.db().all(&q).await?;
    Ok(posts
        .iter()
        .map(|(id, title)| format!("{id} {title}\n"))
        .collect::<String>())
}
```

`ctx.db().json(&q).await?` returns each row as a `serde_json::Value`,
encoded by Postgres. Writes that belong together go in one transaction.
`Ok` commits, and `Err` rolls back:

```rust
ctx.db()
    .transact(async |tx| {
        tx.execute(&debit).await?;
        tx.execute(&credit).await?;
        Ok(())
    })
    .await?;
```

A `brocanth_db::DbError` goes through `?` as a 500. Its message goes to the
log, never to the client.

## Superusers

`src/plugins/superuser/mod.rs` reads the signed-in member of staff with
`staff(ctx)`. A superuser passes every check of the admin. Never set the
superuser flag from a form or a page: an operator sets it, with
`[superuser] email` at start or the `createsuperuser` command.

## Roles

Check a permission with `crate::plugins::roles::require(ctx, "<name>")`.
A role answers which people; it does not answer which rows. Put a row
rule in a `brocanth_policy::Policy`, and combine it with a permission
through `Staff::gate`. Never filter rows after a paginated load.

## Tasks

A command is a `clap::Args` struct with `#[cmd]` on it, in its own file under
`src/commands/`, plus a `pub mod` line at the end of `src/commands/mod.rs`,
below the line that starts `// Your commands`. `brocanth add` rewrites the
lines above it and keeps yours. The command becomes a subcommand
automatically — registration happens at link time.

```
brocanth --help      # lists this project's commands too
brocanth serve
```

## Decisions that belong to the person

The modules in this project are the person's choice. Do not add or
remove one on your own. Before you run `brocanth add`, show the person
the question: `brocanth add --questions` lists each module with its
default, what it needs and what it recommends. Give your
recommendation, and wait for the answer.

Not chosen, and recommended by a module that is here. Ask about each:

- `observability-logs`, for `config`: $<APP>_ENV picks the environment; only observability-logs writes JSON when it is production
- `observability-health`, for `db`: its readiness check, `Db::ping`, is read only by /readyz; without it a deploy sends traffic to an instance whose database is down
- `observability-health`, for `cache`: its readiness check is read only by /readyz; without it a deploy cannot tell that the cache table is unreachable

## House style

- A comment says **why**, next to the decision, and what was chosen
  against. Not what the line does.
- A test name is a sentence describing the guarantee it protects.
- Prefer a type that cannot be built wrong over a check that can be
  forgotten.
- State a limit where someone would look for it rather than leaving it
  to be discovered.
