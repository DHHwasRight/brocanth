//! Written by `brocanth`. Do not edit -- `brocanth add` and `brocanth remove` rewrite this file.
//! Something else belongs here? Put it in `src/app.rs`, which is yours.
//!
//! My Blog -- a brocanth application.
//!
//! No command line was asked for, so this binary does one thing. It takes the
//! flags its settings declare, which `--help` lists and `src/modules/`
//! renames, and nothing else. Everything it contains is in `src/app.rs`;
//! `brocanth add commands` turns this into a command line with `serve` as one
//! of its commands.

mod app;
mod modules;
mod pages;
mod plugins;

use std::process::ExitCode;

fn main() -> ExitCode {
    let served = plugins::read_line().and_then(|helped| {
        if helped {
            Ok(())
        } else {
            app::serve(&app::default_addr())
        }
    });
    match served {
        Ok(()) => ExitCode::SUCCESS,
        Err(message) => {
            // Through `say!`: a JSON line when the log subscriber is up,
            // text when the failure came before it.
            plugins::say!(error, "error: {message}");
            ExitCode::FAILURE
        }
    }
}

//! My Blog -- everything this binary contains, in the order it happens.
//!
//! This file is yours. `brocanth add` does not rewrite it: what it would
//! have to say is in `src/plugins/mod.rs`, which is generated, and this one
//! only asks that file four questions.
//!
//! A plugin that is not named in `plugins::register!` is not registered, and
//! one that is not in `Cargo.toml` is not compiled at all. "The framework
//! decided for you and you cannot see where" is the thing this layout exists
//! to make impossible.
//!
//! `serve` and every other command read this, so there is one answer to what
//! this application is.

use std::sync::Arc;
use std::time::{Duration, Instant};

use brocanth::plugin_app::AppBuilder;
use brocanth::plugin_hooks::{self as hooks, Begin, RequestState};
use brocanth::plugin_routes::BoxFuture;
use brocanth::plugin_shutdown::ShutdownReason;
use brocanth::routing::{AsyncMiddlewareStack, Dispatch, Router, Verb};

use crate::pages::{self, Ctx, Handler, IntoResponse, Response};
use crate::plugins;

/// The application name. Every environment variable this binary reads is
/// `MY_BLOG_<SUFFIX>`: one convention, derived once, whatever spelling the
/// name arrived in.
pub const APP: &str = "my-blog";

/// Where `serve` listens when nothing says else: `$MY_BLOG_ADDR`, or
/// `127.0.0.1:3000`. The container image sets `0.0.0.0:3000`.
pub fn default_addr() -> String {
    std::env::var("MY_BLOG_ADDR").unwrap_or_else(|_| "127.0.0.1:3000".to_string())
}

/// What a request needs, built once and shared by every connection.
pub struct App {
    pub router: Router<Handler>,
    pub stack: AsyncMiddlewareStack<Ctx, Response>,
    pub plugins: Vec<String>,
    /// The application the plugins built.
    ///
    /// Held rather than dropped, because three things need it
    /// after `build` returns: a per-request context, the routes a
    /// plugin contributed, and an HTTP test client. It used to be
    /// a boxed closure over `context()` alone, which cost an
    /// indirect call per request and reached none of the rest.
    ///
    /// Nameable at last: `plugins::Installed` and `plugins::Hooks`
    /// are the two type parameters, and the generator writes both.
    pub core: Core,
}

/// The built application, with its provider list and hook record.
type Core = brocanth::plugin_app::App<plugins::Installed, plugins::Hooks>;

/// Builds it. The only place that decides what this binary is.
pub fn build() -> Result<App, String> {
    // 1. The application, and every plugin it contains. The chain is in
    //    `src/plugins/mod.rs`, one line per plugin and each one a line a
    //    reader can delete. It is there rather than here because the
    //    generator rewrites it whenever the module list changes.
    let app = plugins::register!(AppBuilder::new())
        .setup(|setup| {
            // A plugin *offers* middleware and cannot mount it: a
            // `MiddlewareCatalog` has no `mount`. Mounting takes a pattern, the
            // pattern is a decision, and the decision is the application's --
            // which is why these lines are here, in the file you own.
            // Nothing in this binary offers middleware, so nothing is mounted.
            let _ = setup.router();
            Ok(())
        })
        .build()
        .map_err(|e| e.to_string())?;

    // 2. The route table: the shape of src/pages and nothing else. A file is
    //    a route, and `pages::router` is the only place that says so.
    let router = pages::router().map_err(|e| e.to_string())?;

    // 3. The chain, from exactly what the block above mounted. A name this
    //    binary has no layer for stops the application rather than silently
    //    not running.
    let mut stack: AsyncMiddlewareStack<Ctx, Response> = AsyncMiddlewareStack::new();
    for mount in app.router_plan().mounts() {
        let layer = plugins::layer(&mount.middleware).ok_or_else(|| {
            format!(
                "`{}` is mounted, but this binary has no layer for it",
                mount.middleware
            )
        })?;
        stack
            .use_at(&mount.pattern, layer)
            .map_err(|e| e.to_string())?;
    }

    // 4. What the plugins put on the per-request context, said once.
    plugins::announce!(&app);

    let plugins = app.instance_names().map(str::to_string).collect();

    Ok(App {
        router,
        stack,
        plugins,
        core: app,
    })
}

/// Turns a page's answer into the response the hooks see.
///
/// `text/html` unless the page names its own `content-type`, which then
/// replaces it rather than joining it. Every other header is appended,
/// because `set-cookie` has to repeat. A page that answered with a stream
/// keeps it: the hooks see its headers, and the server polls its body.
fn sent(answered: Response) -> hooks::Response {
    match answered.stream {
        Some(stream) => {
            brocanth_server::bridge::streaming(answered.status, answered.headers, stream)
        }
        None => brocanth_server::bridge::buffered(
            answered.status,
            answered.headers,
            answered.body.into_bytes(),
        ),
    }
}

/// The request headers as text, for the page.
///
/// A value that is not UTF-8 is dropped: a page reads a header as `&str`,
/// and a lossy copy would say something the client did not send.
fn headers(request: &hooks::Request) -> Vec<(String, String)> {
    request
        .headers()
        .iter()
        .filter_map(|(name, value)| Some((name.to_string(), value.to_str().ok()?.to_string())))
        .collect()
}

/// How long a stop waits for open requests and then for the plugins.
///
/// Kubernetes sends SIGTERM, waits `terminationGracePeriodSeconds` (30 by
/// default) and then sends SIGKILL. 25 seconds leaves 5 for the exit.
/// `[server] shutdown_grace_secs` overrides it: `--server.shutdown_grace_secs`,
/// `$<APP>_SERVER__SHUTDOWN_GRACE_SECS`, or with the config plugin the file.
/// See `plugins::shutdown_grace`.
pub const SHUTDOWN_GRACE: Duration = Duration::from_secs(25);

/// Serves until SIGTERM or SIGINT, then stops in two phases (PLAN.md ADR 2).
///
/// 1. The listener closes. Each open request has the grace period to
///    finish, and each open response stream is told to end.
/// 2. Each plugin stops, in reverse registration order, inside what is left
///    of the grace period.
///
/// An error is a non-zero exit, and an orchestrator reads it: the drain ran
/// out of time, the listener failed, or a plugin did not stop cleanly.
///
/// Synchronous on the outside because the command that calls it is, and
/// the runtime is built here rather than wrapped around `main`: a command
/// that never serves should not start a reactor to run.
pub fn serve(addr: &str) -> Result<(), String> {
    let grace = plugins::shutdown_grace()?;
    // Before the socket, so a signal during startup stops the server
    // through the drain and does not kill the process.
    let stop = watch(grace).map_err(|e| e.to_string())?;
    let runtime = tokio::runtime::Builder::new_multi_thread()
        .enable_all()
        .build()
        .map_err(|e| e.to_string())?;

    // The runtime first, and the application inside it. A plugin
    // that opens a connection pool builds it in `before_start`
    // with no connection, and the pool spawns its upkeep task on
    // the current runtime. Building the application first would
    // leave it with no runtime to spawn on.
    let app = {
        let _guard = runtime.enter();
        build()?
    };
    let names = app.plugins.join(", ");
    plugins::say!(info, "plugins: [{names}]"; plugins = %names);

    let result = runtime.block_on(run(addr, app, stop, grace));
    // A request that outlived the drain can still be on a worker. Waiting
    // for it here would undo the grace period, so it is dropped.
    runtime.shutdown_background();
    result
}

/// The async half of [`serve`].
async fn run(
    addr: &str,
    mut app: App,
    mut stop: tokio::sync::mpsc::UnboundedReceiver<Stop>,
    grace: Duration,
) -> Result<(), String> {
    // Every readiness check once, before the socket: the same checks
    // `/readyz` runs. The build waits for no I/O, so a wrong password or
    // an unreachable database stops the start here, with the plugin's
    // message and its `help:` line, and not at the first request. A
    // check can wait for a pool's acquire timeout, so a signal ends the
    // wait.
    tokio::select! {
        checked = app.core.readiness().check_all() => {
            checked.map_err(|not_ready| not_ready.to_string())?;
        }
        _ = stop.recv() => {
            let report = app.core.shutdown(ShutdownReason::Signal, Some(grace));
            plugins::say!(info, "{report}");
            return Ok(());
        }
    }

    // One gate: a stop at the listener and a stop at `App::shutdown`
    // close the same set of response streams.
    let server = brocanth_server::Server::bind(addr)
        .await
        .map_err(|e| e.to_string())?
        .with_streams(app.core.streams().clone());
    let bound = server.local_addr().map_err(|e| e.to_string())?;
    let stopper = server.stopper();

    // After the bind, so a plugin does not start work for a port that
    // turned out to be taken.
    if let Err(e) = app.core.start() {
        let report = app.core.shutdown(ShutdownReason::Failed, None);
        plugins::say!(error, "{report}");
        return Err(e.to_string());
    }

    // The probes and `/metrics` on a port of their own, when the
    // configuration names one.
    plugins::side_listeners(&app).await?;

    let app = Arc::new(app);
    // `plugins::wrap` puts whatever observes a request around it.
    let handler = plugins::wrap(&app, {
        let app = Arc::clone(&app);
        move |request| {
            let app = Arc::clone(&app);
            async move { handle(&app, request).await }
        }
    });
    let mut serving = tokio::spawn(server.run(handler));
    plugins::say!(info, "listening on http://{bound}"; addr = %bound);

    // Phase 1: stop accepting, and give the open requests the grace period.
    let mut failed = tokio::select! {
        event = stop.recv() => {
            let signal = match event {
                Some(Stop::Signal(signal)) => signal,
                _ => "stop",
            };
            plugins::say!(
                info,
                "{signal}: stopping; open requests have {grace:?} to finish";
                signal,
                ?grace
            );
            None
        }
        // The listener stopped by itself. That is a failure, and the
        // plugins still stop.
        served = &mut serving => Some(stopped(served)),
    };
    let reason = match failed {
        None => ShutdownReason::Signal,
        Some(_) => ShutdownReason::Failed,
    };
    let started = Instant::now();
    // Not ready from the first instant of the stop (ADR 2): `/readyz`
    // answers 503, and a plugin that awaits `Readiness::draining` stops
    // taking work now -- the job worker stops claiming -- while the
    // listener still drains.
    app.core.readiness().drain();
    if failed.is_none() {
        // With `observe`, `/readyz` fails first, and the listener closes
        // after `[observability] drain_delay_secs`. Inside the grace period.
        plugins::stop_accepting(&app, stopper).await;
        failed = tokio::select! {
            served = &mut serving => match served {
                Ok(Ok(())) => None,
                other => Some(stopped(other)),
            },
            event = stop.recv() => Some(match event {
                // A second Ctrl-C means "now".
                Some(Stop::Signal(signal)) => {
                    format!("a second {signal} arrived, so open requests were cut off")
                }
                _ => format!("open requests did not finish in {grace:?} and were cut off"),
            }),
        };
    }
    // Nothing when the drain finished. After a timeout, this drops the
    // connections that are still open.
    serving.abort();

    // Phase 2: the plugins, inside what is left of the grace period.
    let remaining = grace.saturating_sub(started.elapsed());
    match owned(app).await {
        Some(mut app) => {
            let report = app.core.shutdown(reason, Some(remaining));
            plugins::say!(info, "{report}");
            if failed.is_none() && (!report.is_clean() || report.overran()) {
                failed = Some(format!("the plugins did not stop cleanly in {grace:?}"));
            }
        }
        None => plugins::say!(
            warn,
            "a request still holds the application, so no plugin stopped"
        ),
    }
    match failed {
        None => Ok(()),
        Some(why) => Err(why),
    }
}

/// Why the listener task ended before a clean drain.
fn stopped(
    served: Result<Result<(), brocanth_server::ServeError>, tokio::task::JoinError>,
) -> String {
    match served {
        Ok(Ok(())) => "the listener stopped with no signal".to_string(),
        Ok(Err(e)) => e.to_string(),
        Err(e) => e.to_string(),
    }
}

/// The application back from the connections that shared it, or `None` when
/// one of them still runs.
///
/// `App::shutdown` needs `&mut`. A finished connection task can be a few
/// instructions from dropping its clone, so this yields a few times first.
async fn owned(mut app: Arc<App>) -> Option<App> {
    for _ in 0..64 {
        match Arc::try_unwrap(app) {
            Ok(app) => return Some(app),
            Err(shared) => {
                app = shared;
                tokio::task::yield_now().await;
            }
        }
    }
    None
}

/// What the stop thread reports to [`run`].
enum Stop {
    /// SIGTERM or SIGINT, by name.
    Signal(&'static str),
    /// The grace period since the first signal is over.
    Deadline,
}

/// Watches for SIGTERM and SIGINT on a thread of its own. It reports each
/// signal, and the end of the grace period after the first one.
///
/// Not on the serving runtime. A handler is synchronous and holds its
/// worker, and while it runs that runtime can leave its signal and timer
/// drivers unpolled: a stop would wait for the slowest request, and the
/// grace period would not run out. This thread has a runtime of its own,
/// and the channel wakes `run` directly.
fn watch(grace: Duration) -> std::io::Result<tokio::sync::mpsc::UnboundedReceiver<Stop>> {
    let runtime = tokio::runtime::Builder::new_current_thread()
        .enable_all()
        .build()?;
    let mut signals = {
        let _guard = runtime.enter();
        Signals::listen()?
    };
    let (tx, rx) = tokio::sync::mpsc::unbounded_channel();
    std::thread::Builder::new()
        .name("stop".to_string())
        .spawn(move || {
            runtime.block_on(async move {
                let first = signals.next().await;
                if tx.send(Stop::Signal(first)).is_err() {
                    return;
                }
                let next = tokio::select! {
                    () = tokio::time::sleep(grace) => Stop::Deadline,
                    second = signals.next() => Stop::Signal(second),
                };
                let _ = tx.send(next);
            });
        })?;
    Ok(rx)
}

/// SIGTERM and SIGINT as one stream. An orchestrator sends SIGTERM, and
/// Ctrl-C is SIGINT.
#[cfg(unix)]
struct Signals {
    term: tokio::signal::unix::Signal,
    int: tokio::signal::unix::Signal,
}

#[cfg(unix)]
impl Signals {
    fn listen() -> std::io::Result<Self> {
        use tokio::signal::unix::{signal, SignalKind};
        Ok(Self {
            term: signal(SignalKind::terminate())?,
            int: signal(SignalKind::interrupt())?,
        })
    }

    /// The next signal, by name.
    async fn next(&mut self) -> &'static str {
        tokio::select! {
            _ = self.term.recv() => "SIGTERM",
            _ = self.int.recv() => "SIGINT",
        }
    }
}

/// Ctrl-C alone, where there is no SIGTERM.
#[cfg(not(unix))]
struct Signals;

#[cfg(not(unix))]
impl Signals {
    fn listen() -> std::io::Result<Self> {
        Ok(Self)
    }

    /// The next signal, by name.
    async fn next(&mut self) -> &'static str {
        let _ = tokio::signal::ctrl_c().await;
        "Ctrl-C"
    }
}

/// One request: the request hooks, the table, then the response hooks.
///
/// A request hook may answer instead of the page -- a CSRF refusal, a
/// 429 -- and then the page never runs. It may also rewrite the request
/// first, so the table reads the request as the hooks left it. The
/// response hooks see every answer, the 404 and the 405 included.
///
/// `begin_async` and not `begin`: it runs a hook that waits as well as
/// one that does not. With no hook that waits, it is ready on its first
/// poll.
pub async fn handle(app: &App, request: brocanth_server::Request) -> brocanth_server::Response {
    let Ok(mut request) = brocanth_server::bridge::to_core(request) else {
        // The path and the query do not make a URI. The table was not
        // asked, so this is not a 404.
        return brocanth_server::Response::text(400, "Bad Request");
    };
    let mut state = RequestState::new();
    let outcome = match app.core.begin_async(&mut request, &mut state).await {
        Begin::Dispatch => Ok(sent(dispatch(app, &request, &mut state).await)),
        Begin::Answered(outcome) => outcome,
    };
    let response = app.core.finish(&mut request, &mut state, outcome);
    brocanth_server::bridge::from_core(response)
}

/// The table, then the chain, then the handler.
///
/// The chain runs only for a route that matched. A 404 has no route and
/// therefore no context. `state` is what the request hooks wrote; the
/// page reads it through `Ctx::hooked`, and it goes back afterwards so
/// the response hooks read what the page left.
///
/// `async` because a page may be: the chain and the page are awaited
/// here, on the worker, and a page that waits on a database does not
/// hold a thread while it does.
pub async fn dispatch(app: &App, request: &hooks::Request, state: &mut RequestState) -> Response {
    let Some(verb) = Verb::from_http(request.method()) else {
        return Response::new(501, "Not Implemented");
    };
    let path = request.uri().path();
    match app.router.dispatch(verb, path) {
        Dispatch::Found { handler, matched } => {
            let mut ctx = Ctx::new(verb, path, matched.params())
                .with_plugins(app.core.context_for(request, state))
                .with_state(std::mem::take(state))
                .sent(
                    request.uri().query().unwrap_or(""),
                    request.body().clone(),
                    headers(request),
                );
            // The one place a failed handler becomes a response --
            // what Echo calls an HTTPErrorHandler. It sits inside
            // the chain's terminal, so a layer still sees what the
            // error turned into and can rewrite it.
            //
            // `handler(c)` is called before the future is built: the
            // future may borrow the context and nothing else.
            let response = app
                .stack
                .run(path, &mut ctx, |c: &mut Ctx| -> BoxFuture<'_, Response> {
                    let answer = handler(c);
                    Box::pin(
                        async move { answer.await.unwrap_or_else(IntoResponse::into_response) },
                    )
                })
                .await;
            *state = ctx.take_state();
            response
        }
        Dispatch::MethodNotAllowed { allow, .. } => Response {
            status: 405,
            headers: vec![("allow".to_string(), allow.to_string())],
            ..Default::default()
        },
        Dispatch::NotFound => Response::new(404, "Not Found"),
    }
}

//! The route table, which is this directory.
//!
//! One file is one route, and the path on disk is the URL: `index.rs` is `/`,
//! `about.rs` is `/about`, `users/[id].rs` is `/users/{id}`.
//!
//! Nothing below lists them. The macro reads this directory while the crate
//! is compiled and writes the `mod` lines and [`router`] from what is there,
//! so adding a page is adding a file and renaming one renames its route.
//! The verb comes off the handler's name: `handle` is GET, `handle_post` is
//! POST, and a `pub fn handle_*` that is not a verb is a build error rather
//! than a handler nothing reaches.
//!
//! `brocanth_pages::pages!` and `cargo expand` is how to read the result.

brocanth_pages::pages!("src/pages", describe);

/// What a handler produces.
///
/// The framework owns rendering: a handler returns a value and never writes a
/// response itself, so a half-written body on an error is not something that
/// can happen. Echo and Gin hand the handler a writer and take an error back,
/// and both carry the scar -- once a 200 has gone out, the central error
/// handler cannot take it back. Here there is nothing to take back.
///
/// The one exception is opt-in: [`Response::stream`] and [`Response::events`]
/// answer with a body that is written after the handler returns. A stream
/// that fails halfway cannot become a 500, because the status is already on
/// the wire. Use it for server-sent events and large downloads, and nowhere
/// else.
pub struct Response {
    pub status: u16,
    pub body: String,
    pub headers: Vec<(String, String)>,
    /// A body the server polls after the handler returns. `None` for every
    /// page that answers with `body`, which is nearly all of them; when it
    /// is `Some`, `body` is ignored. Set it with [`Response::stream`].
    pub stream: Option<Streaming>,
}

/// Written by hand rather than derived, and the difference matters.
///
/// A derived `Default` would give `status: 0`, which is not an HTTP status, so
/// `Response { body: .., ..Default::default() }` would compile and be wrong.
/// 200 is what "I did not say" means for a response.
impl Default for Response {
    fn default() -> Self {
        Self {
            status: 200,
            body: String::new(),
            headers: Vec::new(),
            stream: None,
        }
    }
}

impl Response {
    pub fn new(status: u16, body: impl Into<String>) -> Self {
        Self {
            status,
            body: body.into(),
            ..Default::default()
        }
    }

    pub fn ok(body: impl Into<String>) -> Self {
        Self::new(200, body)
    }

    pub fn empty(status: u16) -> Self {
        Self::new(status, "")
    }

    /// A body written as it is produced: the first bytes reach the client
    /// before the last exist.
    ///
    /// The response hooks see the headers and never the body. The server
    /// tells the stream to end when it stops (`BodyStream::finish`), so a
    /// long-lived stream should implement that, or a deploy waits for it.
    #[allow(dead_code)]
    pub fn stream(status: u16, content_type: &str, stream: impl BodyStream) -> Self {
        Self {
            status,
            headers: vec![("content-type".to_string(), content_type.to_string())],
            stream: Some(Streaming::new(stream)),
            ..Default::default()
        }
    }

    /// Server-sent events: `text/event-stream`, and the two headers that stop
    /// a cache or a proxy from holding the events back.
    ///
    /// `brocanth_server::sse::channel` makes the stream from async code:
    ///
    /// ```ignore
    /// let (events, stream) = brocanth_server::sse::channel(16);
    /// tokio::spawn(async move {
    ///     // `Err` when the client left or the server is stopping.
    ///     let _ = events.send(Event::new("hello").with_event("greeting")).await;
    /// });
    /// Ok(Response::events(stream.heartbeat(Duration::from_secs(15))))
    /// ```
    #[allow(dead_code)]
    pub fn events(stream: impl BodyStream) -> Self {
        Self::stream(200, "text/event-stream", stream)
            .with_header("cache-control", "no-cache")
            .with_header("x-accel-buffering", "no")
    }

    /// For what the struct literal does not cover: several headers, added one
    /// at a time. Rust has no variadic generics, so a tuple return is
    /// fixed-arity and this is the unbounded form.
    #[must_use]
    pub fn with_header(mut self, name: impl Into<String>, value: impl Into<String>) -> Self {
        self.headers.push((name.into(), value.into()));
        self
    }

    /// Drops every header with this name, ignoring case.
    ///
    /// `with_header` appends, because `set-cookie` has to repeat. A
    /// single-valued header does not: two `content-type` lines are two
    /// claims and the client picks. Replacing one drops it first, which
    /// is what `set_header` below does.
    #[allow(dead_code)]
    #[must_use]
    pub fn without_header(mut self, name: &str) -> Self {
        self.headers
            .retain(|(held, _)| !held.eq_ignore_ascii_case(name));
        self
    }

    /// The one value for this header, replacing any already set.
    ///
    /// What to compose with when overriding a type an earlier step chose:
    /// `Ok(render_json(tpl)?.set_header("content-type", "application/problem+json"))`.
    #[allow(dead_code)]
    #[must_use]
    pub fn set_header(self, name: &str, value: impl Into<String>) -> Self {
        self.without_header(name).with_header(name, value)
    }
}

/// What a handler may return instead of building a [`Response`] by hand.
///
/// The Crystal brocanth overloads `write_response` on the handler's return
/// type, so a page returns the simplest thing that says what it means. Rust
/// spells the same idea as a trait: each handler still returns exactly one
/// type, and this is what lets different handlers return different ones.
pub trait IntoResponse {
    fn into_response(self) -> Response;
}

impl IntoResponse for Response {
    fn into_response(self) -> Response {
        self
    }
}

/// A body, answered 200.
impl IntoResponse for String {
    fn into_response(self) -> Response {
        Response::ok(self)
    }
}

impl IntoResponse for &str {
    fn into_response(self) -> Response {
        Response::ok(self)
    }
}

/// Nothing: 200 with no body. Crystal's `Nil` return.
impl IntoResponse for () {
    fn into_response(self) -> Response {
        Response::empty(200)
    }
}

/// A status and nothing else -- `Ok(404)`.
impl IntoResponse for u16 {
    fn into_response(self) -> Response {
        Response::empty(self)
    }
}

/// A body and a status.
impl<T: IntoResponse> IntoResponse for (T, u16) {
    fn into_response(self) -> Response {
        Response {
            status: self.1,
            ..self.0.into_response()
        }
    }
}

/// A body and headers.
impl<T: IntoResponse> IntoResponse for (T, Vec<(String, String)>) {
    fn into_response(self) -> Response {
        let mut response = self.0.into_response();
        response.headers.extend(self.1);
        response
    }
}

/// A body, a status and headers.
impl<T: IntoResponse> IntoResponse for (T, u16, Vec<(String, String)>) {
    fn into_response(self) -> Response {
        let mut response = self.0.into_response();
        response.status = self.1;
        response.headers.extend(self.2);
        response
    }
}

/// What a handler may fail with.
///
/// Deliberately **not** an `impl std::error::Error`, which is what lets the
/// blanket `From` below exist without colliding with the reflexive
/// `impl<T> From<T> for T`. `anyhow::Error` is this shape for the same reason.
/// The effect is that `?` works on anything in a handler.
#[derive(Debug)]
pub struct Error {
    pub status: u16,
    pub message: String,
    /// Headers the failure needs. A `content-type` for a body that is not
    /// HTML, a `www-authenticate` for a 401, a `retry-after` for a 429.
    ///
    /// `Response` always had these and `Error` did not, which made a
    /// machine-readable error body impossible to send.
    pub headers: Vec<(String, String)>,
}

impl Error {
    pub fn new(status: u16, message: impl Into<String>) -> Self {
        Self {
            status,
            message: message.into(),
            headers: Vec::new(),
        }
    }

    /// The default for a failure nobody classified: it is a bug here, not a
    /// mistake by the caller.
    pub fn internal(message: impl Into<String>) -> Self {
        Self::new(500, message)
    }

    /// The same, carrying a header.
    ///
    /// `allow(dead_code)` for the same reason the request readers below
    /// have it: it is here for the first handler that needs it, and a
    /// brand-new project has none yet.
    #[allow(dead_code)]
    #[must_use]
    pub fn with_header(mut self, name: impl Into<String>, value: impl Into<String>) -> Self {
        self.headers.push((name.into(), value.into()));
        self
    }
}

impl<E: std::error::Error + Send + Sync + 'static> From<E> for Error {
    fn from(error: E) -> Self {
        Self::internal(error.to_string())
    }
}

impl IntoResponse for Error {
    fn into_response(self) -> Response {
        // Everything a client sees for a 5xx is the status. The message is for
        // whoever is reading the logs, and putting it in the body is how an
        // internal path ends up on a stranger's screen.
        let withheld = self.status >= 500;
        if withheld {
            // So the message does reach the logs: a database error that came
            // up through `?` says here what failed, and the client sees none
            // of it.
            let (status, message) = (self.status, &self.message);
            crate::plugins::say!(error, "error {status}: {message}"; status, %message);
        }
        let body = if withheld {
            String::from("Internal Server Error")
        } else {
            self.message
        };
        let mut headers = self.headers;
        if withheld {
            // The body is replaced, so a `content-type` describing the
            // original no longer describes anything. Keeping it labelled a
            // plain sentence `application/problem+json` -- harmless while
            // `sent` sent two types and the client could pick the other,
            // and wrong now that the replacement is exact.
            headers.retain(|(name, _)| !name.eq_ignore_ascii_case("content-type"));
        }
        Response {
            status: self.status,
            body,
            // The rest survive: the body is withheld, not the framing the
            // client needs to read what is left.
            headers,
            stream: None,
        }
    }
}

/// What a handler is given. Middleware sees it too, and may change it before
/// the handler runs or after it returns.
pub struct Ctx {
    /// Which verb matched. A layer reads it; a page usually does not,
    /// because the verb is already in the name of the function it is in.
    #[allow(dead_code)]
    pub verb: Verb,
    pub path: String,
    /// Everything after the first `?`, without it. Empty when there was none.
    #[allow(dead_code)]
    pub query: String,
    /// The body as it arrived. Empty for the verbs that carry none, and for
    /// a context built by a test that did not set one.
    #[allow(dead_code)]
    pub body: Vec<u8>,
    params: Vec<(String, String)>,
    headers: Vec<(String, String)>,
    /// What the plugins put here for this request, or `None` for a context
    /// built by hand.
    plugins: Option<Context<plugins::Installed>>,
    /// What the request hooks wrote for this request: a `CsrfToken`, a
    /// client address. Empty for a context built by hand.
    state: RequestState,
}

impl Ctx {
    pub fn new(verb: Verb, path: &str, params: Params<'_>) -> Self {
        Self {
            verb,
            path: path.to_string(),
            query: String::new(),
            body: Vec::new(),
            params: params
                .iter()
                .map(|(name, value)| (name.to_string(), value.to_string()))
                .collect(),
            headers: Vec::new(),
            plugins: None,
            state: RequestState::new(),
        }
    }

    /// The rest of the request: what the client sent, rather than what the
    /// router matched.
    ///
    /// Two calls and not one, because a test that exercises a handler
    /// usually wants the first and not the second. `src/app.rs` makes both.
    ///
    /// The three are copied out of the server's request, which lives for
    /// the connection and not for the handler. One copy per request, and
    /// the handler owns what it reads.
    #[must_use]
    pub fn sent(mut self, query: &str, body: Vec<u8>, headers: Vec<(String, String)>) -> Self {
        self.query = query.to_string();
        self.body = body;
        self.headers = headers;
        self
    }

    /// What the plugins put on this request's context.
    ///
    /// `src/app.rs` calls it once per request, with a context the
    /// application made from its providers. A `Ctx` built in a test has
    /// none, which is what the `Option` says.
    #[must_use]
    pub fn with_plugins(mut self, plugins: Context<plugins::Installed>) -> Self {
        self.plugins = Some(plugins);
        self
    }

    /// What the request hooks wrote. `src/app.rs` hands it over before the
    /// page runs and takes it back after, for the response hooks.
    #[must_use]
    pub fn with_state(mut self, state: RequestState) -> Self {
        self.state = state;
        self
    }

    /// The hook state back, leaving this context an empty one.
    pub fn take_state(&mut self) -> RequestState {
        std::mem::take(&mut self.state)
    }

    /// The value a plugin published for this request.
    ///
    /// **Compiles only when some registered plugin provides `T`.** Ask for
    /// a type nothing installs and the build stops and names it; there is
    /// no string key to misspell and no guard to forget. That check is the
    /// whole point, and it is the compiler's, not this method's.
    ///
    /// `Option` is the other half, and it is not about `T`'s type. It is
    /// `None` in two cases: a `Ctx` built by hand in a test, which has no
    /// application behind it and so no providers; and a provider that
    /// declared `T` and inserted nothing, which `Plugin::provide` allows
    /// because it is one method and `Provides` is a separate declaration.
    /// A dispatched request whose provider wrote its value gets `Some`.
    ///
    /// The `_` is the position of `T` in the provider list, which is
    /// always inferred. It is there so two impls do not overlap, and it is
    /// the same `_` `Context::get` takes.
    ///
    /// ```ignore
    /// let session = ctx.provided::<Session, _>().expect("dispatched");
    /// ```
    #[allow(dead_code)]
    pub fn provided<T, I>(&self) -> Option<&T>
    where
        T: ContextItem,
        plugins::Installed: Has<T, I>,
    {
        // `and_then` with `try_get`, not `map` with `get`: `get` panics
        // when the provider list claims `T` and the provider inserted
        // nothing, which the generated plugin templates make easy --
        // `Provides` is unconditional and `provide` is often
        // `if let Some(..)`. A method returning `Option` must not abort
        // the process to say no.
        self.plugins.as_ref().and_then(|cx| cx.try_get::<T>())
    }

    /// A route parameter, or `None`.
    ///
    /// Never `""`. The Crystal prototype returned an empty string for a name
    /// nobody declared, so a typo read as a present-but-blank value and went
    /// unnoticed; `Option` is that bug deleted rather than documented.
    pub fn param(&self, name: &str) -> Option<&str> {
        self.params
            .iter()
            .find(|(key, _)| key == name)
            .map(|(_, value)| value.as_str())
    }

    /// The same, for a parameter the route declares and so must be there.
    ///
    /// Returns `Result` rather than `Option` so it composes with `?`. A name
    /// the route does not declare is this file disagreeing with its own
    /// filename, which is a 500 and not a 400: nobody sent a bad request.
    pub fn require(&self, name: &str) -> Result<&str, Error> {
        self.param(name)
            .ok_or_else(|| Error::internal(format!("the route declares no `{name}` parameter")))
    }
}

/// What the client sent, as opposed to what the router matched.
///
/// A second block, and `allow(dead_code)`, for one reason: the request is
/// on the context whether or not this project reads it yet. The other way
/// round is a context whose shape depends on which modules were chosen,
/// and then a page that starts reading a header needs the generator rather
/// than a line.
#[allow(dead_code)]
impl Ctx {
    /// A value a request hook wrote for this request, by its type.
    ///
    /// `ctx.hooked::<brocanth_middleware::CsrfToken>()` is the token a form
    /// carries. `None` when no registered hook wrote one, and always for a
    /// context built by hand.
    pub fn hooked<T: Send + Sync + 'static>(&self) -> Option<&T> {
        self.state.get::<T>()
    }

    /// The same value, to change. `None` when no hook wrote one.
    ///
    /// The state goes back to the hooks when the page returns, so a change
    /// made here is what `after_dispatch` and the response hooks read: a
    /// request hook puts a per-request log record on the state, and the
    /// page fills in the fields only it knows.
    pub fn hooked_mut<T: Send + Sync + 'static>(&mut self) -> Option<&mut T> {
        self.state.get_mut::<T>()
    }

    /// Puts a value on the request state for the hooks that run after the
    /// page, and returns the value it replaced.
    ///
    /// For a value that no request hook wrote first. A hook that reads it
    /// must also accept `None`: a 404, a 405 and a request that a request
    /// hook answered run no page.
    pub fn hook_insert<T: Send + Sync + 'static>(&mut self, value: T) -> Option<T> {
        self.state.insert(value)
    }

    /// A request header by name, case-insensitively, or `None`.
    pub fn header(&self, name: &str) -> Option<&str> {
        self.headers
            .iter()
            .find(|(key, _)| key.eq_ignore_ascii_case(name))
            .map(|(_, value)| value.as_str())
    }

    /// Every request header, in the order they arrived.
    pub fn headers(&self) -> impl Iterator<Item = (&str, &str)> {
        self.headers
            .iter()
            .map(|(key, value)| (key.as_str(), value.as_str()))
    }

    /// What the body is written in, or `None` when the client did not say.
    pub fn content_type(&self) -> Option<&str> {
        self.header("content-type")
    }

    /// The body as text, or `None` when it is not UTF-8.
    ///
    /// `Option` and not a lossy conversion: a body that is not text is a
    /// request to answer, not a string to repair.
    pub fn body_text(&self) -> Option<&str> {
        core::str::from_utf8(&self.body).ok()
    }
}

/// A handler, as the table stores it. One per verb a page answers.
///
/// Boxed rather than a bare `fn` pointer, because every handler returns its
/// own type and a `fn` pointer can only hold one of them. [`page`] and
/// [`page_async`] are what unify them, and they run once per route at
/// startup rather than per request.
///
/// It returns a future, so a page may wait -- on a database, on another
/// service -- without holding a thread. A page that waits on nothing is
/// still a plain `fn`, and [`page`] hands back its answer already finished.
pub type Handler =
    Box<dyn for<'a> Fn(&'a mut Ctx) -> BoxFuture<'a, Result<Response, Error>> + Send + Sync>;

/// Puts a `pub fn handle` in the table whatever it returns.
///
/// Generic over the return type, so `-> Result<impl IntoResponse, Error>` on
/// the handler and one concrete `Handler` in the table are the same thing.
/// The answer is ready when the handler returns; the future around it is
/// only there so a sync page and an async page share one table.
pub fn page<H, T>(handler: H) -> Handler
where
    H: Fn(&mut Ctx) -> Result<T, Error> + Send + Sync + 'static,
    T: IntoResponse,
{
    Box::new(move |ctx| {
        let answered = handler(ctx).map(IntoResponse::into_response);
        Box::pin(std::future::ready(answered))
    })
}

/// Puts a `pub async fn handle` in the table.
///
/// `pages!` writes the call, with a closure that awaits the page and turns
/// its answer into a `Response`. The closure is a `fn` pointer and not a
/// generic, because the page's future borrows the context and a generic
/// cannot name that borrow.
///
/// `allow(dead_code)`: nothing calls it until a page is `async`.
#[allow(dead_code)]
pub fn page_async(
    handler: for<'a> fn(&'a mut Ctx) -> BoxFuture<'a, Result<Response, Error>>,
) -> Handler {
    Box::new(handler)
}

/// Every template in this application.
///
/// The `.rsx` files sit next to the pages they render -- `index.rsx` beside
/// `index.rs` -- and `load_dir!` compiles the tree while this crate is being
/// compiled. Nothing is read from disk when a request arrives, and a template

///
/// A module of their own, because `load_dir!` emits a `pub mod index` for
/// `index.rsx` and `pages!` emits one for `index.rs`: unnested they collide.
/// The path is relative to the package root and not to this file, because a
/// proc macro cannot learn where it was written on stable Rust.
///
/// A file whose name is not an identifier gets the nearest one: `[id].rsx` is
/// `views::users::_id_`.
pub mod views {

/// Renders a page inside the layouts that cover it.
///
/// `layout.rsx` wraps every page in its directory and in every directory
/// below it, and they stack from the root down. A page never names one,
/// because which layout covers a page is a property of where the page is --
/// the rule Crystal's page_router uses for `layout.cr`.
///
/// The macro reads that off the directory tree at compile time: it knows
/// which file it was written in, walks up to `src/`, and wraps the page in
/// every `layout.rsx` it passes. Add one to a directory and every page under
/// it is wrapped, with nothing to register and no chain to keep in step.
///
/// Re-exported here so a page imports it from the same place as everything
/// else it uses, and so the layout rule lives beside the route table rather
/// than in a crate name.
pub use brocanth_pages::html;

/// The URL prefix a layout covers, for a layout under a parameter.
///
/// `html!` passes a layout the prefix it wraps, which is the marker dp-swap
/// reads to work out how much of a page two URLs share. A directory named
/// after a parameter has no prefix of its own until a request arrives, so
/// the macro hands this function the template and the request fills it:
/// `/{section}` under `/alpha/card-2` is `/alpha`.
///
/// Called only from the expansion, and only for a layout that sits under a
/// parameter. A marker reading `/[section]` would be the same string on
/// every section, and the client would conclude that two pages share a
/// layer they do not.
/// `allow(dead_code)`: `html!` emits a call to this only for a layout that
/// defines an `Oob` component, and no scaffolded layout does. It is here
/// for the first one that needs it, which is the same reason the request
/// readers on `Ctx` carry the attribute.
#[allow(dead_code)]
pub fn prefix(ctx: &Ctx, template: &str) -> String {
    let mut out = String::with_capacity(template.len());
    let mut rest = template;
    while let Some(open) = rest.find('{') {
        out.push_str(&rest[..open]);
        let Some(close) = rest[open..].find('}') else {
            break;
        };
        // A parameter the router did not match cannot happen: this template
        // is built from the directory the route was read out of, so the
        // name is one the pattern declares.
        out.push_str(ctx.param(&rest[open + 1..open + close]).unwrap_or_default());
        rest = &rest[open + close + 1..];
    }
    out.push_str(rest);
    out
}

/// A partial, and the replacements for the chrome it left behind.
///
/// A layout the client keeps is a layout this answer does not contain, which
/// is the point of asking for a partial and the reason its chrome goes
/// stale: the sidebar still marks the section you came from. A layout that
/// has something to correct defines `Oob` beside its `Layout`, and `html!`
/// renders one per skipped layer and passes them here.
///
/// They lead, because the static version of this writes them that way. The
/// client reads the whole body before it swaps anything, so the order is a
/// convention rather than a requirement, and one convention is better than
/// two.
/// `allow(dead_code)`: `html!` emits a call to this only for a layout that
/// defines an `Oob` component, and no scaffolded layout does. It is here
/// for the first one that needs it, which is the same reason the request
/// readers on `Ctx` carry the attribute.
#[allow(dead_code)]
pub fn render_with(
    out_of_band: Vec<String>,
    content: impl rsx::Template,
) -> Result<Response, Error> {
    let mut body = out_of_band.concat();
    body.push_str(&content.render_to_string()?);
    Ok(Response::ok(body))
}

/// One skipped layout's replacement chrome, rendered on its own.
///
/// `html!` calls this once per layer it left out, and `render_with` puts
/// the results in front of the page. A function here rather than a call to
/// `rsx::Template` inside the macro: the other two renders already come
/// back through this file, and the macro naming one engine was the reason
/// a project could have only that engine.
/// `allow(dead_code)`: `html!` emits a call to this only for a layout that
/// defines an `Oob` component, and no scaffolded layout does. It is here
/// for the first one that needs it, which is the same reason the request
/// readers on `Ctx` carry the attribute.
#[allow(dead_code)]
pub fn render_oob(content: impl rsx::Template) -> Result<String, Error> {
    Ok(content.render_to_string()?)
}

/// A page with no layout at all: a fragment, a partial, an htmx swap.
///
/// What `html!` expands into, once it has wrapped the page, so this is also
/// the escape hatch -- name a layout by hand and pass it here.
///
/// It returns the handler's own `Result` rather than a value, so a template
/// that bails propagates with `?` instead of turning into a blank 200. For
/// another status or a header, compose with the forms above:
/// `Ok((html!(ctx, tpl)?, 404))`, or `Ok(html!(ctx, tpl)?.with_header("x-kind", "page"))`.
///
/// Deliberately a function and not `impl<T: rsx::Template> IntoResponse for T`.
/// `()`, `Option<T>` and `Vec<T>` are templates, so a blanket impl overlaps
/// the ones above and does not compile -- and an `into_response` that cannot
/// fail would have to swallow a render error. A blanket impl could not apply
/// a layout either, since it would not know which page it was rendering.
pub fn render(content: impl rsx::Template) -> Result<Response, Error> {
    // `render_to_string` builds into its own String, so a page that fails
    // half way through its layout leaves nothing behind. There is no partial
    // body to take back, which is the property the rest of this file is
    // built on.
    Ok(Response::ok(content.render_to_string()?))
}

impl Ctx {
    /// How many layout layers the client already holds. `html!` asks it.
    ///
    /// Always zero: nothing here sends a partial request, so every page
    /// goes out whole. `brocanth add rsx-htmx` reads `DP-Partial` here.
    pub fn partial_depth(&self) -> u32 {
        0
    }
}

// -- the entrance ----------------------------------------------------------
//
// Written because this project chose `input`. A body arrives as bytes and
// leaves as a struct, or as every reason it could not.

/// A body the framework reads and checks before the handler runs.
///
/// `pages!` registers a handler through this one when its signature takes
/// a second parameter, so the page cannot forget to check and cannot
/// check twice.
///
/// `allow(dead_code)`: nothing references it until a page declares a body
/// or a `Json<T>` answer, and a brand-new project has neither. It is here
/// for the first one that does, like the request readers on `Ctx`.
#[allow(dead_code)]
pub fn page_with_body<H, B, T>(handler: H) -> Handler
where
    H: Fn(&mut Ctx, B) -> Result<T, Error> + Send + Sync + 'static,
    B: zod::Entry,
    T: IntoResponse,
{
    Box::new(move |ctx| {
        let answered = ctx
            .body_as::<B>()
            .and_then(|body| handler(ctx, body).map(IntoResponse::into_response));
        Box::pin(std::future::ready(answered))
    })
}

/// [`page_with_body`] for a `pub async fn handle_post(ctx, body)`.
///
/// The body is read and checked before the page's future exists, so a
/// refused body never reaches the page. `pages!` writes the call.
///
/// `allow(dead_code)`: nothing calls it until a page that takes a body is
/// `async`.
#[allow(dead_code)]
pub fn page_with_body_async<B>(
    handler: for<'a> fn(&'a mut Ctx, B) -> BoxFuture<'a, Result<Response, Error>>,
) -> Handler
where
    B: zod::Entry + 'static,
{
    Box::new(move |ctx| match ctx.body_as::<B>() {
        Ok(body) => handler(ctx, body),
        Err(refused) => Box::pin(std::future::ready(Err(refused))),
    })
}

/// An answer sent as JSON, and described in the document.
///
/// Wrapping the return type is the opt-in: a page that writes
/// `-> Result<impl IntoResponse, Error>` keeps working and says nothing
/// about its answer, and one that writes `-> Result<Json<Created>, Error>`
/// publishes `Created` as its response schema. The wrapper is the only
/// place that difference is spelled.
///
/// `allow(dead_code)`: nothing references it until a page declares a body
/// or a `Json<T>` answer, and a brand-new project has neither. It is here
/// for the first one that does, like the request readers on `Ctx`.
#[allow(dead_code)]
pub struct Json<T>(pub T);

impl<T: zod::Give> IntoResponse for Json<T> {
    fn into_response(self) -> Response {
        Response::new(200, zod::write(&self.0.give()))
            .with_header("content-type", "application/json")
    }
}

/// What came of reading a body.
///
/// Three answers and not two, because they are three different statuses. A
/// body that broke a rule is 422 and the client can fix it. A body nobody
/// could read is 400 or 415, and no amount of fixing the fields will help.
///
/// `allow(dead_code)` on this and on what follows: they are written for
/// the first handler that reads a body, and a brand-new project has none.
/// A scaffold that hands a developer three warnings on their first
/// `cargo clippy` is a scaffold that got it wrong.
#[allow(dead_code)]
pub enum Sent<T> {
    /// It read, and every rule passed.
    Good(T),
    /// It read, and these rules did not. Draw the form again, with
    /// `faults.by_name()` beside the inputs that caused them.
    Faulty(zod::Faults),
    /// Nobody could read it at all.
    Unreadable(brocanth_input::Unreadable),
}

#[allow(dead_code)]
impl Ctx {
    /// The three parts of the request a schema reads: the content type,
    /// the query string and the body.
    pub fn entrance(&self) -> brocanth_input::Entrance<'_> {
        brocanth_input::Entrance::new(self.content_type(), &self.query, &self.body)
    }

    /// Read the body into `T`, keeping every answer apart.
    ///
    /// `T` derives `zod::Entry`, which puts the rules on the struct:
    ///
    /// ```ignore
    /// #[derive(zod::Entry)]
    /// struct Signup {
    ///     #[check(zod::Text::new().min(3).max(16).slug())]
    ///     #[label("Handle")]
    ///     #[says("Pick 3 to 16 lowercase letters, digits or hyphens")]
    ///     handle: String,
    ///
    ///     #[check(zod::Int::new().from(13).to(130))]
    ///     age: i64,
    /// }
    /// ```
    ///
    /// A form and a JSON body both work, and the content type decides how
    /// text is read: a form has no types to lose, so `age=36` is a number
    /// there and is text in JSON.
    pub fn read<T: zod::Entry>(&self) -> Sent<T> {
        match self.entrance().input() {
            Err(why) => Sent::Unreadable(why),
            Ok(input) => match T::parse(input) {
                Ok(value) => Sent::Good(value),
                Err(faults) => Sent::Faulty(faults),
            },
        }
    }

    /// The same, as one `Result` a handler can use `?` on.
    ///
    /// ```ignore
    /// pub fn handle_post(ctx: &mut Ctx) -> Result<impl IntoResponse, Error> {
    ///     let signup: Signup = ctx.body_as()?;
    ///     Ok(Response::empty(303).with_header("location", "/"))
    /// }
    /// ```
    pub fn body_as<T: zod::Entry>(&self) -> Result<T, Error> {
        match self.read() {
            Sent::Good(value) => Ok(value),
            Sent::Faulty(faults) => Err(refused(&faults)),
            Sent::Unreadable(why) => Err(Error::new(why.status(), why.to_string())),
        }
    }

    /// Read the query string into `T`. The same rules, from `?page=2`.
    pub fn query_as<T: zod::Entry>(&self) -> Result<T, Error> {
        T::parse(self.entrance().asked()).map_err(|faults| refused(&faults))
    }
}

/// Every fault, as the answer this application sends.
///
/// One line per field. An HTML application usually wants the form back with
/// the messages beside the inputs, which only the handler can draw: match
/// on [`Sent::Faulty`] there and render `faults.by_name()`. This is what a
/// handler that has not done that yet sends.
///
/// Not a `From<zod::Faults> for Error`. `Faults` is a `std::error::Error`,
/// so the blanket `From` above already covers it -- and that one answers
/// 500, which is why a handler calls `body_as` rather than `?` on a
/// `Faults`.
#[allow(dead_code)]
pub fn refused(faults: &zod::Faults) -> Error {
    let lines: Vec<String> = faults
        .by_name()
        .into_iter()
        .map(|(name, says)| format!("{name}: {}", says.join("; ")))
        .collect();
    Error::new(422, lines.join("\n"))
}

/// Every plugin this application contains, in dependency order.
///
/// A macro and not a function, and the reason is the builder's type:
/// registering a plugin that provides a context item widens
/// `AppBuilder`'s type-level provider list, so the type this chain ends
/// on is one nested associated type per plugin. A function would have to
/// spell that type out and re-spell it on every change. The macro expands
/// at its one call site in `src/app.rs`, where the compiler works it out
/// and nobody has to read it.
///
/// Each registration is still one line, and a line a reader can delete.
macro_rules! register {
    ($builder:expr) => {{
        use brocanth_config::AppBuilderConfigExt;

        // The files, the environment and, where src/modules/ turns them
        // on, the flags: `brocanth-config` layers them and types each
        // section, and it is the only reason a TOML parser is in the tree.
        let config = crate::plugins::config()?;
        let builder = $builder.with_config(config);

        // -- brocanth-middleware ------------------------------------------
        //
        // Five edge hooks, none of them on. Each encodes a fact about this
        // deployment, so each is turned on by somebody who knows it.
        //
        // This file is generated, so do not uncomment them here. Copy the
        // line to `build()` in `src/app.rs`, after `plugins::register!`,
        // and widen `type Core` there by the plugin's hook families:
        //
        //     brocanth::plugin_hooks::Both<
        //         <brocanth_middleware::Cors as brocanth::plugin_hooks::HookSource>::Families,
        //         plugins::Hooks,
        //     >

        // Who the client is, behind a proxy. The list is the addresses of
        // YOUR load balancers: `X-Forwarded-For` is a claim anybody who
        // reaches the socket may make, so there is no "trust everything".
        // let builder = builder.plugin_hooking(
        //     <brocanth_middleware::ProxyHeaders as brocanth::plugin_api::Plugin>::new(
        //         brocanth_middleware::ProxyHeadersConfig::trusting([
        //             "10.0.0.1".parse().expect("an address"),
        //         ]),
        //     ),
        // );

        // A DELETE sent from an HTML form, through `_method` or
        // `X-HTTP-Method-Override`. Only a POST is overridden, and only
        // into PUT, PATCH or DELETE.
        // let builder = builder.plugin_hooking(
        //     <brocanth_middleware::MethodOverride as brocanth::plugin_api::Plugin>::new(
        //         brocanth_middleware::MethodOverrideConfig::default(),
        //     ),
        // );

        // One host, so cookies and caches agree. 308 keeps the verb; 301
        // is what crawlers prefer and may turn a POST into a GET.
        // let builder = builder.plugin_hooking(
        //     <brocanth_middleware::CanonicalHost as brocanth::plugin_api::Plugin>::new(
        //         brocanth_middleware::CanonicalHostConfig::to("https://example.com"),

        //     ),
        // );

        //
        // Messages for the next page, in a signed cookie. A page writes
        // `ctx.hooked_mut::<brocanth_middleware::Flash>()` and redirects;

    /// What it sets.
    pub setting: &'static Setting,
    /// Where it came from, as typed: `--cookie`, or `$MY_APP_SESSIONS__COOKIE`.

/// The name each flag answers to after the renames in src/modules/, whether
/// the flags are read or not.
///
/// A name that is not spelled as a flag, and two flags under one name, each
/// stop the binary: each would otherwise be a flag that does nothing, or does
/// something else. A rename of a key no flag sets does not compile: `Key` has
/// no variant for it.
fn names() -> Result<Vec<(&'static Setting, &'static str)>, String> {
    let renames = renames();
    let mut names: Vec<(&'static Setting, &'static str)> = Vec::new();
    for setting in SETTINGS.iter().filter(|setting| setting.flag) {
        let name = renames
            .iter()
            .find(|(path, _)| *path == setting.path)
            .map_or(setting.path, |(_, flag)| flag.name());
        if name.is_empty() {
            continue;
        }
        let spelled = name
            .chars()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || "._-".contains(c));
        let taken = ["help", "version", "settings"].contains(&name);
        if !spelled || name.starts_with(['-', '.', '_']) || taken {
            return Err(format!(
                "{} names a flag `{name}`\nhelp: a flag is lower case, digits, `.`, `_` and `-`, starts with a letter or a digit, and is not `help`, `version` or `settings`",
                setting.file
            ));
        }
        if let Some((other, _)) = names.iter().find(|(_, taken)| *taken == name) {
            let files = if other.file == setting.file {
                other.file.to_string()
            } else {
                format!("{} or {}", other.file, setting.file)
            };
            return Err(format!(
                "`--{name}` would set both `{}` and `{}`\nhelp: rename one of them in {files}",
                other.path, setting.path
            ));
        }
        names.push((setting, name));
    }
    Ok(names)
}

/// Whether a module's flags are read: its file says `Flags::On` or
/// `Flags::Off`, or follows the project rule. With a config file, the file is
/// the source until src/modules/brocanth.rs turns the flags on.
fn flags_on(rule: Flags) -> bool {
    match rule {
        Flags::On => true,
        Flags::Off => false,
        Flags::Project => crate::modules::brocanth::MODULE.flags_with_config,
    }
}

/// The flags this run reads: each module file decides, and
/// src/modules/brocanth.rs holds the project rule. Every name is checked
/// first, read or not.
pub fn flag_names() -> Result<Vec<(&'static Setting, &'static str)>, String> {
    Ok(names()?
        .into_iter()
        .filter(|(setting, _)| flags_on(setting.flags))
        .collect())
}

/// A flag given while the flags are off, refused by name with what to do
/// instead. Silently ignoring it would run with a value nobody chose.
pub fn refuse_off_flags() -> Result<(), String> {
    let names = names()?;
    for arg in std::env::args_os().skip(1) {
        let arg = arg.to_string_lossy();
        if arg == "--" {
            break;
        }
        let Some(written) = arg.strip_prefix("--") else {
            continue;
        };
        let name = written.split_once('=').map_or(written, |(name, _)| name);
        let Some((setting, _)) = names.iter().find(|(_, known)| *known == name) else {
            continue;
        };
        if flags_on(setting.flags) {
            continue;
        }
        let (section, key) = setting.path.rsplit_once('.').unwrap_or(("", setting.path));
        let fix = if setting.file.ends_with("/brocanth.rs") {
            "flags_with_config: true"
        } else {
            "flags: Flags::On"
        };
        return Err(format!(
            "`--{name}` is off: this project reads its settings from config/<env>.toml\nhelp: set `{fix}` in {}, or set `{key}` under [{section}] in config/<env>.toml",
            setting.file
        ));
    }
    Ok(())
}

/// The flags this run was given, in the order typed. The environment is not
/// here: `brocanth-config` reads it, under the file and over the flags.
///
/// `--name=VALUE` and `--name VALUE`. Every other argument is left alone: it
/// is a command's, and the command line's own parser has already refused what
/// nobody declared.
pub fn given() -> Result<Vec<Given>, String> {
    let mut given = Vec::new();
    let names = flag_names()?;
    let args: Vec<String> = std::env::args_os()
        .skip(1)
        .map(|arg| arg.to_string_lossy().into_owned())
        .collect();
    let mut rest = args.iter();
    while let Some(arg) = rest.next() {
        if arg == "--" {
            break;
        }
        let Some(written) = arg.strip_prefix("--") else {
            continue;
        };
        let (name, inline) = match written.split_once('=') {
            Some((name, value)) => (name, Some(value.to_string())),
            None => (written, None),
        };
        let Some((setting, name)) = names.iter().copied().find(|(_, known)| *known == name) else {
            continue;
        };
        let value = match inline {
            Some(value) => value,
            None => rest
                .next()
                .cloned()
                .ok_or_else(|| format!("--{name} needs a value\nhelp: write --{name}=VALUE"))?,
        };
        given.push(Given {
            setting,
            from: format!("--{name}"),
            value,
        });
    }
    Ok(given)
}

/// The configuration this run reads: the files, the environment, and the
/// flags on top where src/modules/ turns them on.
pub fn config() -> Result<brocanth_config::Config, String> {
    refuse_off_flags()?;
    let mut loader = brocanth_config::Loader::new(crate::app::APP);
    for setting in SETTINGS {
        if let Some(alias) = setting.none_as {
            loader = loader.none_as(setting.path, alias);
        }
    }
    for given in given()? {
        let flag = given.from.trim_start_matches('-');
        loader = loader.flag(given.setting.path, flag, &given.value, given.setting.text);
    }
    loader.load().map_err(|e| e.to_string())
}

/// Refuses anything on the command line that is not a flag, and answers
/// `--help` and `--settings`. `true` when it printed one of those, so `main`
/// stops there.
///
/// This binary has no command line of its own: `brocanth add commands` gives
/// it one, and `clap` does this.
pub fn read_line() -> Result<bool, String> {
    refuse_off_flags()?;
    let names = flag_names()?;
    let mut rest = std::env::args_os()
        .skip(1)
        .map(|arg| arg.to_string_lossy().into_owned());
    while let Some(arg) = rest.next() {
        if arg == "--help" || arg == "-h" {
            println!("Flags:");
            for (setting, name) in &names {
                println!("  --{name} <VALUE>  {}", setting.help);
            }
            println!("  --settings  every settings layer, then the merged values");
            return Ok(true);
        }
        if arg == "--settings" {
            print!("{}", settings_report()?);
            return Ok(true);
        }
        let written = arg.strip_prefix("--").unwrap_or_default();
        let name = written.split_once('=').map_or(written, |(name, _)| name);
        if name.is_empty() || !names.iter().any(|(_, known)| *known == name) {
            return Err(format!(
                "unknown argument `{arg}`\nhelp: `--help` lists the flags"
            ));
        }
        // `--name VALUE`: the next argument is the value, not a flag.
        if !written.contains('=') {
            rest.next();
        }
    }
    Ok(false)
}

/// Every settings layer on its own, lowest first, then the merged values.
/// A secret is masked in every layer.
pub fn settings_report() -> Result<String, String> {
    let config = config()?;
    let mut layers = config.layers().defaults(defaults());
    for setting in SETTINGS.iter().filter(|setting| !setting.flag) {
        layers = layers.secret(setting.path);
    }
    Ok(layers.to_string())
}

/// `[server] shutdown_grace_secs` from the files, the environment and the
/// flags, or [`SHUTDOWN_GRACE`](crate::app::SHUTDOWN_GRACE) when none sets it.
pub fn shutdown_grace() -> Result<std::time::Duration, String> {
    let config = config()?;
    let Some(value) = config
        .raw_section("server")
        .and_then(|server| server.get("shutdown_grace_secs"))
    else {
        return Ok(crate::app::SHUTDOWN_GRACE);
    };
    // An integer in a file, a flag or a variable; a string when a layer
    // could not type it.
    value
        .as_integer()
        .and_then(|secs| u64::try_from(secs).ok())
        .or_else(|| value.as_str().and_then(|secs| secs.trim().parse().ok()))
        .map(std::time::Duration::from_secs)
        .ok_or_else(|| {
            format!(
                "`[server] shutdown_grace_secs` is {}, and it must be a whole number of seconds\nhelp: set it to a whole number of seconds, such as 25",
                value.display()
            )
        })
}

/// What `serve` answers a request with: `next`, as it is. `brocanth add
/// observe` logs and counts each request here, and answers the probes.
pub fn wrap<F, Fut>(
    app: &std::sync::Arc<crate::app::App>,
    next: F,
) -> impl Fn(brocanth_server::Request) -> Answer + Send + Sync + 'static
where
    F: Fn(brocanth_server::Request) -> Fut + Send + Sync + 'static,
    Fut: std::future::Future<Output = brocanth_server::Response> + Send + 'static,

    let _ = app;
    move |request| -> Answer { Box::pin(next(request)) }
}

//! `/account/code` -- the second factor, after the password passed.
//!
//! The password's proof waits in the `auth_pending` cookie, sealed with
//! the session key and good for a few minutes. Six digits are a TOTP code;
//! anything longer is a recovery code.

use brocanth_auth::AuthError;
use brocanth_middleware::Level;

use crate::pages::{html, views, Ctx, Error, Response};
use crate::plugins::auth_screens::{self as screens, csrf_field, see_other, PENDING};

#[derive(zod::Entry)]
pub struct Presented {
    #[check(zod::Text::new().min(6).max(64))]
    code: String,
}

pub fn handle(ctx: &mut Ctx) -> Result<Response, Error> {
    again(ctx, "", 200)
}

pub async fn handle_post(ctx: &mut Ctx) -> Result<Response, Error> {
    let auth = screens::services(ctx)?;
    let Some(proof) =
        screens::cookie(ctx, PENDING).and_then(|sealed| auth.sessions().unseal_proof(&sealed))
    else {
        screens::flash(ctx, Level::Error, "That took too long. Sign in again.");
        return Ok(see_other("/account/signin"));
    };
    let Some(form) = screens::form::<Presented>(ctx)? else {
        return again(ctx, "Enter the code your app shows.", 422);
    };
    let code: String = form.code.chars().filter(|c| !c.is_whitespace()).collect();
    let mfa = auth.mfa()?;
    let account = proof.account().clone();
    let second = if code.len() == 6 && code.bytes().all(|b| b.is_ascii_digit()) {
        mfa.verify_totp(&account, &code).await
    } else {
        mfa.verify_recovery_code(&account, &code)
            .await
            .map(|accepted| accepted.proof)
    };
    let second = match second {
        Ok(second) => second,
        Err(AuthError::InvalidCredentials) => {
            return again(ctx, "That code is wrong or already used.", 422)
        }
        Err(other) => return Err(other.into()),
    };
    screens::admit(ctx, &auth, proof.and(second)?).await
}

fn again(ctx: &mut Ctx, error: &str, status: u16) -> Result<Response, Error> {
    let csrf = csrf_field(ctx);
    let mut page = html!(ctx, views::account::screens::Code { csrf: &csrf, error })?;
    page.status = status;
    Ok(page)
}

//! `/account/signin` -- a password, then a second factor if the account has
//! one.

use brocanth_auth::Attempt;

use crate::pages::{html, views, Ctx, Error, Response};
use crate::plugins::auth_screens::{self as screens, csrf_field};

#[derive(zod::Entry)]
pub struct Credentials {
    #[check(zod::Text::new().min(1).max(254))]
    email: String,
    #[check(zod::Text::new().min(1).max(1024))]
    password: String,
}

/// One sentence for a wrong password, an unknown address and an account
/// with no password: which one it was is what an attacker wants.
const WRONG: &str = "That email and password do not match an account.";

pub fn handle(ctx: &mut Ctx) -> Result<Response, Error> {
    again(ctx, "", "", 200)
}

pub async fn handle_post(ctx: &mut Ctx) -> Result<Response, Error> {
    let auth = screens::services(ctx)?;
    let Some(form) = screens::form::<Credentials>(ctx)? else {
        return again(ctx, "", "Enter your email and your password.", 422);
    };
    let lockout = crate::plugins::auth_lockout::lockout(&auth)?;
    let attempt = lockout
        .login(auth.accounts(), &form.email, &form.password)
        .await?;
    let proof = match attempt {
        Attempt::Admitted(signed) => signed.into_proof(),
        Attempt::Refused => return again(ctx, &form.email, WRONG, 422),
        Attempt::JustLocked { account, .. } => {
            screens::mail_locked(ctx, &account, lockout.policy().lock_for()).await?;
            return again(ctx, &form.email, WRONG, 422);
        }
    };
    screens::admit(ctx, &auth, proof).await
}

fn again(ctx: &mut Ctx, email: &str, error: &str, status: u16) -> Result<Response, Error> {
    let csrf = csrf_field(ctx);
    let mut page = html!(
        ctx,
        views::account::screens::Signin {
            csrf: &csrf,
            email,
            error
        }
    )?;
    page.status = status;
    Ok(page)
}

//!
//! The GET draws a new key and stores it unconfirmed; the POST confirms it
//! with a code from the app and shows the recovery codes, once.

use brocanth_auth::{AccountHolder, AuthError};
use brocanth_middleware::Level;

use crate::pages::{html, views, Ctx, Error, Response};
use crate::plugins::auth_screens::{self as screens, csrf_field, see_other};

pub struct Confirm {
    #[check(zod::Text::new().min(6).max(16))]
    code: String,
}

pub async fn handle(ctx: &mut Ctx) -> Result<Response, Error> {
    let auth = screens::services(ctx)?;
    let links = screens::screens(ctx)?;
    let Some(session) = screens::signed_in(ctx, &auth).await else {
        screens::flash(ctx, Level::Info, "Sign in first.");
        return Ok(see_other("/account/signin"));
    };
    let holder = AccountHolder::of(&session)?;
    let label = screens::email_of(&auth, &session).await?;
    let enrolment = auth
        .mfa()?
        .begin_totp_enrolment(&holder, links.issuer(), &label)
        .await?;

    page(ctx, &secret, &enrolment.uri, "", &[], 200)
}

pub async fn handle_post(ctx: &mut Ctx) -> Result<Response, Error> {
    let auth = screens::services(ctx)?;
    let Some(session) = screens::signed_in(ctx, &auth).await else {
        screens::flash(ctx, Level::Info, "Sign in first.");
        return Ok(see_other("/account/signin"));
    };
    let holder = AccountHolder::of(&session)?;
    let Some(form) = screens::form::<Confirm>(ctx)? else {
        return page(
            ctx,
            "",
            "",
            "Enter the six digits your app shows.",
            &[],
            422,
        );
    };

        .mfa()?
        .confirm_totp_enrolment(&holder, form.code.trim())

            ctx,
            "",
            "",
            "That code does not match. Check the clock of your device, or start again.",
            &[],
            422,
        ),
        Err(other) => Err(other.into()),
    }
}

===== src/plugins/auth/mod.rs =====
//! Accounts and sessions, as this application registers them.
//!
//! One interface, no providers yet, and a local account path that does not
//! leak who has one: `register`, `login`, `request_password_reset` and
//! `verify_email` answer the same shape and take the same time whether or not
//! the address exists, because the expensive hash runs on both paths.
//!
//! `brocanth-auth` is in `Cargo.toml` because this file is here, and through
//! no other route. Delete the registration in `src/plugins/mod.rs` and
//! delete this file, and the application has no accounts and links nothing
//! to hash a password with.
//!
//! # Signing somebody in
//!
//! Everything is behind one context item, and a handler in an application
//! that never registered this plugin does not compile:
//!
//! ```ignore
//! use brocanth_auth::mfa::{Admission, Enrolled, FactorPolicy, FactorSet};
//!
//! let auth = ctx.provided::<Services, _>().expect("a dispatched request");
//!
//! // 1. The credential. `login` answers the same shape and takes the same
//! //    time whether or not the address has an account.
//! let who = auth.accounts().login(email, password).await?;
//!
//! // 2. What this account could prove. Nothing: `Mfa` needs a store trait
//! //    this application does not implement, so the enrolment is empty and
//! //    the constructor is named so that a reviewer stops on it. The day
//! //    there is an `MfaStore`, `Mfa::second_factors` answers this properly
//! //    and this line goes.
//! let enrolled =
//!     Enrolled::without_asking_the_store(who.account().id.clone(), FactorSet::empty());
//!
//! // 3. The policy, which consumes the proof. `Admitted` has no public
//! //    constructor, so this step cannot be skipped -- a sign-in that
//! //    forgot the second factor is a program that does not compile rather
//! //    than one that ships.
//! let Admission::Admit(admitted) =
//!     FactorPolicy::default().admit(&enrolled, who.into_proof())
//! else {
//!     return Err(Error::new(401, "sign in"));
//! };
//!
//! // 4. The session, which takes the admission by value.
//! let started = auth.sessions().begin_with_factors(admitted, None).await?;
//! // `started.cookie` is the `Set-Cookie` header to send back.
//! ```
//!
//! **A page cannot do that yet.** A handler here is
//! `Box<dyn Fn(&mut Ctx) -> Result<Response, Error>>`, which is synchronous,
//! and every method above is `async fn`. What runs today is a test, a
//! `manage` subcommand, and anything else of yours that already has a
//! runtime. The sign-in form arrives with the core's request hooks, which are
//! also what the two middleware offers below are waiting for.
//!
//! # Three decisions this file makes, each in one line
//!
//! They are type aliases so that changing one changes one line and nothing
//! else in the project changes type.
//!
//! 1. **`Store = MemoryStore`.** It is a `Mutex<HashMap>`. Accounts and
//!    sessions last exactly as long as the process, so a restart is a logout
//!    for everybody and a deploy loses every account. That is the one thing
//!    in this file to replace before anybody else uses this application: write
//!    an `AuthStore` over your database -- fourteen methods, of which the two
//!    that carry a requirement are a unique email and a `consume_token` that
//!    deletes and returns under one lock -- and change the line below.
//!
//!    When you do, the build moves: a pool is opened asynchronously and no
//!    stage may wait for it, so `configure` becomes `before_start`, builds the
//!    pool lazily and registers a readiness check that `serve` awaits before
//!    it listens. `brocanth-cache` does it in about ten lines and they are
//!    worth copying rather than inventing.
//! 2. **`Http = StubHttp`.** There are no providers, so nothing in this
//!    binary ever performs a provider request, and a client that answers "no
//!    stubbed route" is a louder failure than one that opens a socket. It is
//!    also the line that keeps reqwest, rustls and their trees out of the lock
//!    file. Adding a provider changes it; see below.
//! 3. **`Passwords::recommended()`, not a setting.** The hash is argon2id and
//!    there is no choice to make. The one question a deployment gets to answer
//!    is cost, and it is a named constructor rather than three numbers in a
//!    file, because three numbers in a file is how an application ends up
//!    hashing at a cost nobody chose.
//!
//! # Adding a provider
//!
//! Three steps, and the flow code never learns the provider's name.
//!
//! 1. In `Cargo.toml`, `brocanth-auth = { ..., features = ["reqwest-client"] }`.
//! 2. Here, `pub type Http = brocanth_auth::http_client::ReqwestClient;`.
//! 3. Here, in `configure`, push one `ConfiguredProvider`:
//!
//! ```ignore
//! use brocanth_auth::oauth::ClientConfig;
//! use brocanth_auth::plugin::ConfiguredProvider;
//! use brocanth_auth::providers::catalog::GOOGLE;
//!
//! providers: vec![ConfiguredProvider {
//!     provider: GOOGLE,
//!     client: ClientConfig::new(
//!         std::env::var("GOOGLE_CLIENT_ID")?,
//!         std::env::var("GOOGLE_CLIENT_SECRET")?,
//!         "https://example.test/auth/google/callback",
//!     ),
//! }]
//! ```
//!
//! A provider is data -- endpoints, scopes, and a mapping from its profile
//! JSON to a common identity -- so the second one is another entry in that
//! `vec!` and no new code anywhere. `brocanth_auth::providers::catalog` has
//! the ones that are already written down.
//!
//! Two things change with the first provider. `same_site` below must become
//! `Lax`, because the callback is a cross-site top-level navigation and
//! `Strict` drops the cookie on the way back. And a matching email still does
//! not link an account to a provider: `Linking` returns a refusal the
//! application answers with a page that asks, because "this provider says
//! alice@example.com" is not Alice consenting, and several real products have
//! shipped that account takeover.
//!
//! # What is not here
//!
//! Second factors, one-time codes and login alerts. Each needs a store trait
//! of its own -- `MfaStore`, `OtpStore`, `SignInLocationStore` -- and
//! `brocanth-auth` deliberately keeps all three off `Auth`, so that an
//! application wanting a password form does not have to implement a TOTP
//! table. An application that wants them builds them here, from the same
//! `Arc<Store>`, and puts them on the context itself: one line each, in this
//! file, where the decision is visible.

use std::sync::Arc;

use brocanth::plugin_api::{Lifecycle, Plugin, PluginError};
use brocanth::plugin_context::{ContextBuilder, ProvidesContext};
use brocanth_auth::http_client::StubHttp;
use brocanth_auth::plugin::AuthConfig;
use brocanth_auth::session::{SameSite, SessionConfig, SigningKey};
use brocanth_auth::{MemoryStore, Passwords, SystemClock};

/// Where identities, sessions and single-use tokens live.
///
/// In memory, for now. See the file header: this is the line to change first.
pub type Store = MemoryStore;

/// How this application reaches a provider.
///
/// It does not. There are no providers, so nothing calls this.
pub type Http = StubHttp;

/// Where the time comes from.
pub type Clock = SystemClock;

/// Everything a handler needs, behind one `Arc`: registration, login, the
/// token service, sessions and identity linking.
///
/// An alias rather than a wrapper, because `brocanth_auth::Auth` is already a
/// `ContextItem`. Cloning is cheap and the clone shares the provider key
/// cache, which is why there is one of these and not one per request.
pub type Services = brocanth_auth::Auth<Store, Http, Clock>;

/// What this plugin is built from.
///
/// Five values a file can carry. The store, the client, the clock and the
/// hasher are built below, because none of them is a thing a TOML file can
/// say.
///
/// It derives `Deserialize` because this application reads it from
/// `[auth]` in a TOML file. Drop the config plugin and the derive goes
/// with it: `Plugin::Config` carries no bounds the core imposes.
#[derive(Debug, serde::Deserialize)]
#[serde(default, deny_unknown_fields)]
pub struct Settings {
    /// The *name* of the environment variable holding the session signing
    /// key, not the key. 32 bytes, base64url.
    pub signing_key_env: String,
    /// The session cookie's name. `__Host-` in production: it is the only
    /// prefix a browser enforces, and what it enforces is `Secure`, a path of
    /// `/` and no `Domain` -- so a subdomain takeover cannot set a session
    /// cookie for the parent.
    pub cookie_name: String,
    /// Whether the cookie is `Secure`. False only on plain-HTTP localhost,
    /// and then the name above must lose its `__Host-` prefix as well.
    pub secure: bool,
    /// Seconds of inactivity before the session is dead.
    pub idle_timeout: i64,
    /// Seconds from creation before it is dead however much it was used.
    pub absolute_timeout: i64,
}

impl Default for Settings {
    fn default() -> Self {
        Self {
            signing_key_env: "MY_BLOG_SESSION_KEY".to_string(),
            cookie_name: "__Host-session".to_string(),
            secure: true,
            idle_timeout: 8 * 60 * 60,
            absolute_timeout: 30 * 24 * 60 * 60,
        }
    }
}

pub struct Auth {
    settings: Settings,
    /// Built at `configure`, so this is an `Option` for exactly the length of
    /// the build.
    services: Option<Services>,
}

impl Plugin for Auth {
    type Config = Settings;
    type Provides = ProvidesContext<Services>;
    const NAME: &'static str = "auth";

    fn new(config: Settings) -> Self {
        Self {
            settings: config,
            services: None,
        }
    }

    fn configure(&mut self, lc: &mut Lifecycle<'_>) -> Result<(), PluginError> {
        if self.settings.cookie_name.is_empty() {
            return Err(PluginError::new(
                "[auth] cookie_name must not be empty\n\
                 help: set `cookie_name` under [auth] in the config file, such as \"__Host-session\"",
            ));
        }
        // A `__Host-` cookie is `Secure` by definition, so a configuration
        // where the two disagree produces a cookie no browser stores and a
        // login that silently never takes. Caught here, at startup, where it
        // can still be answered.
        if self.settings.cookie_name.starts_with("__Host-") && !self.settings.secure {
            return Err(PluginError::new(
                "[auth] a __Host- cookie is Secure by definition, and secure is false\n\
                 help: set secure = true under [auth], or drop the prefix for plain-HTTP development",
            ));
        }

        let name = self.settings.signing_key_env.clone();
        let signing_key = match std::env::var(&name) {
            // Rejected rather than stretched. Padding a four-character secret
            // to 32 bytes is how a deployment ends up with a forgeable cookie
            // and no warning anywhere.
            Ok(raw) => SigningKey::parse(&raw).map_err(|e| {
                PluginError::new(format!(
                    "[auth] ${name}: {e}\n\
                     help: set it to 32 random bytes, base64url:\n  \
                     export {name}=$(head -c 32 /dev/urandom | base64 | tr '+/' '-_' | tr -d '=')"
                ))
            })?,
            // This project's own tests build the application with no secret
            // in the environment. The key is fixed and public, and `cfg(test)`
            // keeps it out of every binary.
            Err(_) if cfg!(test) => SigningKey::from_bytes([7; 32]),
            Err(_) => {
                return Err(PluginError::new(format!(
                    "[auth] ${name} is not set. There is no default: a key invented at startup \
                     would log every session out on each restart and differ between the \
                     instances of a fleet.\n\
                     help: set it to 32 random bytes, base64url:\n  \
                     export {name}=$(head -c 32 /dev/urandom | base64 | tr '+/' '-_' | tr -d '=')"
                )));
            }
        };

        let passwords = Passwords::recommended().map_err(|e| {
            PluginError::new(format!(
                "[auth] argon2id: {e}\n\
                 help: the parameters are brocanth-auth's own, so this is a bug; report it with this message"
            ))
        })?;

        let session = SessionConfig {
            cookie_name: self.settings.cookie_name.clone(),
            secure: self.settings.secure,
            idle_timeout: self.settings.idle_timeout,
            absolute_timeout: self.settings.absolute_timeout,
            // Strict, and not the crate's own `Lax` default. `Lax` is there
            // for the cross-site top-level navigation an OAuth callback is,
            // and this application has no provider. Add one and this line
            // has to become `Lax` or the cookie is dropped on the way back.
            same_site: SameSite::Strict,
            // Path `/`, and thirty minutes for an impersonated session. Both
            // are the crate's defaults and both are arguments rather than
            // numbers; `SessionConfig` carries them.
            ..SessionConfig::default()
        };

        // Nothing here opens anything, so there is no `before_start`:
        // `MemoryStore` is a `HashMap` and `Auth::build` is synchronous. A
        // store over a database is a pool, a pool is opened asynchronously,
        // and no stage may wait for it -- so that swap moves this block into
        // `before_start`, where it builds the pool lazily and registers a
        // readiness check that `serve` awaits. `brocanth-cache` does that.
        self.services = Some(
            brocanth_auth::Auth::build(AuthConfig {
                store: Arc::new(Store::new()),
                http: Arc::new(Http::new()),
                clock: Arc::new(SystemClock),
                signing_key,
                session,
                passwords: Arc::new(passwords),
                // Add one here. The file header has the three lines.
                providers: Vec::new(),
            })
            .map_err(|e| PluginError::new(format!("[auth] {e}")))?,
        );
        // In the services too, for a plugin that reads the accounts at
        // startup, such as `superuser`.
        if let Some(services) = &self.services {
            lc.services_mut().insert(services.clone());
        }
        Ok(())
    }

    fn provide(&self, cx: &mut ContextBuilder) {
        // `configure` runs before the server is handed the application, so
        // the services are here by the time any request builds a context.
        if let Some(services) = &self.services {
            cx.insert(services.clone());
        }
    }
}

//! Account lockout: after too many wrong passwords, an account refuses
//! password sign-in for a while. The policy is in
//! `src/modules/auth_lockout.rs`.
//!
//! Sign in through `lockout(&auth)?.login(auth.accounts(), email, password)`
//! and render `Attempt::Refused` and `Attempt::JustLocked` alike: a page
//! that says "locked" tells a stranger the address has an account. On
//! `JustLocked`, mail the owner. After a password reset, call `clear`.

use brocanth_auth::{AuthError, Lockout, LockoutPolicy};

use crate::plugins::auth::{Clock, Services, Store};

/// The policy in `src/modules/auth_lockout.rs`.
///
/// # Errors
/// [`AuthError::Misconfigured`] when an option is out of its range.
pub fn policy() -> Result<LockoutPolicy, AuthError> {
    let max = crate::modules::auth_lockout::MODULE.max_failures;
    let window = crate::modules::auth_lockout::MODULE.window_secs;
    let lock_for = crate::modules::auth_lockout::MODULE.lock_for_secs;
    LockoutPolicy::new(
        u32::try_from(max).unwrap_or(u32::MAX),
        i64::try_from(window).unwrap_or(i64::MAX),
        i64::try_from(lock_for).unwrap_or(i64::MAX),
    )
}

/// Password sign-in that counts failures, over the auth store.
///
/// # Errors
/// [`AuthError::Misconfigured`] when an option is out of its range.
pub fn lockout(auth: &Services) -> Result<Lockout<Store, Clock>, AuthError> {
    Ok(auth.lockout(policy()?))
}

//! The account screens' plugin, and what every screen under
//! `src/pages/account/` shares.
//!
//! The pages are yours: change the words in `src/pages/account/screens.rsx`
//! and the flow in each page. Keep three rules:
//!
//! 1. Answer alike for an address that has an account and one that has
//!    not. A different word is how a stranger learns who signed up.
//! 2. Build a link in mail from [`Screens::link`], never from the request.
//! 3. Every form carries [`csrf_field`].

use brocanth::plugin_api::{Lifecycle, Plugin, PluginError};
use brocanth::plugin_context::{ContextBuilder, ContextItem, ProvidesContext};
use brocanth_auth::{Admission, AuthStore, FactorPolicy, SessionRecord, Verified};

impl Plugin for AuthScreens {

/// The hidden input a form carries for the CSRF check. HTML, so a
/// template writes it with `{! }`.
pub fn csrf_field(ctx: &Ctx) -> String {
    ctx.hooked::<brocanth_middleware::CsrfToken>()
        .map(|token| token.hidden_field())
        .unwrap_or_default()
}

/// The form, or `None` when a field broke its rule. A body nobody can read
/// is an error: no form sent it.
pub fn form<T: zod::Entry>(ctx: &Ctx) -> Result<Option<T>, Error> {
    match ctx.read::<T>() {
        Sent::Good(value) => Ok(Some(value)),
        Sent::Faulty(_) => Ok(None),
        Sent::Unreadable(why) => Err(Error::new(why.status(), why.to_string())),
    }
}

/// One cookie of the request, by name.
pub fn cookie(ctx: &Ctx, name: &str) -> Option<String> {
    ctx.headers()
        .filter(|(header, _)| header.eq_ignore_ascii_case("cookie"))
        .flat_map(|(_, line)| line.split(';'))
        .find_map(|pair| {
            let (key, value) = pair.split_once('=')?;
            (key.trim() == name).then(|| value.trim().to_string())
        })
        .filter(|value| !value.is_empty())
}

/// The session this request carries, if it is a live one.
pub async fn signed_in(ctx: &Ctx, auth: &Services) -> Option<SessionRecord> {
    let value = cookie(ctx, &auth.sessions().config().cookie_name)?;
    auth.sessions().load(&value).await.ok()
}

/// A message for the next page.
pub fn flash(ctx: &mut Ctx, level: Level, text: &str) {
    if let Some(flash) = ctx.hooked_mut::<Flash>() {
        flash.push(level, text);
    }
}

/// `303 See Other`: after a post, the browser asks for `location` with GET.
pub fn see_other(location: &str) -> Response {
    Response::empty(303).with_header("location", location)
}

/// A mail with a plain-text body, through the outbox.
pub async fn mail(ctx: &Ctx, to: &str, subject: &str, text: String) -> Result<(), Error> {
    let outbox = ctx.provided::<Outbox, _>().cloned().ok_or_else(|| {
        Error::new(
            500,
            "no outbox on this context\nhelp: register the mail plugin",
        )
    })?;
    let message = outbox
        .message()
        .to(brocanth_mail::Address::parse(to)?)
        .subject(subject)
        .text(text)
        .build()?;
    outbox.deliver_later(message).await?;
    Ok(())
}

/// The email of the account a session belongs to, for a label.
pub async fn email_of(auth: &Services, session: &SessionRecord) -> Result<String, Error> {
    let account = auth
        .store()
        .account(&session.account)
        .await
        .map_err(|e| Error::new(500, e.to_string()))?;
    Ok(account
        .and_then(|account| account.email)
        .map(|email| email.as_str().to_string())
        .unwrap_or_default())
}

/// The second half of every sign-in: ask the policy, then open a session,
/// or wait for a second factor, or refuse.
pub async fn admit(ctx: &mut Ctx, auth: &Services, proof: Verified) -> Result<Response, Error> {
    let enrolled = auth.mfa()?.second_factors(&proof).await?;
    let secure = if auth.sessions().config().secure {
        "; Secure"
    } else {
        ""
    };
    match FactorPolicy::recommended().admit(&enrolled, proof) {
        Admission::Admit(admitted) => {
            // The session the browser had, if any, is replaced: a new
            // identifier at sign-in is the defence against fixation.
            let before = cookie(ctx, &auth.sessions().config().cookie_name)
                .and_then(|value| auth.sessions().unseal(&value));
            let started = auth
                .sessions()
                .begin_with_factors(admitted, before.as_ref())
                .await?;
            flash(ctx, Level::Success, "You are signed in.");
            Ok(see_other(crate::modules::auth_screens::MODULE.after_sign_in)
                .with_header("set-cookie", started.cookie.to_string())
                .with_header(
                    "set-cookie",
                    format!("{PENDING}=; Path=/account; Max-Age=0; HttpOnly; SameSite=Strict{secure}"),
                ))
        }
        Admission::Incomplete { proof, .. } => {
            let ttl = crate::modules::auth_screens::MODULE.pending_secs;
            let sealed = auth.sessions().seal_proof(&proof, ttl as i64);
            Ok(see_other("/account/code").with_header(
                "set-cookie",
                format!(
                    "{PENDING}={sealed}; Path=/account; Max-Age={ttl}; HttpOnly; SameSite=Strict{secure}"
                ),
            ))
        }
        Admission::EnrolmentRequired { .. } | Admission::Insufficient => {
            flash(
                ctx,
                Level::Error,
                "This account needs a second factor that it does not have.",
            );
            Ok(see_other("/account/signin"))
        }
    }
}

/// Tells the owner of an account that just locked, with the way back in.
pub async fn mail_locked(
    ctx: &Ctx,
    account: &brocanth_auth::Account,
    lock_for: i64,
) -> Result<(), Error> {
    let Some(email) = &account.email else {
        return Ok(());
    };
    let link = screens(ctx)?.link("/account/forgot");
    let text = format!(
        "Somebody entered a wrong password for your account too many times, \
         so it refuses to sign in for {} minutes. Nobody signed in.\n\n\
         If it was you, wait, or choose a new password now:\n\n{link}\n",
        lock_for / 60
    );
    mail(ctx, email.as_str(), "Your account is locked", text).await
}

#[cfg(test)]
mod tests {
    use brocanth::routing::Verb;
    use brocanth_server::{Body, Request, Response};

    use crate::plugins::auth::Services;

    const PASSWORD: &str = "correct horse battery";

    /// One application and one cookie jar: what a browser keeps between
    /// requests. Each `Browser` builds its own application, so each has its
    /// own memory store.
    struct Browser {
        app: crate::app::App,
        jar: Vec<(String, String)>,
    }

    impl Browser {
        async fn new() -> Self {
            let mut browser = Self {
                app: crate::app::build().expect("the application builds"),
                jar: Vec::new(),
            };
            let first = browser.get("/account/signup").await;
            assert_eq!(first.status, 200);
            assert!(browser.cookie("csrf_token").is_some(), "a CSRF cookie");
            browser
        }

        fn auth(&self) -> Services {
            self.app.core.context().get::<Services, _>().clone()
        }

        fn cookie(&self, name: &str) -> Option<String> {
            self.jar
                .iter()
                .find(|(key, _)| key == name)
                .map(|(_, value)| value.clone())
        }

        fn set_cookie(&mut self, name: &str, value: &str) {
            self.jar.retain(|(key, _)| key != name);
            self.jar.push((name.to_string(), value.to_string()));
        }

        async fn send(&mut self, mut request: Request) -> Response {
            if !self.jar.is_empty() {
                let line: Vec<String> = self.jar.iter().map(|(k, v)| format!("{k}={v}")).collect();
                request = request.with_header("cookie", &line.join("; "));
            }
            let response = crate::app::handle(&self.app, request).await;
            for (name, line) in response.headers() {
                if !name.eq_ignore_ascii_case("set-cookie") {
                    continue;
                }
                let pair = line.split(';').next().unwrap_or_default();
                let Some((key, value)) = pair.split_once('=') else {
                    continue;
                };
                self.jar.retain(|(k, _)| k != key);
                if !value.is_empty() && !line.contains("Max-Age=0") {
                    self.jar.push((key.to_string(), value.to_string()));
                }
            }
            response
        }

        async fn get(&mut self, path: &str) -> Response {
            self.send(Request::new(Verb::Get, path)).await
        }

        /// A form post with the CSRF field, as the page's form sends it.
        async fn post(&mut self, path: &str, form: &str) -> Response {
            let token = self.cookie("csrf_token").expect("a CSRF cookie");
            self.post_raw(path, &format!("_csrf_token={token}&{form}"))
                .await
        }

        async fn post_raw(&mut self, path: &str, body: &str) -> Response {
            let (path, query) = path.split_once('?').unwrap_or((path, ""));
            let mut request = Request::new(Verb::Post, path)
                .with_header("content-type", "application/x-www-form-urlencoded");
            request.query = query.to_string();
            request.body = body.to_string().into();
            self.send(request).await
        }

        /// The page a redirect points at, as the browser asks for it next.
        async fn follow(&mut self, response: &Response) -> String {
            assert_eq!(response.status, 303, "a redirect");
            let location = response.header("location").expect("a location").to_string();
            let page = self.get(&location).await;
            text(&page)
        }

        async fn sign_up(&mut self, email: &str) -> Response {
            self.post("/account/signup", &credentials(email, PASSWORD))
                .await
        }

        async fn sign_in(&mut self, email: &str, password: &str) -> Response {
            self.post("/account/signin", &credentials(email, password))
                .await
        }

        /// Whether the jar holds a session.
        fn signed_in(&self) -> bool {
            let name = self.auth().sessions().config().cookie_name.clone();
            self.cookie(&name).is_some()
        }
    }

    fn credentials(email: &str, password: &str) -> String {
        format!(
            "email={}&password={}",
            email.replace('@', "%40"),
            password.replace(' ', "+")
        )
    }

    fn text(response: &Response) -> String {
        match &response.body {
            Body::Bytes(bytes) => String::from_utf8_lossy(bytes).into_owned(),
            Body::Stream(_) => panic!("a page, not a stream"),
        }
    }

    /// The `<p role="alert">` of a form drawn again.
    fn alert(response: &Response) -> String {
        let body = text(response);
        let start = body.find(r#"<p role="alert">"#).expect("an alert");
        let rest = &body[start..];
        rest[..rest.find("</p>").expect("closed")].to_string()
    }

    /// A reset token, as the link in the reset mail carries it.
    async fn reset_token(browser: &Browser, email: &str) -> String {
        match browser
            .auth()
            .accounts()
            .request_password_reset(email)
            .await
            .expect("a reset")

    /// Two-step sign-in, from turning it on to a replayed code.
    #[tokio::test]
    async fn a_second_factor_is_asked_for_and_a_wrong_or_spent_code_is_refused() {
        use brocanth_auth::{Clock, SystemClock, Totp, TotpSecret};

        let mut browser = Browser::new().await;
        browser.sign_up("ada@example.com").await;
        browser.sign_in("ada@example.com", PASSWORD).await;
        let page = text(&browser.get("/account/totp").await);
        let start = page.find("<code>").expect("the key") + "<code>".len();
        let secret = &page[start..start + page[start..].find("</code>").expect("closed")];
        let secret = TotpSecret::from_base32(secret).expect("base32");
        let code = |at: i64| Totp::interoperable().code_at(&secret, at);
        let now = SystemClock.now();

        let wrong = if code(now) == "000000" {
            "111111"
        } else {
            "000000"
        };
        let refused = browser
            .post("/account/totp", &format!("code={wrong}"))
            .await;
        assert_eq!(refused.status, 422);
        let on = browser
            .post("/account/totp", &format!("code={}", code(now)))
            .await;
        assert_eq!(on.status, 200);
        let page = text(&on);
        let recovery: Vec<String> = page
            .split("<li><code>")
            .skip(1)
            .map(|rest| rest[..rest.find("</code>").expect("closed")].to_string())
            .collect();
        assert!(!recovery.is_empty(), "the recovery codes, once: {page}");

        // The password alone now leads to the code page, not a session.
        browser.post("/account/signout", "").await;
        let first = browser.sign_in("ada@example.com", PASSWORD).await;
        assert_eq!(first.header("location"), Some("/account/code"));
        assert!(!browser.signed_in());
        assert!(browser.cookie(super::PENDING).is_some());

        let wrong = browser
            .post("/account/code", &format!("code={wrong}"))
            .await;
        assert_eq!(wrong.status, 422);
        assert!(!browser.signed_in());

        // The code that turned it on is spent; the next step's code works.
        let next = code(now + 30);
        let replayed = browser
            .post("/account/code", &format!("code={}", code(now)))
            .await;
        assert_eq!(replayed.status, 422, "the enrolment code is spent");
        let admitted = browser.post("/account/code", &format!("code={next}")).await;
        assert_eq!(
            admitted.header("location"),
            Some(crate::modules::auth_screens::MODULE.after_sign_in)
        );
        assert!(browser.signed_in());
        assert!(
            browser.cookie(super::PENDING).is_none(),
            "the pending cookie is cleared"
        );

        // A recovery code works once.
        browser.post("/account/signout", "").await;
        browser.sign_in("ada@example.com", PASSWORD).await;
        let replayed = browser.post("/account/code", &format!("code={next}")).await;
        assert_eq!(replayed.status, 422, "a TOTP code works once");
        let recovered = browser
            .post("/account/code", &format!("code={}", recovery[0]))
            .await;
        assert_eq!(recovered.status, 303);
        assert!(browser.signed_in());
        browser.post("/account/signout", "").await;
        browser.sign_in("ada@example.com", PASSWORD).await;
        let spent = browser
            .post("/account/code", &format!("code={}", recovery[0]))
            .await;
        assert_eq!(spent.status, 422, "a recovery code works once");
    }

//! Mail, as this application sends it.
//!
//! A page reads the `Outbox` from the context and builds a message from it,
//! so `From` is always the configured sender:
//!
//! ```ignore
//! let outbox = ctx.provided::<Outbox, _>().expect("a dispatched request");
//! let message = outbox
//!     .message()
//!     .to(brocanth_mail::Address::parse("person@example.test")?)
//!     .subject("Welcome")
//!     .text("Hello")
//!     .build()?;
//! outbox.deliver_later(message).await?;
//! ```
//!
//! # Where mail goes
//!
//! `[mail] transport` decides: `log` writes each message to the log, `folder`
//! writes each one as an `.eml` file into `[mail] folder`, and `smtp` sends it
//! to `[mail] smtp_host`. Only `smtp` delivers, so a development file copied
//! to production sends nothing to real people, and the start says so.
//!
//! The SMTP password has no flag: every user on the host can read a process's
//! arguments. Set `$<APP>_MAIL__SMTP_PASSWORD`.
//!
//! # Later
//!
//! There is no queue in this project, so `deliver_later` sends at once.
//! Add the `jobs` module and it puts the message on a queue instead, with
//! no change to the pages that call it.
use std::sync::Arc;

use brocanth::plugin_api::{Lifecycle, Plugin, PluginError};
use brocanth::plugin_context::{ContextBuilder, ContextItem, ProvidesContext};
use brocanth_mail::{
    Address, Folder, Log, Mail, MailError, Message, MessageBuilder, Security, Smtp, SmtpConfig,
};

/// What this plugin is built from.
///
/// It derives `Deserialize` because this application reads it from
/// `[mail]` in a TOML file. Drop the config plugin and the derive goes
/// with it: `Plugin::Config` carries no bounds the core imposes.
#[derive(serde::Deserialize)]
#[serde(default, deny_unknown_fields)]
pub struct Settings {
    /// `smtp`, `log` or `folder`.
    pub transport: String,
    /// The address every message comes from.
    pub from: String,
    /// The name shown beside it. Empty for none.
    pub from_name: String,
    /// Where `folder` writes.
    pub folder: String,
    /// The relay.
    pub smtp_host: String,
    /// The relay's port.
    pub smtp_port: u16,
    /// `tls`, `starttls`, or `plain` for a local catcher only.
    pub smtp_security: String,
    /// The relay's user name. Empty for none.
    pub smtp_username: String,
    /// The relay's password. Empty for none. A secret: no flag sets it.
    pub smtp_password: String,
}

/// Every field but a secret, which prints as `<redacted>`: a log line or
/// a panic message that prints the settings leaks no secret.
impl std::fmt::Debug for Settings {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        f.debug_struct("Settings")
            .field("transport", &self.transport)
            .field("from", &self.from)
            .field("from_name", &self.from_name)
            .field("folder", &self.folder)
            .field("smtp_host", &self.smtp_host)
            .field("smtp_port", &self.smtp_port)
            .field("smtp_security", &self.smtp_security)
            .field("smtp_username", &self.smtp_username)
            .field("smtp_password", &"<redacted>")
            .finish()
    }
}

impl Default for Settings {
    fn default() -> Self {
        Self {
            transport: "log".to_string(),
            from: "no-reply@example.com".to_string(),
            from_name: "My Blog".to_string(),
            folder: "tmp/mail".to_string(),
            smtp_host: "localhost".to_string(),
            smtp_port: 587,
            smtp_security: "starttls".to_string(),
            smtp_username: String::new(),
            smtp_password: String::new(),
        }
    }
}

/// What a page sends mail through.
#[derive(Clone)]
pub struct Outbox {
    mail: Mail,
    transport: String,
}

impl Outbox {
    /// A message from the configured sender.
    pub fn message(&self) -> MessageBuilder {
        self.mail.message()
    }

    /// Sends the message now, and waits for the transport to take it.
    ///
    /// `allow(dead_code)`: it is here for the first page that sends at once,
    /// and a project that only queues mail has none.
    #[allow(dead_code)]
    pub async fn send(&self, message: &Message) -> Result<(), MailError> {
        self.mail.send(message).await
    }

    /// Sends the message. This project has no queue, so it sends at once;
    /// with the `jobs` module it goes on a queue.
    pub async fn deliver_later(&self, message: Message) -> Result<(), MailError> {
        self.mail.send(&message).await
    }

    /// Where mail goes: `smtp`, `log` or `folder`.
    #[allow(dead_code)]
    pub fn transport(&self) -> &str {
        &self.transport
    }
}

impl ContextItem for Outbox {}

pub struct MailPlugin {
    settings: Settings,
    /// Built at `before_start`, so this is an `Option` for the length of the
    /// build.
    outbox: Option<Outbox>,
}

impl MailPlugin {
    /// `[mail] from`, with `[mail] from_name` beside it.
    fn sender(&self) -> Result<Address, PluginError> {
        let s = &self.settings;
        if s.from_name.is_empty() {
            Address::parse(&s.from)
        } else {
            Address::named(&s.from_name, &s.from)
        }
        .map_err(|e| PluginError::new(format!("[mail] from: {e}\nhelp: set a plain address")))
    }

    /// The backend `[mail] transport` names.
    fn mailer(&self) -> Result<Arc<dyn brocanth_mail::Mailer>, PluginError> {
        let s = &self.settings;
        Ok(match s.transport.as_str() {
            "log" => Arc::new(Log::new(|message: &Message| {
                let to: Vec<&str> = message.to().iter().map(Address::address).collect();
                let to = to.join(", ");
                let subject = message.subject().as_str();
                let text = message.text();
                crate::plugins::say!(info, "mail to {to}: {subject}\n{text}"; %to, %subject);
            })),
            "folder" => Arc::new(Folder::new(&s.folder)),
            "smtp" => {
                let security = match s.smtp_security.as_str() {
                    "tls" => Security::ImplicitTls,
                    "starttls" => Security::StartTls,
                    "plain" => Security::PlaintextForLocalCatcherOnly,
                    other => {
                        return Err(PluginError::new(format!(
                            "[mail] smtp_security is `{other}`\n\
                             help: set it to `tls`, `starttls` or `plain`"
                        )))
                    }
                };
                let mut config = SmtpConfig::new(&s.smtp_host)
                    .with_security(security)
                    .with_port(s.smtp_port);
                if !s.smtp_username.is_empty() {
                    config = config.with_credentials(&s.smtp_username, &s.smtp_password);
                }
                Arc::new(
                    Smtp::connect(&config).map_err(|e| PluginError::new(format!("[mail] {e}")))?,
                )
            }
            other => {
                return Err(PluginError::new(format!(
                    "[mail] transport is `{other}`\nhelp: set it to `smtp`, `log` or `folder`"
                )))
            }
        })
    }
}

impl Plugin for MailPlugin {
    type Config = Settings;
    type Provides = ProvidesContext<Outbox>;
    const NAME: &'static str = "mail";

    fn new(config: Settings) -> Self {
        Self {
            settings: config,
            outbox: None,
        }
    }

    fn before_start(&mut self, _lc: &mut Lifecycle<'_>) -> Result<(), PluginError> {
        let from = self.sender()?;
        // The SMTP pool spawns its upkeep on the runtime, so it needs the
        // runtime entered. It opens no connection here.
        let mailer = match tokio::runtime::Handle::try_current() {
            Ok(runtime) => {
                let _entered = runtime.enter();
                self.mailer()?
            }
            Err(_) => self.mailer()?,
        };
        if self.settings.transport != "smtp" {
            let transport = &self.settings.transport;
            crate::plugins::say!(
                warn,
                "[mail] transport is {transport}: nothing is delivered";
                %transport
            );
        }
        let mail = Mail::new(mailer, from);
        self.outbox = Some(Outbox {
            mail,
            transport: self.settings.transport.clone(),
        });
        Ok(())
    }

    fn provide(&self, cx: &mut ContextBuilder) {
        // `before_start` runs before the server is handed the application, so
        // the outbox is here by the time any request builds a context.
        if let Some(outbox) = &self.outbox {
            cx.insert(outbox.clone());
        }
    }
}

# Working in My Blog

A brief for a coding agent. Written by `brocanth`, so the lists
below are this project's and not the framework's in general.

## Check your work

Run these after every change, in this order. They are ordered by how
fast they fail.

```
cargo test
cargo clippy --all-targets -- -D warnings
cargo fmt --check
```

The route table is `pages::router()` and the test beside it asserts the
shape of the directory. Read it rather than guessing what covers a
request.

## What this binary contains

- `core`  the router, the context, the plugin seams
- `config`  config/<env>.toml, merged; without it nothing links a TOML parser
- `auth`  argon2id accounts and signed sessions; no provider, so no HTTP client
- `auth-screens`  sign-up, sign-in, sign-out, reset, verify and TOTP pages under /account
- `rsx`  `.rsx` templates compiled into the binary; without it a page builds a String
- `input`  a form or a JSON body, read into a struct, with every fault at once
- `middleware`  proxy headers, method override, canonical host, content types, CORS
- `middleware-csrf`  a double-submit cookie, checked on every verb that is not a read
- `mail`  send mail through SMTP, or to the log or a folder in development
- `flash`  one-shot messages across a redirect, in a signed cookie, drawn into the layout
- `auth-lockout`  locks an account for minutes after too many wrong passwords

**Absent:** `logging`, `observability`, `observability-logs`, `observability-health`, `observability-metrics`, `observability-tracing`, `analytics`, `beacon`, `sessions`, `db`, `model`, `cache`, `cache-memory`, `cache-redis`, `jobs`, `jobs-memory`, `flags`, `audit`, `auth-oauth`, `superuser`, `roles`, `policy`, `rls`, `storage`, `storage-routes`, `client`, `middleware-security-headers`, `middleware-rate-limit`, `middleware-conditional-get`, `middleware-compression`, `gtag`, `openapi`, `validation`, `resource`, `admin`, `admin-model`, `admin-actions`, `webhooks`, `webhooks-in`, `webhooks-out`, `oauth`, `oauth-client`, `oauth-client-memory`, `realtime`, `realtime-sse`, `realtime-websocket`, `commands`, `jt`, `rsx-tailwind`, `rsx-htmx`, `sanitize`, `static`, `robots`, `sitemap`, `pagination`, `filters`, `search`, `i18n`, `i18n-routing`, `meta`, `errors`, `docker`, `docker-kubernetes`, `ci`, `ci-github` — not compiled, not linked, absent from `Cargo.lock`. Do not write code that uses them; add the dependency and name the plugin in `main` first.

That is the framework's central rule: a capability nobody asked for is
not compiled and not linked. If you need one, add it to `Cargo.toml`
*and* name its plugin in `src/main.rs`. Nothing registers itself.

## Adding a route

A route is a file. `src/pages/about.rs` is `/about`;
`src/pages/users/[id].rs` is `/users/{id}`. **Write the file. That is
the whole of it** -- no `mod` line, no row to register, because there is
no list. `brocanth_pages::pages!` in `src/pages/mod.rs` reads this
directory while the crate compiles and writes both.

The verb comes off the handler's name: `handle` is GET, then
`handle_head`, `handle_post`, `handle_put`, `handle_patch`,
`handle_delete`, `handle_options`. A `pub fn handle_*` outside that
list is a build error naming the list, rather than a handler nothing
reaches; a page with no handler is a build error too. A private
`fn handle_*` is a helper and is left alone.

A handler may be `pub async fn` when it waits on something, such as a
database: the table awaits it and the wait holds no thread. A page that
waits on nothing stays a plain `fn`. An async handler is dropped at an
`.await` when the client goes away, so do not leave a write half done
across one.

**Adding a file does not rebuild on its own** -- nothing referenced it
yet. Run `touch src/pages/mod.rs` after creating a page or a layout.
Editing or deleting one is noticed.

A handler returns a value and never writes a response itself. That is
what makes a half-written body on an error impossible rather than
unlikely.

## Views

The markup for a route is a `.rsx` file **next to the page that renders it**:
`src/pages/index.rsx` beside `src/pages/index.rs`. There is no second tree and
nothing to register -- `rsx::load_dir!("src/pages")` in `src/pages/mod.rs`
compiles them all into the binary. One file is one module, under `views`, so
`index.rsx` is `views::index` and `users/[id].rsx` is `views::users::_id_`.

### Layouts

`layout.rsx` wraps every page in its directory **and every directory below**,
and a page never names it: which layout wraps a page is a property of where
the page is. That is the Crystal page_router's rule for `layout.cr`.

They stack root to leaf, and **the chain is not written down anywhere**.
`html!` is a macro: it knows which file it was written in, walks up to
`src/`, and wraps the page in every `layout.rsx` it passes. So
`src/pages/users/layout.rsx` wraps the pages under `users/` inside the root
layout, and deleting that file removes exactly that layer with no other edit.

Every page uses the same import and the same call, whatever directory it is
in. Do not add a chain of your own; if a page is wrapped wrongly, the
directory tree is what to change.

A layout takes `children: Element`, and may also take `layout: &str`, which
`html!` fills with the URL prefix that layout covers. Those two are the only
parameters it fills, because it cannot know what a third one would mean.
A value every page shares comes from a `const` -- the root layout reads
`pages::TITLE` that way. A value belonging to one request belongs on the page
component, which the handler builds and which already has the context.

For a fragment that is not a document -- an htmx swap, a partial -- call
`render` instead of `html!`: the same thing with no shell.

**Adding a `layout.rsx` where there was none does not rebuild anything**,
because nothing referenced the file yet. Run `touch src/pages/mod.rs` after
creating one. Editing or deleting an existing layout is noticed.

A component is a function returning `Element`. Its parameters are its contract
with the handler, and a caller fills them as struct fields, so a field the
component does not declare is a compile error rather than a blank in the page.
A parameter of type `Element` receives children, which is all a layout is.

Reach a template from a handler with `html!`, and with nothing else:

```rust
pub fn handle(_ctx: &mut Ctx) -> Result<impl IntoResponse, Error> {
    html!(ctx, views::index::Index { title: TITLE })
}
```

It returns the handler's own `Result`, so there is no `Ok(..)` to write and a
template that bails propagates with `?`. For another status or a header,
compose with the other return shapes: `Ok((html!(ctx, tpl)?, 404))`.

Adding a page with markup is two files in one directory -- `<name>.rs` and
`<name>.rsx` -- and nothing else anywhere. Neither needs a line in
`src/pages/mod.rs`, because the directory is the registry.

Not every response is a page. A redirect is `Response::empty(303)` with a
header and no template anywhere near it.

An interpolated value is escaped on the way out. Templates are read at
**compile** time, so an edit needs a rebuild, and a template that does not
parse is a build failure on its own line rather than a 500.

## Reading a body

A handler never parses `ctx.body` itself. It declares a struct with
`#[derive(zod::Entry)]`, puts the rules on the fields, and calls
`ctx.body_as()` or `ctx.read()`. The rules are values: a `const`
shared by two structs is one rule, and `zod::accepts!` makes the
compiler check a literal against it.

Report every fault, never the first. `Faults` already holds them all
and `by_name()` groups them per field; a handler that returns after
the first one makes a person submit the form three times.

Do not write `?` on a `zod::Faults`. The blanket `From` on `Error`
turns any `std::error::Error` into a 500, and a bad body is a 422.
`body_as` is the call that gets the status right.

## Edge middleware

`src/plugins/mod.rs` holds five commented registrations from
`brocanth-middleware`. Each encodes a fact about the deployment --
which proxy to trust, which host is canonical, which origins may call
-- so each is turned on by somebody who knows that fact. Do not
enable them as a set, and do not invent an address or an origin to
make one compile. To turn one on, copy it to `build()` in
`src/app.rs` and widen `type Core` by its hook families; the comment
above the five shows the type.

## Flash

`brocanth_middleware::FlashPlugin` is registered in
`src/plugins/mod.rs`. After a form POST, write a message and redirect:

```rust
if let Some(flash) = ctx.hooked_mut::<brocanth_middleware::Flash>() {
    flash.success("Saved.");
}
Ok(Response::empty(303).with_header("location", "/"))
```

The next HTML page shows it inside `<div id="flash">`, which the root
layout carries. Keep that element in the layout. A page that draws a
form again without a redirect uses `flash.now(level, text)`. The text
is plain: the hook escapes it. Never put HTML or a secret in a message.

## CSRF

`brocanth_middleware::Csrf` is registered in `src/plugins/mod.rs`. It
is a double-submit cookie: the hook mints a token, sets it as its own
cookie, and refuses any verb that is not GET, HEAD, OPTIONS or TRACE
unless the request sends the same value back -- in the `_csrf_token`
field, or in the `X-CSRF-Token` header.

So every form that posts carries the hidden field. In a page, read
the token with `ctx.hooked::<brocanth_middleware::CsrfToken>()` and
write `token.hidden_field()` inside the `<form>`. For HTML this
project did not template, `token.stamp_forms(html)` adds the field to
every posting form; the admin uses it.

A 403 from this hook is a form that lost its token, or a caller that
is not a browser. Add that path to `exempt`; do not remove the
registration.

## Mail

`src/plugins/mail/mod.rs` registers the mail plugin. A page reads the
`Outbox` with `ctx.provided::<crate::plugins::mail::Outbox, _>()` and
builds a message with `outbox.message()`, which sets `From`.

Use `outbox.deliver_later(message)` in a request, not `send`: with the
`jobs` module the message goes on a queue and a slow relay does not hold
the response. Never build an `Smtp` in a page, and never put the SMTP
password in a file that is in git or on a command line.

In development `[mail] transport` is `log`: each message is in the log,
and nothing is sent.

## Account lockout

`src/plugins/auth_lockout.rs` builds the lockout with the policy in
`src/modules/auth_lockout.rs`. Sign in through `lockout(&auth)?.login(..)`,
not `auth.accounts().login(..)`, or wrong passwords are not counted.
Render `Attempt::Refused` and `Attempt::JustLocked` with the same words
and status; mail the owner on `JustLocked`. Call `clear` after a password
reset.

## Account screens

`src/pages/account/` holds the sign-up, sign-in, sign-out, reset,
verify and TOTP pages; their markup is `src/pages/account/screens.rsx`,
and the helpers are in `src/plugins/auth_screens/mod.rs`. Where a person
lands after signing in (`after_sign_in`) and how long a sign-in waits
for its second factor (`pending_secs`) are in
`src/modules/auth_screens.rs`.

Rules when you change them:

- Answer alike for an address with an account and one without, in the
  words and the status. Say which it was only in the mail.
- Build a link in mail with `Screens::link`, which reads
  `[auth_screens] base_url`. Never build one from the `Host` header.
- Every form carries `csrf_field(ctx)`, written with `{! }`.
- A page that changes state answers a POST, never a GET.
- Read who is signed in with `auth_screens::signed_in`. Do not trust a
  cookie value you did not unseal.

## Decisions that belong to the person

The modules in this project are the person's choice. Do not add or
remove one on your own. Before you run `brocanth add`, show the person
the question: `brocanth add --questions` lists each module with its
default, what it needs and what it recommends. Give your
recommendation, and wait for the answer.

Not chosen, and recommended by a module that is here. Ask about each:

- `observability-logs`, for `config`: $<APP>_ENV picks the environment; only observability-logs writes JSON when it is production

## House style

- A comment says **why**, next to the decision, and what was chosen
  against. Not what the line does.
- A test name is a sentence describing the guarantee it protects.
- Prefer a type that cannot be built wrong over a check that can be
  forgotten.
- State a limit where someone would look for it rather than leaving it
  to be discovered.
