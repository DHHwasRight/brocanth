//! Written by `brocanth`. Do not edit -- `brocanth add` and `brocanth remove` rewrite this file.
//! Something else belongs here? Put it in `src/app.rs`, which is yours.
//!
//! My Blog -- a brocanth application.
//!
//! No command line was asked for, so this binary does one thing. It takes the
//! flags its settings declare, which `--help` lists and `src/modules/`
//! renames, and nothing else. Everything it contains is in `src/app.rs`;
//! `brocanth add commands` turns this into a command line with `serve` as one
//! of its commands.

mod app;
mod modules;
mod pages;
mod plugins;

use std::process::ExitCode;

fn main() -> ExitCode {
    let served = plugins::read_line().and_then(|helped| {
        if helped {
            Ok(())
        } else {
            app::serve(&app::default_addr())
        }
    });
    match served {
        Ok(()) => ExitCode::SUCCESS,
        Err(message) => {
            // Through `say!`: a JSON line when the log subscriber is up,
            // text when the failure came before it.
            plugins::say!(error, "error: {message}");
            ExitCode::FAILURE
        }
    }
}

//! My Blog -- everything this binary contains, in the order it happens.
//!
//! This file is yours. `brocanth add` does not rewrite it: what it would
//! have to say is in `src/plugins/mod.rs`, which is generated, and this one
//! only asks that file four questions.
//!
//! A plugin that is not named in `plugins::register!` is not registered, and
//! one that is not in `Cargo.toml` is not compiled at all. "The framework
//! decided for you and you cannot see where" is the thing this layout exists
//! to make impossible.
//!
//! `serve` and every other command read this, so there is one answer to what
//! this application is.

use std::sync::Arc;
use std::time::{Duration, Instant};

use brocanth::plugin_app::AppBuilder;
use brocanth::plugin_hooks::{self as hooks, Begin, RequestState};
use brocanth::plugin_routes::BoxFuture;
use brocanth::plugin_shutdown::ShutdownReason;
use brocanth::routing::{AsyncMiddlewareStack, Dispatch, Router, Verb};

use crate::pages::{self, Ctx, Handler, IntoResponse, Response};
use crate::plugins;

/// The application name. Every environment variable this binary reads is
/// `MY_BLOG_<SUFFIX>`: one convention, derived once, whatever spelling the
/// name arrived in.
pub const APP: &str = "my-blog";

/// Where `serve` listens when nothing says else: `$MY_BLOG_ADDR`, or
/// `127.0.0.1:3000`. The container image sets `0.0.0.0:3000`.
pub fn default_addr() -> String {
    std::env::var("MY_BLOG_ADDR").unwrap_or_else(|_| "127.0.0.1:3000".to_string())
}

/// What a request needs, built once and shared by every connection.
pub struct App {
    pub router: Router<Handler>,
    pub stack: AsyncMiddlewareStack<Ctx, Response>,
    pub plugins: Vec<String>,
    /// The application the plugins built.
    ///
    /// Held rather than dropped, because three things need it
    /// after `build` returns: a per-request context, the routes a
    /// plugin contributed, and an HTTP test client. It used to be
    /// a boxed closure over `context()` alone, which cost an
    /// indirect call per request and reached none of the rest.
    ///
    /// Nameable at last: `plugins::Installed` and `plugins::Hooks`
    /// are the two type parameters, and the generator writes both.
    pub core: Core,
}

/// The built application, with its provider list and hook record.
type Core = brocanth::plugin_app::App<plugins::Installed, plugins::Hooks>;

/// Builds it. The only place that decides what this binary is.
pub fn build() -> Result<App, String> {
    // 1. The application, and every plugin it contains. The chain is in
    //    `src/plugins/mod.rs`, one line per plugin and each one a line a
    //    reader can delete. It is there rather than here because the
    //    generator rewrites it whenever the module list changes.
    let app = plugins::register!(AppBuilder::new())
        .setup(|setup| {
            // A plugin *offers* middleware and cannot mount it: a
            // `MiddlewareCatalog` has no `mount`. Mounting takes a pattern, the
            // pattern is a decision, and the decision is the application's --
            // which is why these lines are here, in the file you own.
            // Nothing in this binary offers middleware, so nothing is mounted.
            let _ = setup.router();
            Ok(())
        })
        .build()
        .map_err(|e| e.to_string())?;

    // 2. The route table: the shape of src/pages and nothing else. A file is
    //    a route, and `pages::router` is the only place that says so.
    let router = pages::router().map_err(|e| e.to_string())?;

    // 3. The chain, from exactly what the block above mounted. A name this
    //    binary has no layer for stops the application rather than silently
    //    not running.
    let mut stack: AsyncMiddlewareStack<Ctx, Response> = AsyncMiddlewareStack::new();
    for mount in app.router_plan().mounts() {
        let layer = plugins::layer(&mount.middleware).ok_or_else(|| {
            format!(
                "`{}` is mounted, but this binary has no layer for it",
                mount.middleware
            )
        })?;
        stack
            .use_at(&mount.pattern, layer)
            .map_err(|e| e.to_string())?;
    }

    // 4. What the plugins put on the per-request context, said once.
    plugins::announce!(&app);

    let plugins = app.instance_names().map(str::to_string).collect();

    Ok(App {
        router,
        stack,
        plugins,
        core: app,
    })
}

/// Turns a page's answer into the response the hooks see.
///
/// `text/html` unless the page names its own `content-type`, which then
/// replaces it rather than joining it. Every other header is appended,
/// because `set-cookie` has to repeat. A page that answered with a stream
/// keeps it: the hooks see its headers, and the server polls its body.
fn sent(answered: Response) -> hooks::Response {
    match answered.stream {
        Some(stream) => {
            brocanth_server::bridge::streaming(answered.status, answered.headers, stream)
        }
        None => brocanth_server::bridge::buffered(
            answered.status,
            answered.headers,
            answered.body.into_bytes(),
        ),
    }
}

/// The request headers as text, for the page.
///
/// A value that is not UTF-8 is dropped: a page reads a header as `&str`,
/// and a lossy copy would say something the client did not send.
fn headers(request: &hooks::Request) -> Vec<(String, String)> {
    request
        .headers()
        .iter()
        .filter_map(|(name, value)| Some((name.to_string(), value.to_str().ok()?.to_string())))
        .collect()
}

/// How long a stop waits for open requests and then for the plugins.
///
/// Kubernetes sends SIGTERM, waits `terminationGracePeriodSeconds` (30 by
/// default) and then sends SIGKILL. 25 seconds leaves 5 for the exit.
/// `[server] shutdown_grace_secs` overrides it: `--server.shutdown_grace_secs`,
/// `$<APP>_SERVER__SHUTDOWN_GRACE_SECS`, or with the config plugin the file.
/// See `plugins::shutdown_grace`.
pub const SHUTDOWN_GRACE: Duration = Duration::from_secs(25);

/// Serves until SIGTERM or SIGINT, then stops in two phases (PLAN.md ADR 2).
///
/// 1. The listener closes. Each open request has the grace period to
///    finish, and each open response stream is told to end.
/// 2. Each plugin stops, in reverse registration order, inside what is left
///    of the grace period.
///
/// An error is a non-zero exit, and an orchestrator reads it: the drain ran
/// out of time, the listener failed, or a plugin did not stop cleanly.
///
/// Synchronous on the outside because the command that calls it is, and
/// the runtime is built here rather than wrapped around `main`: a command
/// that never serves should not start a reactor to run.
pub fn serve(addr: &str) -> Result<(), String> {
    let grace = plugins::shutdown_grace()?;
    // Before the socket, so a signal during startup stops the server
    // through the drain and does not kill the process.
    let stop = watch(grace).map_err(|e| e.to_string())?;
    let runtime = tokio::runtime::Builder::new_multi_thread()
        .enable_all()
        .build()
        .map_err(|e| e.to_string())?;

    // The runtime first, and the application inside it. A plugin
    // that opens a connection pool builds it in `before_start`
    // with no connection, and the pool spawns its upkeep task on
    // the current runtime. Building the application first would
    // leave it with no runtime to spawn on.
    let app = {
        let _guard = runtime.enter();
        build()?
    };
    let names = app.plugins.join(", ");
    plugins::say!(info, "plugins: [{names}]"; plugins = %names);

    let result = runtime.block_on(run(addr, app, stop, grace));
    // A request that outlived the drain can still be on a worker. Waiting
    // for it here would undo the grace period, so it is dropped.
    runtime.shutdown_background();
    result
}

/// The async half of [`serve`].
async fn run(
    addr: &str,
    mut app: App,
    mut stop: tokio::sync::mpsc::UnboundedReceiver<Stop>,
    grace: Duration,
) -> Result<(), String> {
    // Every readiness check once, before the socket: the same checks
    // `/readyz` runs. The build waits for no I/O, so a wrong password or
    // an unreachable database stops the start here, with the plugin's
    // message and its `help:` line, and not at the first request. A
    // check can wait for a pool's acquire timeout, so a signal ends the
    // wait.
    tokio::select! {
        checked = app.core.readiness().check_all() => {
            checked.map_err(|not_ready| not_ready.to_string())?;
        }
        _ = stop.recv() => {
            let report = app.core.shutdown(ShutdownReason::Signal, Some(grace));
            plugins::say!(info, "{report}");
            return Ok(());
        }
    }

    // One gate: a stop at the listener and a stop at `App::shutdown`
    // close the same set of response streams.
    let server = brocanth_server::Server::bind(addr)
        .await
        .map_err(|e| e.to_string())?
        .with_streams(app.core.streams().clone());
    let bound = server.local_addr().map_err(|e| e.to_string())?;
    let stopper = server.stopper();

    // After the bind, so a plugin does not start work for a port that
    // turned out to be taken.
    if let Err(e) = app.core.start() {
        let report = app.core.shutdown(ShutdownReason::Failed, None);
        plugins::say!(error, "{report}");
        return Err(e.to_string());
    }

    // The probes and `/metrics` on a port of their own, when the
    // configuration names one.
    plugins::side_listeners(&app).await?;

    let app = Arc::new(app);
    // `plugins::wrap` puts whatever observes a request around it.
    let handler = plugins::wrap(&app, {
        let app = Arc::clone(&app);
        move |request| {
            let app = Arc::clone(&app);
            async move { handle(&app, request).await }
        }
    });
    let mut serving = tokio::spawn(server.run(handler));
    plugins::say!(info, "listening on http://{bound}"; addr = %bound);

    // Phase 1: stop accepting, and give the open requests the grace period.
    let mut failed = tokio::select! {
        event = stop.recv() => {
            let signal = match event {
                Some(Stop::Signal(signal)) => signal,
                _ => "stop",
            };
            plugins::say!(
                info,
                "{signal}: stopping; open requests have {grace:?} to finish";
                signal,
                ?grace
            );
            None
        }
        // The listener stopped by itself. That is a failure, and the
        // plugins still stop.
        served = &mut serving => Some(stopped(served)),
    };
    let reason = match failed {
        None => ShutdownReason::Signal,
        Some(_) => ShutdownReason::Failed,
    };
    let started = Instant::now();
    // Not ready from the first instant of the stop (ADR 2): `/readyz`
    // answers 503, and a plugin that awaits `Readiness::draining` stops
    // taking work now -- the job worker stops claiming -- while the
    // listener still drains.
    app.core.readiness().drain();
    if failed.is_none() {
        // With `observe`, `/readyz` fails first, and the listener closes
        // after `[observability] drain_delay_secs`. Inside the grace period.
        plugins::stop_accepting(&app, stopper).await;
        failed = tokio::select! {
            served = &mut serving => match served {
                Ok(Ok(())) => None,
                other => Some(stopped(other)),
            },
            event = stop.recv() => Some(match event {
                // A second Ctrl-C means "now".
                Some(Stop::Signal(signal)) => {
                    format!("a second {signal} arrived, so open requests were cut off")
                }
                _ => format!("open requests did not finish in {grace:?} and were cut off"),
            }),
        };
    }
    // Nothing when the drain finished. After a timeout, this drops the
    // connections that are still open.
    serving.abort();

    // Phase 2: the plugins, inside what is left of the grace period.
    let remaining = grace.saturating_sub(started.elapsed());
    match owned(app).await {
        Some(mut app) => {
            let report = app.core.shutdown(reason, Some(remaining));
            plugins::say!(info, "{report}");
            if failed.is_none() && (!report.is_clean() || report.overran()) {
                failed = Some(format!("the plugins did not stop cleanly in {grace:?}"));
            }
        }
        None => plugins::say!(
            warn,
            "a request still holds the application, so no plugin stopped"
        ),
    }
    match failed {
        None => Ok(()),
        Some(why) => Err(why),
    }
}

/// Why the listener task ended before a clean drain.
fn stopped(
    served: Result<Result<(), brocanth_server::ServeError>, tokio::task::JoinError>,
) -> String {
    match served {
        Ok(Ok(())) => "the listener stopped with no signal".to_string(),
        Ok(Err(e)) => e.to_string(),
        Err(e) => e.to_string(),
    }
}

/// The application back from the connections that shared it, or `None` when
/// one of them still runs.
///
/// `App::shutdown` needs `&mut`. A finished connection task can be a few
/// instructions from dropping its clone, so this yields a few times first.
async fn owned(mut app: Arc<App>) -> Option<App> {
    for _ in 0..64 {
        match Arc::try_unwrap(app) {
            Ok(app) => return Some(app),
            Err(shared) => {
                app = shared;
                tokio::task::yield_now().await;
            }
        }
    }
    None
}

/// What the stop thread reports to [`run`].
enum Stop {
    /// SIGTERM or SIGINT, by name.
    Signal(&'static str),
    /// The grace period since the first signal is over.
    Deadline,
}

/// Watches for SIGTERM and SIGINT on a thread of its own. It reports each
/// signal, and the end of the grace period after the first one.
///
/// Not on the serving runtime. A handler is synchronous and holds its
/// worker, and while it runs that runtime can leave its signal and timer
/// drivers unpolled: a stop would wait for the slowest request, and the
/// grace period would not run out. This thread has a runtime of its own,
/// and the channel wakes `run` directly.
fn watch(grace: Duration) -> std::io::Result<tokio::sync::mpsc::UnboundedReceiver<Stop>> {
    let runtime = tokio::runtime::Builder::new_current_thread()
        .enable_all()
        .build()?;
    let mut signals = {
        let _guard = runtime.enter();
        Signals::listen()?
    };
    let (tx, rx) = tokio::sync::mpsc::unbounded_channel();
    std::thread::Builder::new()
        .name("stop".to_string())
        .spawn(move || {
            runtime.block_on(async move {
                let first = signals.next().await;
                if tx.send(Stop::Signal(first)).is_err() {
                    return;
                }
                let next = tokio::select! {
                    () = tokio::time::sleep(grace) => Stop::Deadline,
                    second = signals.next() => Stop::Signal(second),
                };
                let _ = tx.send(next);
            });
        })?;
    Ok(rx)
}

/// SIGTERM and SIGINT as one stream. An orchestrator sends SIGTERM, and
/// Ctrl-C is SIGINT.
#[cfg(unix)]
struct Signals {
    term: tokio::signal::unix::Signal,
    int: tokio::signal::unix::Signal,
}

#[cfg(unix)]
impl Signals {
    fn listen() -> std::io::Result<Self> {
        use tokio::signal::unix::{signal, SignalKind};
        Ok(Self {
            term: signal(SignalKind::terminate())?,
            int: signal(SignalKind::interrupt())?,
        })
    }

    /// The next signal, by name.
    async fn next(&mut self) -> &'static str {
        tokio::select! {
            _ = self.term.recv() => "SIGTERM",
            _ = self.int.recv() => "SIGINT",
        }
    }
}

/// Ctrl-C alone, where there is no SIGTERM.
#[cfg(not(unix))]
struct Signals;

#[cfg(not(unix))]
impl Signals {
    fn listen() -> std::io::Result<Self> {
        Ok(Self)
    }

    /// The next signal, by name.
    async fn next(&mut self) -> &'static str {
        let _ = tokio::signal::ctrl_c().await;
        "Ctrl-C"
    }
}

/// One request: the request hooks, the table, then the response hooks.
///
/// A request hook may answer instead of the page -- a CSRF refusal, a
/// 429 -- and then the page never runs. It may also rewrite the request
/// first, so the table reads the request as the hooks left it. The
/// response hooks see every answer, the 404 and the 405 included.
///
/// `begin_async` and not `begin`: it runs a hook that waits as well as
/// one that does not. With no hook that waits, it is ready on its first
/// poll.
pub async fn handle(app: &App, request: brocanth_server::Request) -> brocanth_server::Response {
    let Ok(mut request) = brocanth_server::bridge::to_core(request) else {
        // The path and the query do not make a URI. The table was not
        // asked, so this is not a 404.
        return brocanth_server::Response::text(400, "Bad Request");
    };
    let mut state = RequestState::new();
    let outcome = match app.core.begin_async(&mut request, &mut state).await {
        Begin::Dispatch => Ok(sent(dispatch(app, &request, &mut state).await)),
        Begin::Answered(outcome) => outcome,
    };
    let response = app.core.finish(&mut request, &mut state, outcome);
    brocanth_server::bridge::from_core(response)
}

/// The table, then the chain, then the handler.
///
/// The chain runs only for a route that matched. A 404 has no route and
/// therefore no context. `state` is what the request hooks wrote; the
/// page reads it through `Ctx::hooked`, and it goes back afterwards so
/// the response hooks read what the page left.
///
/// `async` because a page may be: the chain and the page are awaited
/// here, on the worker, and a page that waits on a database does not
/// hold a thread while it does.
pub async fn dispatch(app: &App, request: &hooks::Request, state: &mut RequestState) -> Response {
    let Some(verb) = Verb::from_http(request.method()) else {
        return Response::new(501, "Not Implemented");
    };
    let path = request.uri().path();
    match app.router.dispatch(verb, path) {
        Dispatch::Found { handler, matched } => {
            let mut ctx = Ctx::new(verb, path, matched.params())
                .with_plugins(app.core.context_for(request, state))
                .with_state(std::mem::take(state))
                .sent(
                    request.uri().query().unwrap_or(""),
                    request.body().clone(),
                    headers(request),
                );
            // The one place a failed handler becomes a response --
            // what Echo calls an HTTPErrorHandler. It sits inside
            // the chain's terminal, so a layer still sees what the
            // error turned into and can rewrite it.
            //
            // `handler(c)` is called before the future is built: the
            // future may borrow the context and nothing else.
            let response = app
                .stack
                .run(path, &mut ctx, |c: &mut Ctx| -> BoxFuture<'_, Response> {
                    let answer = handler(c);
                    Box::pin(
                        async move { answer.await.unwrap_or_else(IntoResponse::into_response) },
                    )
                })
                .await;
            *state = ctx.take_state();
            response
        }
        Dispatch::MethodNotAllowed { allow, .. } => Response {
            status: 405,
            headers: vec![("allow".to_string(), allow.to_string())],
            ..Default::default()
        },
        Dispatch::NotFound => Response::new(404, "Not Found"),
    }
}

//! The route table, which is this directory.
//!
//! One file is one route, and the path on disk is the URL: `index.rs` is `/`,
//! `about.rs` is `/about`, `users/[id].rs` is `/users/{id}`.
//!
//! Nothing below lists them. The macro reads this directory while the crate
//! is compiled and writes the `mod` lines and [`router`] from what is there,
//! so adding a page is adding a file and renaming one renames its route.
//! The verb comes off the handler's name: `handle` is GET, `handle_post` is
//! POST, and a `pub fn handle_*` that is not a verb is a build error rather
//! than a handler nothing reaches.
//!
//! `brocanth_pages::pages!` and `cargo expand` is how to read the result.

brocanth_pages::pages!("src/pages");

use brocanth::plugin_context::{Context, ContextItem, Has};

/// What a handler produces.
///
/// The framework owns rendering: a handler returns a value and never writes a
/// response itself, so a half-written body on an error is not something that
/// can happen. Echo and Gin hand the handler a writer and take an error back,
/// and both carry the scar -- once a 200 has gone out, the central error
/// handler cannot take it back. Here there is nothing to take back.
///
/// The one exception is opt-in: [`Response::stream`] and [`Response::events`]
/// answer with a body that is written after the handler returns. A stream
/// that fails halfway cannot become a 500, because the status is already on
/// the wire. Use it for server-sent events and large downloads, and nowhere
/// else.
pub struct Response {
    pub status: u16,
    pub body: String,
    pub headers: Vec<(String, String)>,
    /// A body the server polls after the handler returns. `None` for every
    /// page that answers with `body`, which is nearly all of them; when it
    /// is `Some`, `body` is ignored. Set it with [`Response::stream`].
    pub stream: Option<Streaming>,
}

/// Written by hand rather than derived, and the difference matters.
///
/// A derived `Default` would give `status: 0`, which is not an HTTP status, so
/// `Response { body: .., ..Default::default() }` would compile and be wrong.
/// 200 is what "I did not say" means for a response.
impl Default for Response {
    fn default() -> Self {
        Self {
            status: 200,
            body: String::new(),
            headers: Vec::new(),
            stream: None,
        }
    }
}

impl Response {
    pub fn new(status: u16, body: impl Into<String>) -> Self {
        Self {
            status,
            body: body.into(),
            ..Default::default()
        }
    }

    pub fn ok(body: impl Into<String>) -> Self {
        Self::new(200, body)
    }

    pub fn empty(status: u16) -> Self {
        Self::new(status, "")
    }

    /// A body written as it is produced: the first bytes reach the client
    /// before the last exist.
    ///
    /// The response hooks see the headers and never the body. The server
    /// tells the stream to end when it stops (`BodyStream::finish`), so a
    /// long-lived stream should implement that, or a deploy waits for it.
    #[allow(dead_code)]
    pub fn stream(status: u16, content_type: &str, stream: impl BodyStream) -> Self {
        Self {
            status,
            headers: vec![("content-type".to_string(), content_type.to_string())],
            stream: Some(Streaming::new(stream)),
            ..Default::default()
        }
    }

    /// Server-sent events: `text/event-stream`, and the two headers that stop
    /// a cache or a proxy from holding the events back.
    ///
    /// `brocanth_server::sse::channel` makes the stream from async code:
    ///
    /// ```ignore
    /// let (events, stream) = brocanth_server::sse::channel(16);
    /// tokio::spawn(async move {
    ///     // `Err` when the client left or the server is stopping.
    ///     let _ = events.send(Event::new("hello").with_event("greeting")).await;
    /// });
    /// Ok(Response::events(stream.heartbeat(Duration::from_secs(15))))
    /// ```
    #[allow(dead_code)]
    pub fn events(stream: impl BodyStream) -> Self {
        Self::stream(200, "text/event-stream", stream)
            .with_header("cache-control", "no-cache")
            .with_header("x-accel-buffering", "no")
    }

    /// For what the struct literal does not cover: several headers, added one
    /// at a time. Rust has no variadic generics, so a tuple return is
    /// fixed-arity and this is the unbounded form.
    #[must_use]
    pub fn with_header(mut self, name: impl Into<String>, value: impl Into<String>) -> Self {
        self.headers.push((name.into(), value.into()));
        self
    }

    /// Drops every header with this name, ignoring case.
    ///
    /// `with_header` appends, because `set-cookie` has to repeat. A
    /// single-valued header does not: two `content-type` lines are two
    /// claims and the client picks. Replacing one drops it first, which
    /// is what `set_header` below does.
    #[allow(dead_code)]
    #[must_use]
    pub fn without_header(mut self, name: &str) -> Self {
        self.headers
            .retain(|(held, _)| !held.eq_ignore_ascii_case(name));
        self
    }

    /// The one value for this header, replacing any already set.
    ///
    /// What to compose with when overriding a type an earlier step chose:
    /// `Ok(render_json(tpl)?.set_header("content-type", "application/problem+json"))`.
    #[allow(dead_code)]
    #[must_use]
    pub fn set_header(self, name: &str, value: impl Into<String>) -> Self {
        self.without_header(name).with_header(name, value)
    }
}

/// What a handler may return instead of building a [`Response`] by hand.
///
/// The Crystal brocanth overloads `write_response` on the handler's return
/// type, so a page returns the simplest thing that says what it means. Rust
/// spells the same idea as a trait: each handler still returns exactly one
/// type, and this is what lets different handlers return different ones.
pub trait IntoResponse {
    fn into_response(self) -> Response;
}

impl IntoResponse for Response {
    fn into_response(self) -> Response {
        self
    }
}

/// A body, answered 200.
impl IntoResponse for String {
    fn into_response(self) -> Response {
        Response::ok(self)
    }
}

impl IntoResponse for &str {
    fn into_response(self) -> Response {
        Response::ok(self)
    }
}

/// Nothing: 200 with no body. Crystal's `Nil` return.
impl IntoResponse for () {
    fn into_response(self) -> Response {
        Response::empty(200)
    }
}

/// A status and nothing else -- `Ok(404)`.
impl IntoResponse for u16 {
    fn into_response(self) -> Response {
        Response::empty(self)
    }
}

/// A body and a status.
impl<T: IntoResponse> IntoResponse for (T, u16) {
    fn into_response(self) -> Response {
        Response {
            status: self.1,
            ..self.0.into_response()
        }
    }
}

/// A body and headers.
impl<T: IntoResponse> IntoResponse for (T, Vec<(String, String)>) {
    fn into_response(self) -> Response {
        let mut response = self.0.into_response();
        response.headers.extend(self.1);
        response
    }
}

/// A body, a status and headers.
impl<T: IntoResponse> IntoResponse for (T, u16, Vec<(String, String)>) {
    fn into_response(self) -> Response {
        let mut response = self.0.into_response();
        response.status = self.1;
        response.headers.extend(self.2);
        response
    }
}

/// What a handler may fail with.
///
/// Deliberately **not** an `impl std::error::Error`, which is what lets the
/// blanket `From` below exist without colliding with the reflexive
/// `impl<T> From<T> for T`. `anyhow::Error` is this shape for the same reason.
/// The effect is that `?` works on anything in a handler.
#[derive(Debug)]
pub struct Error {
    pub status: u16,
    pub message: String,
    /// Headers the failure needs. A `content-type` for a body that is not
    /// HTML, a `www-authenticate` for a 401, a `retry-after` for a 429.
    ///
    /// `Response` always had these and `Error` did not, which made a
    /// machine-readable error body impossible to send.
    pub headers: Vec<(String, String)>,
}

impl Error {
    pub fn new(status: u16, message: impl Into<String>) -> Self {
        Self {
            status,
            message: message.into(),
            headers: Vec::new(),
        }
    }

    /// The default for a failure nobody classified: it is a bug here, not a
    /// mistake by the caller.
    pub fn internal(message: impl Into<String>) -> Self {
        Self::new(500, message)
    }

    /// The same, carrying a header.
    ///
    /// `allow(dead_code)` for the same reason the request readers below
    /// have it: it is here for the first handler that needs it, and a
    /// brand-new project has none yet.
    #[allow(dead_code)]
    #[must_use]
    pub fn with_header(mut self, name: impl Into<String>, value: impl Into<String>) -> Self {
        self.headers.push((name.into(), value.into()));
        self
    }
}

impl<E: std::error::Error + Send + Sync + 'static> From<E> for Error {
    fn from(error: E) -> Self {
        Self::internal(error.to_string())
    }
}

impl IntoResponse for Error {
    fn into_response(self) -> Response {
        // Everything a client sees for a 5xx is the status. The message is for
        // whoever is reading the logs, and putting it in the body is how an
        // internal path ends up on a stranger's screen.
        let withheld = self.status >= 500;
        if withheld {
            // So the message does reach the logs: a database error that came
            // up through `?` says here what failed, and the client sees none
            // of it.
            let (status, message) = (self.status, &self.message);
            crate::plugins::say!(error, "error {status}: {message}"; status, %message);
        }
        let body = if withheld {
            String::from("Internal Server Error")
        } else {
            self.message
        };
        let mut headers = self.headers;
        if withheld {
            // The body is replaced, so a `content-type` describing the
            // original no longer describes anything. Keeping it labelled a
            // plain sentence `application/problem+json` -- harmless while
            // `sent` sent two types and the client could pick the other,
            // and wrong now that the replacement is exact.
            headers.retain(|(name, _)| !name.eq_ignore_ascii_case("content-type"));
        }
        Response {
            status: self.status,
            body,
            // The rest survive: the body is withheld, not the framing the
            // client needs to read what is left.
            headers,
            stream: None,
        }
    }
}

/// What a handler is given. Middleware sees it too, and may change it before
/// the handler runs or after it returns.
pub struct Ctx {
    /// Which verb matched. A layer reads it; a page usually does not,
    /// because the verb is already in the name of the function it is in.
    #[allow(dead_code)]
    pub verb: Verb,
    pub path: String,
    /// Everything after the first `?`, without it. Empty when there was none.
    #[allow(dead_code)]
    pub query: String,
    /// The body as it arrived. Empty for the verbs that carry none, and for
    /// a context built by a test that did not set one.
    #[allow(dead_code)]
    pub body: Vec<u8>,
    params: Vec<(String, String)>,
    headers: Vec<(String, String)>,
    /// What the plugins put here for this request, or `None` for a context
    /// built by hand.
    plugins: Option<Context<plugins::Installed>>,
    /// What the request hooks wrote for this request: a `CsrfToken`, a
    /// client address. Empty for a context built by hand.
    state: RequestState,
}

impl Ctx {
    pub fn new(verb: Verb, path: &str, params: Params<'_>) -> Self {
        Self {
            verb,
            path: path.to_string(),
            query: String::new(),
            body: Vec::new(),
            params: params
                .iter()
                .map(|(name, value)| (name.to_string(), value.to_string()))
                .collect(),
            headers: Vec::new(),
            plugins: None,
            state: RequestState::new(),
        }
    }

    /// The rest of the request: what the client sent, rather than what the
    /// router matched.
    ///
    /// Two calls and not one, because a test that exercises a handler
    /// usually wants the first and not the second. `src/app.rs` makes both.
    ///
    /// The three are copied out of the server's request, which lives for
    /// the connection and not for the handler. One copy per request, and
    /// the handler owns what it reads.
    #[must_use]
    pub fn sent(mut self, query: &str, body: Vec<u8>, headers: Vec<(String, String)>) -> Self {
        self.query = query.to_string();
        self.body = body;
        self.headers = headers;
        self
    }

    /// What the plugins put on this request's context.
    ///
    /// `src/app.rs` calls it once per request, with a context the
    /// application made from its providers. A `Ctx` built in a test has
    /// none, which is what the `Option` says.
    #[must_use]
    pub fn with_plugins(mut self, plugins: Context<plugins::Installed>) -> Self {
        self.plugins = Some(plugins);
        self
    }

    /// What the request hooks wrote. `src/app.rs` hands it over before the
    /// page runs and takes it back after, for the response hooks.
    #[must_use]
    pub fn with_state(mut self, state: RequestState) -> Self {
        self.state = state;
        self
    }

    /// The hook state back, leaving this context an empty one.
    pub fn take_state(&mut self) -> RequestState {
        std::mem::take(&mut self.state)
    }

    /// The value a plugin published for this request.
    ///
    /// **Compiles only when some registered plugin provides `T`.** Ask for
    /// a type nothing installs and the build stops and names it; there is
    /// no string key to misspell and no guard to forget. That check is the
    /// whole point, and it is the compiler's, not this method's.
    ///
    /// `Option` is the other half, and it is not about `T`'s type. It is
    /// `None` in two cases: a `Ctx` built by hand in a test, which has no
    /// application behind it and so no providers; and a provider that
    /// declared `T` and inserted nothing, which `Plugin::provide` allows
    /// because it is one method and `Provides` is a separate declaration.
    /// A dispatched request whose provider wrote its value gets `Some`.
    ///
    /// The `_` is the position of `T` in the provider list, which is
    /// always inferred. It is there so two impls do not overlap, and it is
    /// the same `_` `Context::get` takes.
    ///
    /// ```ignore
    /// let session = ctx.provided::<Session, _>().expect("dispatched");
    /// ```
    #[allow(dead_code)]
    pub fn provided<T, I>(&self) -> Option<&T>
    where
        T: ContextItem,
        plugins::Installed: Has<T, I>,
    {
        // `and_then` with `try_get`, not `map` with `get`: `get` panics
        // when the provider list claims `T` and the provider inserted
        // nothing, which the generated plugin templates make easy --
        // `Provides` is unconditional and `provide` is often
        // `if let Some(..)`. A method returning `Option` must not abort
        // the process to say no.
        self.plugins.as_ref().and_then(|cx| cx.try_get::<T>())
    }

    /// A route parameter, or `None`.
    ///
    /// Never `""`. The Crystal prototype returned an empty string for a name
    /// nobody declared, so a typo read as a present-but-blank value and went
    /// unnoticed; `Option` is that bug deleted rather than documented.
    pub fn param(&self, name: &str) -> Option<&str> {
        self.params
            .iter()
            .find(|(key, _)| key == name)
            .map(|(_, value)| value.as_str())
    }

    /// The same, for a parameter the route declares and so must be there.
    ///
    /// Returns `Result` rather than `Option` so it composes with `?`. A name
    /// the route does not declare is this file disagreeing with its own
    /// filename, which is a 500 and not a 400: nobody sent a bad request.
    pub fn require(&self, name: &str) -> Result<&str, Error> {
        self.param(name)
            .ok_or_else(|| Error::internal(format!("the route declares no `{name}` parameter")))
    }
}

/// What the client sent, as opposed to what the router matched.
///
/// A second block, and `allow(dead_code)`, for one reason: the request is
/// on the context whether or not this project reads it yet. The other way
/// round is a context whose shape depends on which modules were chosen,
/// and then a page that starts reading a header needs the generator rather
/// than a line.
#[allow(dead_code)]
impl Ctx {
    /// A value a request hook wrote for this request, by its type.
    ///
    /// `ctx.hooked::<brocanth_middleware::CsrfToken>()` is the token a form
    /// carries. `None` when no registered hook wrote one, and always for a
    /// context built by hand.
    pub fn hooked<T: Send + Sync + 'static>(&self) -> Option<&T> {
        self.state.get::<T>()
    }

    /// The same value, to change. `None` when no hook wrote one.
    ///
    /// The state goes back to the hooks when the page returns, so a change
    /// made here is what `after_dispatch` and the response hooks read: a
    /// request hook puts a per-request log record on the state, and the
    /// page fills in the fields only it knows.
    pub fn hooked_mut<T: Send + Sync + 'static>(&mut self) -> Option<&mut T> {
        self.state.get_mut::<T>()
    }

    /// Puts a value on the request state for the hooks that run after the
    /// page, and returns the value it replaced.
    ///
    /// For a value that no request hook wrote first. A hook that reads it
    /// must also accept `None`: a 404, a 405 and a request that a request
    /// hook answered run no page.
    pub fn hook_insert<T: Send + Sync + 'static>(&mut self, value: T) -> Option<T> {
        self.state.insert(value)
    }

    /// A request header by name, case-insensitively, or `None`.
    pub fn header(&self, name: &str) -> Option<&str> {
        self.headers
            .iter()
            .find(|(key, _)| key.eq_ignore_ascii_case(name))
            .map(|(_, value)| value.as_str())
    }

    /// Every request header, in the order they arrived.
    pub fn headers(&self) -> impl Iterator<Item = (&str, &str)> {
        self.headers
            .iter()
            .map(|(key, value)| (key.as_str(), value.as_str()))
    }

    /// What the body is written in, or `None` when the client did not say.
    pub fn content_type(&self) -> Option<&str> {
        self.header("content-type")
    }

    /// The body as text, or `None` when it is not UTF-8.
    ///
    /// `Option` and not a lossy conversion: a body that is not text is a
    /// request to answer, not a string to repair.
    pub fn body_text(&self) -> Option<&str> {
        core::str::from_utf8(&self.body).ok()
    }
}

/// A handler, as the table stores it. One per verb a page answers.
///
/// Boxed rather than a bare `fn` pointer, because every handler returns its
/// own type and a `fn` pointer can only hold one of them. [`page`] and
/// [`page_async`] are what unify them, and they run once per route at
/// startup rather than per request.
///
/// It returns a future, so a page may wait -- on a database, on another
/// service -- without holding a thread. A page that waits on nothing is
/// still a plain `fn`, and [`page`] hands back its answer already finished.
pub type Handler =
    Box<dyn for<'a> Fn(&'a mut Ctx) -> BoxFuture<'a, Result<Response, Error>> + Send + Sync>;

/// Puts a `pub fn handle` in the table whatever it returns.
///
/// Generic over the return type, so `-> Result<impl IntoResponse, Error>` on
/// the handler and one concrete `Handler` in the table are the same thing.
/// The answer is ready when the handler returns; the future around it is
/// only there so a sync page and an async page share one table.
pub fn page<H, T>(handler: H) -> Handler
where
    H: Fn(&mut Ctx) -> Result<T, Error> + Send + Sync + 'static,
    T: IntoResponse,
{
    Box::new(move |ctx| {
        let answered = handler(ctx).map(IntoResponse::into_response);
        Box::pin(std::future::ready(answered))
    })
}

/// Puts a `pub async fn handle` in the table.
///
/// `pages!` writes the call, with a closure that awaits the page and turns
/// its answer into a `Response`. The closure is a `fn` pointer and not a
/// generic, because the page's future borrows the context and a generic
/// cannot name that borrow.
///
/// `allow(dead_code)`: nothing calls it until a page is `async`.
#[allow(dead_code)]
pub fn page_async(
    handler: for<'a> fn(&'a mut Ctx) -> BoxFuture<'a, Result<Response, Error>>,
) -> Handler {
    Box::new(handler)
}

/// Every template in this application.
///
/// The `.rsx` files sit next to the pages they render -- `index.rsx` beside
/// `index.rs` -- and `load_dir!` compiles the tree while this crate is being
/// compiled. Nothing is read from disk when a request arrives, and a template

///
/// A module of their own, because `load_dir!` emits a `pub mod index` for
/// `index.rsx` and `pages!` emits one for `index.rs`: unnested they collide.
/// The path is relative to the package root and not to this file, because a
/// proc macro cannot learn where it was written on stable Rust.
///
/// A file whose name is not an identifier gets the nearest one: `[id].rsx` is
/// `views::users::_id_`.
pub mod views {

/// Renders a page inside the layouts that cover it.
///
/// `layout.rsx` wraps every page in its directory and in every directory
/// below it, and they stack from the root down. A page never names one,
/// because which layout covers a page is a property of where the page is --
/// the rule Crystal's page_router uses for `layout.cr`.
///
/// The macro reads that off the directory tree at compile time: it knows
/// which file it was written in, walks up to `src/`, and wraps the page in
/// every `layout.rsx` it passes. Add one to a directory and every page under
/// it is wrapped, with nothing to register and no chain to keep in step.
///
/// Re-exported here so a page imports it from the same place as everything
/// else it uses, and so the layout rule lives beside the route table rather
/// than in a crate name.
pub use brocanth_pages::html;

/// The URL prefix a layout covers, for a layout under a parameter.
///
/// `html!` passes a layout the prefix it wraps, which is the marker dp-swap
/// reads to work out how much of a page two URLs share. A directory named
/// after a parameter has no prefix of its own until a request arrives, so
/// the macro hands this function the template and the request fills it:
/// `/{section}` under `/alpha/card-2` is `/alpha`.
///
/// Called only from the expansion, and only for a layout that sits under a
/// parameter. A marker reading `/[section]` would be the same string on
/// every section, and the client would conclude that two pages share a
/// layer they do not.
/// `allow(dead_code)`: `html!` emits a call to this only for a layout that
/// defines an `Oob` component, and no scaffolded layout does. It is here
/// for the first one that needs it, which is the same reason the request
/// readers on `Ctx` carry the attribute.
#[allow(dead_code)]
pub fn prefix(ctx: &Ctx, template: &str) -> String {
    let mut out = String::with_capacity(template.len());
    let mut rest = template;
    while let Some(open) = rest.find('{') {
        out.push_str(&rest[..open]);
        let Some(close) = rest[open..].find('}') else {
            break;
        };
        // A parameter the router did not match cannot happen: this template
        // is built from the directory the route was read out of, so the
        // name is one the pattern declares.
        out.push_str(ctx.param(&rest[open + 1..open + close]).unwrap_or_default());
        rest = &rest[open + close + 1..];
    }
    out.push_str(rest);
    out
}

/// A partial, and the replacements for the chrome it left behind.
///
/// A layout the client keeps is a layout this answer does not contain, which
/// is the point of asking for a partial and the reason its chrome goes
/// stale: the sidebar still marks the section you came from. A layout that
/// has something to correct defines `Oob` beside its `Layout`, and `html!`
/// renders one per skipped layer and passes them here.
///
/// They lead, because the static version of this writes them that way. The
/// client reads the whole body before it swaps anything, so the order is a
/// convention rather than a requirement, and one convention is better than
/// two.
/// `allow(dead_code)`: `html!` emits a call to this only for a layout that
/// defines an `Oob` component, and no scaffolded layout does. It is here
/// for the first one that needs it, which is the same reason the request
/// readers on `Ctx` carry the attribute.
#[allow(dead_code)]
pub fn render_with(
    out_of_band: Vec<String>,
    content: impl rsx::Template,
) -> Result<Response, Error> {
    let mut body = out_of_band.concat();
    body.push_str(&content.render_to_string()?);
    Ok(Response::ok(body))
}

/// One skipped layout's replacement chrome, rendered on its own.
///
/// `html!` calls this once per layer it left out, and `render_with` puts
/// the results in front of the page. A function here rather than a call to
/// `rsx::Template` inside the macro: the other two renders already come
/// back through this file, and the macro naming one engine was the reason
/// a project could have only that engine.
/// `allow(dead_code)`: `html!` emits a call to this only for a layout that
/// defines an `Oob` component, and no scaffolded layout does. It is here
/// for the first one that needs it, which is the same reason the request
/// readers on `Ctx` carry the attribute.
#[allow(dead_code)]
pub fn render_oob(content: impl rsx::Template) -> Result<String, Error> {
    Ok(content.render_to_string()?)
}

/// A page with no layout at all: a fragment, a partial, an htmx swap.
///
/// What `html!` expands into, once it has wrapped the page, so this is also
/// the escape hatch -- name a layout by hand and pass it here.
///
/// It returns the handler's own `Result` rather than a value, so a template
/// that bails propagates with `?` instead of turning into a blank 200. For
/// another status or a header, compose with the forms above:
/// `Ok((html!(ctx, tpl)?, 404))`, or `Ok(html!(ctx, tpl)?.with_header("x-kind", "page"))`.
///
/// Deliberately a function and not `impl<T: rsx::Template> IntoResponse for T`.
/// `()`, `Option<T>` and `Vec<T>` are templates, so a blanket impl overlaps
/// the ones above and does not compile -- and an `into_response` that cannot
/// fail would have to swallow a render error. A blanket impl could not apply
/// a layout either, since it would not know which page it was rendering.
pub fn render(content: impl rsx::Template) -> Result<Response, Error> {
    // `render_to_string` builds into its own String, so a page that fails
    // half way through its layout leaves nothing behind. There is no partial
    // body to take back, which is the property the rest of this file is
    // built on.
    Ok(Response::ok(content.render_to_string()?))
}

impl Ctx {
    /// How many layout layers the client already holds. `html!` asks it.
    ///
    /// Always zero: nothing here sends a partial request, so every page
    /// goes out whole. `brocanth add rsx-htmx` reads `DP-Partial` here.
    pub fn partial_depth(&self) -> u32 {
        0
    }
}

/// `allow(dead_code)`: it is here for the first page that queries, and a

    }

//! Written by `brocanth`. Do not edit -- `brocanth add` and `brocanth remove` rewrite this file.
//! Something else belongs here? Put it in `src/app.rs`, which is yours.
//!
//! The plugins this application contains, and the four things `src/app.rs`
//! asks this file for.
//!
//! Each plugin is also named in `register!`: being here makes it compile,
//! being named there makes it run. The core depends on none of them, and the
//! dependency arrow points this way and never back.
//!
//! The registration chain sits here and not in `src/app.rs` because the
//! generator rewrites it every time the module list changes, and `src/app.rs`
//! is yours.

pub mod db;
pub mod policy;
pub mod shape;

/// Every plugin this application contains, in dependency order.
///
/// A macro and not a function, and the reason is the builder's type:
/// registering a plugin that provides a context item widens
/// `AppBuilder`'s type-level provider list, so the type this chain ends
/// on is one nested associated type per plugin. A function would have to
/// spell that type out and re-spell it on every change. The macro expands
/// at its one call site in `src/app.rs`, where the compiler works it out
/// and nobody has to read it.
///
/// Each registration is still one line, and a line a reader can delete.
macro_rules! register {
    ($builder:expr) => {{
        use brocanth::plugin_api::Plugin;

        // No file and no TOML parser. Each plugin below starts from its
        // defaults and takes what this run was given on top: the
        // environment, then the flags.
        let given = crate::plugins::given()?;
        let builder = $builder;
        let builder = builder.plugin(crate::plugins::db::Database::new(
            crate::plugins::db_settings(&given)?,
        ));
        let builder = builder.plugin(crate::plugins::policy::Policies::new(
            crate::plugins::policy_settings(&given)?,
        ));
        builder
    }};
}
pub(crate) use register;

/// What this binary installs on a request's context, as a type.
///
/// One `Install` per registered plugin, innermost first, which is exactly
/// what `register!` builds. The two have to be the same type: if they
/// drift this file does not compile, which is why the list is written
/// once and both readers derive from it.
///
/// A plugin that puts nothing on the context folds away, because
/// `NoContext::Install<E>` is `E`, so this is `Base` in a project whose
/// plugins all provide nothing.
type WithDb = After<crate::plugins::db::Database, Base>;
type WithPolicy = After<crate::plugins::policy::Policies, WithDb>;
pub type Installed = WithPolicy;

/// Which hook families this binary installed, as a type.
///
/// One `Both` per plugin registered with `plugin_hooking`, innermost first,
/// which is exactly the record `register!` builds. A hook family nobody
/// installed costs nothing, and this type is how the compiler knows: every
/// hook entry point opens with a constant `if` on this record.
pub type Hooks = brocanth::plugin_hooks::NoHooks;
/// What this binary puts on the context, said once at startup.
///
/// The lines are a demonstration and they are load-bearing: the installed
/// provider set lives in the application's *type*, so an `extract` for a
/// plugin nobody registered does not compile. That is the modularity claim,
/// checked by the compiler rather than written in a README.
///
/// A handler reads the same values through `Ctx::provided`, which carries
/// the same check. This runs at startup so a misconfiguration is a failure
/// to start rather than a failure on the first request that needed it.
///
/// A macro for the reason `register!` is one: the context's type is the
/// builder's type, and neither is worth spelling.
macro_rules! announce {
    ($app:expr) => {{
        let cx = $app.context();
        let db: &brocanth_db::Db = cx.extract();
        let connections = db.pool().size();
        crate::plugins::say!(info, "db: {connections} connection(s) open"; connections);
        let rules: &crate::plugins::policy::Rules = cx.extract();
        let dialect = &rules.dialect;
        crate::plugins::say!(info, "policy: compiling for {dialect:?}"; ?dialect);
    }};
}
pub(crate) use announce;

/// A line about the application itself: startup, the listener, the stop.
///
/// A line on stderr, prefixed with the application name. The level and the
/// fields after `;` are what `observability-logs` would record; without it
/// there is no subscriber, so they are dropped and the message is printed.
///
/// ```ignore
/// plugins::say!(info, "listening on http://{bound}"; %bound);
/// ```
macro_rules! say {
    ($level:ident, $message:literal $(; $($field:tt)+)?) => {
        eprintln!("{}: {}", crate::app::APP, format_args!($message))
    };
}

///
/// A layer may wait, so it returns a future. The lifetime is named

/// One value this run was given, above the defaults.
pub struct Given {
    /// What it sets.
    pub setting: &'static Setting,
    /// Where it came from, as typed: `--cookie`, or `$MY_APP_SESSIONS__COOKIE`.
    pub from: String,
    /// The value, as typed. Empty is `None` for an `Option` setting.
    pub value: String,
}

/// The name each flag answers to after the renames in src/modules/, whether
/// the flags are read or not.
///
/// A name that is not spelled as a flag, and two flags under one name, each
/// stop the binary: each would otherwise be a flag that does nothing, or does
/// something else. A rename of a key no flag sets does not compile: `Key` has
/// no variant for it.
fn names() -> Result<Vec<(&'static Setting, &'static str)>, String> {
    let renames = renames();
    let mut names: Vec<(&'static Setting, &'static str)> = Vec::new();
    for setting in SETTINGS.iter().filter(|setting| setting.flag) {
        let name = renames
            .iter()
            .find(|(path, _)| *path == setting.path)
            .map_or(setting.path, |(_, flag)| flag.name());
        if name.is_empty() {
            continue;
        }
        let spelled = name
            .chars()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || "._-".contains(c));
        let taken = ["help", "version", "settings"].contains(&name);
        if !spelled || name.starts_with(['-', '.', '_']) || taken {
            return Err(format!(
                "{} names a flag `{name}`\nhelp: a flag is lower case, digits, `.`, `_` and `-`, starts with a letter or a digit, and is not `help`, `version` or `settings`",
                setting.file
            ));
        }
        if let Some((other, _)) = names.iter().find(|(_, taken)| *taken == name) {
            let files = if other.file == setting.file {
                other.file.to_string()
            } else {
                format!("{} or {}", other.file, setting.file)
            };
            return Err(format!(
                "`--{name}` would set both `{}` and `{}`\nhelp: rename one of them in {files}",
                other.path, setting.path
            ));
        }
        names.push((setting, name));
    }
    Ok(names)
}

/// Every rename in src/modules/, as the setting's path and its flag.
fn renames() -> Vec<(&'static str, Flag)> {
    let mut renames = Vec::new();
    for (key, flag) in crate::modules::brocanth::MODULE.rename {
        renames.push((key.path(), *flag));
    }
    for (key, flag) in crate::modules::db::MODULE.rename {
        renames.push((key.path(), *flag));
    }
    for (key, flag) in crate::modules::policy::MODULE.rename {
        renames.push((key.path(), *flag));
    }
    renames
}

/// Whether a module's flags are read: its file says `Flags::On` or
/// `Flags::Off`, or follows the project rule. Without a config file the
/// project rule is on: the flags and the environment are the only sources.
fn flags_on(rule: Flags) -> bool {
    match rule {
        Flags::On | Flags::Project => true,
        Flags::Off => false,
    }
}

/// The flags this run reads: each module file decides, and
/// src/modules/brocanth.rs holds the project rule. Every name is checked
/// first, read or not.
pub fn flag_names() -> Result<Vec<(&'static Setting, &'static str)>, String> {
    Ok(names()?
        .into_iter()
        .filter(|(setting, _)| flags_on(setting.flags))
        .collect())
}

/// What this run was given above the defaults, lowest first: each variable
/// that is set, then each flag in the order typed. A later value for one key
/// replaces an earlier one.
///
/// `--name=VALUE` and `--name VALUE`. Every other argument is left alone: it
/// is a command's, and the command line's own parser has already refused what
/// nobody declared.
pub fn given() -> Result<Vec<Given>, String> {
    let mut given = Vec::new();
    for setting in SETTINGS {
        if let Some(value) = std::env::var_os(setting.env) {
            given.push(Given {
                setting,
                from: format!("${}", setting.env),
                value: value.to_string_lossy().into_owned(),
            });
        }
    }
    let names = flag_names()?;
    let args: Vec<String> = std::env::args_os()
        .skip(1)
        .map(|arg| arg.to_string_lossy().into_owned())
        .collect();
    let mut rest = args.iter();
    while let Some(arg) = rest.next() {
        if arg == "--" {
            break;
        }
        let Some(written) = arg.strip_prefix("--") else {
            continue;
        };
        let (name, inline) = match written.split_once('=') {
            Some((name, value)) => (name, Some(value.to_string())),
            None => (written, None),
        };
        let Some((setting, name)) = names.iter().copied().find(|(_, known)| *known == name) else {
            continue;
        };
        let value = match inline {
            Some(value) => value,
            None => rest
                .next()
                .cloned()
                .ok_or_else(|| format!("--{name} needs a value\nhelp: write --{name}=VALUE"))?,
        };
        given.push(Given {
            setting,
            from: format!("--{name}"),
            value,
        });
    }
    Ok(given)
}

/// Refuses anything on the command line that is not a flag, and answers
/// `--help` and `--settings`. `true` when it printed one of those, so `main`
/// stops there.
///
/// This binary has no command line of its own: `brocanth add commands` gives
/// it one, and `clap` does this.
pub fn read_line() -> Result<bool, String> {
    let names = flag_names()?;
    let mut rest = std::env::args_os()
        .skip(1)
        .map(|arg| arg.to_string_lossy().into_owned());
    while let Some(arg) = rest.next() {
        if arg == "--help" || arg == "-h" {
            println!("Flags:");
            for (setting, name) in &names {
                println!("  --{name} <VALUE>  {}", setting.help);
            }
            println!("  --settings  every settings layer, then the merged values");
            return Ok(true);
        }
        if arg == "--settings" {
            print!("{}", settings_report()?);
            return Ok(true);
        }
        let written = arg.strip_prefix("--").unwrap_or_default();
        let name = written.split_once('=').map_or(written, |(name, _)| name);
        if name.is_empty() || !names.iter().any(|(_, known)| *known == name) {
            return Err(format!(
                "unknown argument `{arg}`\nhelp: `--help` lists the flags"
            ));
        }
        // `--name VALUE`: the next argument is the value, not a flag.
        if !written.contains('=') {
            rest.next();
        }
    }
    Ok(false)
}

/// One value, as the type of the field it sets.
fn parsed<T>(given: &Given) -> Result<T, String>
where
    T: std::str::FromStr,
    T::Err: std::fmt::Display,
{
    given
        .value
        .parse()
        .map_err(|e| format!("{} is {:?}: {e}", given.from, given.value))
}

/// One value for an `Option` field: empty, or the setting's declared alias,
/// is `None`, decided at this layer.
fn optional<T>(given: &Given) -> Result<Option<T>, String>
where
    T: std::str::FromStr,
    T::Err: std::fmt::Display,
{
    if given.value.is_empty() || given.setting.none_as == Some(given.value.as_str()) {
        Ok(None)
    } else {
        parsed(given).map(Some)
    }
}

/// Every settings layer on its own, lowest first, then the merged values.
/// A secret is masked in every layer.
pub fn settings_report() -> Result<String, String> {
    let given = given()?;
    let secret = |path: &str| SETTINGS.iter().any(|s| s.path == path && !s.flag);
    let shown = |path: &str, value: String| {
        if secret(path) {
            "(secret)".to_string()
        } else {
            value
        }
    };
    let width = SETTINGS.iter().map(|s| s.path.len()).max().unwrap_or(0);
    let mut out = String::from("layers, lowest precedence first\n\n   1  defaults\n");
    for (path, value) in defaults() {
        out.push_str(&format!(
            "        {path:<width$}  = {}\n",
            shown(path, value)
        ));
    }
    for (n, label, prefix) in [(2, "environment", "$"), (3, "flags", "--")] {
        out.push_str(&format!("\n  {n:>2}  {label}\n"));
        let mine: Vec<&Given> = given
            .iter()
            .filter(|g| g.from.starts_with(prefix))
            .collect();
        if mine.is_empty() {
            out.push_str("        (sets nothing)\n");
        }
        for g in mine {
            let value = shown(g.setting.path, format!("{:?}", g.value));
            let path = g.setting.path;
            out.push_str(&format!("        {path:<width$}  = {value}  {}\n", g.from));
        }
    }
    out.push_str("\nmerged\n");
    for (path, value) in merged(&given)? {
        let layer = match given.iter().rev().find(|g| g.setting.path == path) {
            Some(g) if g.from.starts_with('$') => "2 environment",
            Some(_) => "3 flags",
            None => "1 defaults",
        };
        let value = shown(path, value);
        out.push_str(&format!(
            "  {path:<w$}  = {value}  {layer}\n",
            w = width + 6
        ));
    }
    Ok(out)
}

//! `/search` -- the site search, over `search_documents`.
//!
//! `?q=words&page=2`. The term is read by Postgres' `websearch_to_tsquery`:
//! words, `"a phrase"`, `-not`, `or`. A typo in a title still matches, by
//! trigram. The markup is in `search.rsx` beside this file.

use brocanth_search::documents::{self, Document, DocumentColumn, Hit};
use brocanth_search::pagination::{query, Pager};
use brocanth_search::policy::{Dialect, Pred};

use crate::pages::{html, views, Ctx, Error, IntoResponse};

/// Results per page: `per_page` in `src/modules/search.rs`.
const PER_PAGE: u64 = crate::modules::search::MODULE.per_page;

/// GET. An empty `q` shows the form alone.
pub async fn handle(ctx: &mut Ctx) -> Result<impl IntoResponse, Error> {
    let term = query::param(&ctx.query, "q").unwrap_or_default();
    let params = ctx.page_params()?;
    // The rule. Every row of the site index is public: index only what any
    // visitor may read. To hide rows, write the rule here. It goes into the
    // `WHERE` of the page and of the count, so a hidden row is never ranked.
    let rule = Pred::<DocumentColumn>::Always;
    let condition = rule.condition::<Document>(Dialect::Postgres)?;
    let hits = documents::index()
        .search::<Hit>(ctx.db(), &condition, &term, params.page, PER_PAGE)
        .await?;
    let pager = Pager::new(hits.page, &ctx.links());
    html!(
        ctx,
        views::search::Search {
            term: &term,
            hits: &hits.items,
            pager: &pager,
        }
    )
}

//! The database, as this application registers it.
//!
//! One Postgres pool, built at startup and shared by every request. It opens
//! no connection in the build; `serve` checks it once before it listens. A page
//! reads it with `ctx.db()`. Another plugin reads it from the application's
//! services: `lc.require_service::<brocanth_db::Db>()`, from its own
//! `before_start`, registered after this one.
//!
//! `brocanth-db` is in `Cargo.toml` because this file is here, and through no
//! other route. Delete the registration in `src/plugins/mod.rs` and this
//! file, and the application links no SQL driver.

use std::time::Duration;

use brocanth::plugin_api::{Lifecycle, Plugin, PluginError};
use brocanth::plugin_context::{ContextBuilder, ProvidesContext};
use brocanth::plugin_shutdown::ShutdownCx;
use brocanth_db::{Db, DbConfig, DbError};

/// What this plugin is built from: `[db]`.
///
/// A duration is a whole number of seconds, and 0 turns an optional one
/// off.
///
/// Nothing deserialises it: `main` builds it from a literal, and the
/// compiler checks that literal. `Plugin::Config` carries no bounds the
/// core imposes, so there is no deserialiser here to link.
#[derive(Debug)]
pub struct Settings {
    /// `postgres://user:password@host:5432/database`.
    pub url: String,
    /// The most connections the pool opens.
    pub max_connections: u32,
    /// The connections the pool keeps open when idle.
    pub min_connections: u32,
    /// Seconds a statement waits for a connection, opening one included.
    pub acquire_timeout: u64,
    /// Seconds before an idle connection closes. 0 keeps it open.
    pub idle_timeout: u64,
    /// Seconds before the server cancels a statement. 0 is no limit.
    pub statement_timeout: u64,
    /// The schemas for a table name with no schema. Empty is the server's
    /// default.
    pub search_path: Vec<String>,
    /// Every transaction is read-only.
    pub read_only: bool,
    /// The name `pg_stat_activity` shows. Empty sends none.
    pub application_name: String,
    /// A PEM bundle to trust, with `verify-full`. `None` uses the system's
    /// roots.
    pub ca_file: Option<String>,
}

impl Default for Settings {
    fn default() -> Self {
        Self {
            url: "postgres://localhost/development".to_string(),
            max_connections: 10,
            min_connections: 0,
            acquire_timeout: 30,
            idle_timeout: 600,
            statement_timeout: 0,
            search_path: Vec::new(),
            read_only: false,
            application_name: "my-blog".to_string(),
            ca_file: None,
        }
    }
}

impl Settings {
    /// The pool's configuration, from these settings.
    pub fn config(&self) -> DbConfig {
        let seconds = |secs: u64| (secs > 0).then(|| Duration::from_secs(secs));
        let mut config = DbConfig::new(&self.url);
        config.max_connections = self.max_connections;
        config.min_connections = self.min_connections;
        config.acquire_timeout = Duration::from_secs(self.acquire_timeout);
        config.idle_timeout = seconds(self.idle_timeout);
        config.statement_timeout = seconds(self.statement_timeout);
        config.search_path = self.search_path.clone();
        config.read_only = self.read_only;
        config.application_name =
            Some(self.application_name.clone()).filter(|name| !name.is_empty());
        config.ca_file = self.ca_file.clone().map(Into::into);
        config
    }
}

pub struct Database {
    config: DbConfig,
    /// Open from `before_start` until `shutdown`.
    db: Option<Db>,
}

impl Plugin for Database {
    type Config = Settings;
    type Provides = ProvidesContext<Db>;
    const NAME: &'static str = "db";

    fn new(settings: Settings) -> Self {
        Self {
            config: settings.config(),
            db: None,
        }
    }

    fn configure(&mut self, _lc: &mut Lifecycle<'_>) -> Result<(), PluginError> {
        if self.config.url.is_empty() {
            return Err(PluginError::new(
                "[db] url is required; there is no default database\n\
                 help: set `url` under [db] in the config file, or the variable for `db.url`",
            ));
        }
        Ok(())
    }

    fn before_start(&mut self, lc: &mut Lifecycle<'_>) -> Result<(), PluginError> {
        // The hook is synchronous and connecting is not, so the pool opens
        // lazily and nothing here waits. Waiting would hang a current-thread
        // runtime, where this thread is the one that drives the I/O. The
        // pool spawns its upkeep task, so it needs the runtime entered.
        let runtime = tokio::runtime::Handle::try_current().map_err(|e| {
            PluginError::new(format!(
                "[db] needs a Tokio runtime at startup to open its pool ({e})"
            ))
        })?;
        let db = {
            let _entered = runtime.enter();
            Db::connect_lazy(&self.config).map_err(|e| PluginError::new(format!("[db] {e}")))?
        };
        // Ready while `SELECT 1` answers on a pooled connection. `serve`
        // awaits this once before it listens, so a wrong URL, password or
        // certificate stops the start there and not on the first request.
        let probe = db.clone();
        let url = self.config.redacted_url();
        lc.ready_when(move || {
            let probe = probe.clone();
            let url = url.clone();
            async move {
                probe.ping().await.map_err(|e| match e {
                    DbError::Connect { .. } => format!("connecting to {url}: {e}"),
                    other => other.to_string(),
                })
            }
        });
        // For a plugin registered after this one: the model layer wraps the
        // same pool, so there is one set of connections to size.
        lc.services_mut().insert(db.clone());
        self.db = Some(db);
        Ok(())
    }

    fn shutdown(&mut self, sd: &mut ShutdownCx<'_>) -> Result<(), PluginError> {
        let Some(db) = self.db.take() else {
            return Ok(());
        };
        // Without a runtime there is nothing to wait on: the connections
        // close when the pool drops.
        let Ok(runtime) = tokio::runtime::Handle::try_current() else {
            return Ok(());
        };
        let remaining = sd.remaining();
        // `close` waits for the connections in use to come back, so it
        // waits inside what is left of the grace period and no longer.
        let close = async move {
            match remaining {
                Some(left) => tokio::time::timeout(left, db.close()).await.is_ok(),
                None => {
                    db.close().await;
                    true
                }
            }
        };
        // On a current-thread runtime this thread drives the I/O, so a wait
        // here never ends. The close runs as a task, under the same limit.
        if runtime.runtime_flavor() == tokio::runtime::RuntimeFlavor::CurrentThread {
            runtime.spawn(close);
            return Ok(());
        }
        // On a multi-thread runtime the workers drive it, and a scoped
        // thread waits without blocking one of them.
        std::thread::scope(|scope| {
            scope
                .spawn(|| runtime.block_on(close))
                .join()
                .map_err(|_| PluginError::new("[db] the closing thread panicked"))
        })?
        .then_some(())
        .ok_or_else(|| PluginError::new("[db] connections were still in use at the deadline"))
    }

    fn provide(&self, cx: &mut ContextBuilder) {
        // `before_start` runs before the server is handed the application,
        // so the pool is here by the time a request builds a context.
        if let Some(db) = &self.db {
            cx.insert(db.clone());
        }
    }
}
===== src/plugins/policy/mod.rs =====
//! Authorisation, as this application writes it.
//!
//! A rule is a predicate and a check is a witness. `PostPolicy` below is the
//! first; `Verified` is the second; the two are not alternatives.
//!
//! `brocanth-policy` is in `Cargo.toml` because this file is here, and through
//! no other route. Delete the registration in `src/app.rs` and delete this
//! file, and the application has no rule algebra and links nothing to have one
//! with.
//!
//! # The bug this file cannot have
//!
//! PocketBase validates a rule against an empty request object, so a
//! misspelled field in a negated rule becomes `'' IS NOT 'banned'`, which is
//! true, which grants access -- silently, for ever. Two things here make that
//! impossible, and both are visible below:
//!
//! 1. **A field name is not a string.** `Pred::eq` takes a `PostColumn`, so
//!    `PostColumn::Autor` is a compile error. There is no
//!    `Pred::eq("author_id", ..)` to fall back to and no string-to-filter
//!    parser anywhere in this framework.
//! 2. **An absent value is a word somebody wrote.** An `Option` does not turn
//!    into a `Bound`. The author writes `Bound::or_deny`, `or_allow` or
//!    `or_null`, and the word is in the diff for a reviewer to disagree with.
//!
//! # Where the rules go
//!
//! `Post` is here so that this file compiles into something that runs. Move
//! it, and whatever follows it, into `src/policies/` once there is more than
//! one -- what must stay here is the plugin, because that is what reads the
//! dialect and puts it on the context.
//!
//! The column enum is written by hand on purpose: the `policy!` macro is
//! deliberately late, after a real application has used the algebra, so that
//! the syntax is informed rather than imitated.
//!
//! # Listing, and then writing
//!
//! A list is a `WHERE` fragment, so the paginator keeps its totals:
//!
//! ```ignore
//! let rules = ctx.provided::<Rules, _>().expect("a dispatched request");
//! let condition = PostPolicy.list(&visitor).condition::<Post>(rules.dialect)?;
//! // "SELECT .. FROM posts WHERE " + condition.sql(), bound to condition.binds()
//! ```
//!
//! A write is the same predicate carried into the statement, which is what
//! closes the window between the check and the change:
//!
//! ```ignore
//! let admission = PostPolicy.update(&visitor).admission::<Post>(rules.dialect, id)?;
//! let rows = /* run admission.query(), count the rows */;
//! let verified = admission.resolve(rows)?;          // zero rows is a denial
//! let write = verified.update(rules.dialect, [(PostColumn::Status, "published".into())])?;
//! // run write.query(); then:
//! write.applied(affected)?;                          // zero affected is RaceLost
//! ```
//!
//! `Verified` is taken by value everywhere it is used, and a debug build fails
//! the test that checked one and then forgot to spend it.

use brocanth::plugin_api::{Lifecycle, Plugin, PluginError};
use brocanth::plugin_context::{ContextBuilder, ContextItem, ProvidesContext};
use brocanth_policy::{Bound, Column, Dialect, Entity, EntityPred, FieldMask, Ident, Policy, Pred};

// ---------------------------------------------------------------------------
// An entity
// ---------------------------------------------------------------------------

/// The columns of `posts`, as an enum, because a rule is written in them.
#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
pub enum PostColumn {
    /// Primary key.
    Id,
    /// Who wrote it.
    AuthorId,
    /// `draft` or `published`.
    Status,
}

impl Column for PostColumn {
    /// Required rather than derived: a changed-field name that is not one of
    /// these is rejected instead of quietly permitted.
    const ALL: &'static [Self] = &[Self::Id, Self::AuthorId, Self::Status];

    fn name(&self) -> &'static str {
        match self {
            Self::Id => "id",
            Self::AuthorId => "author_id",
            Self::Status => "status",
        }
    }
}

/// The table the rules below are about.
pub struct Post;

impl Entity for Post {
    type Column = PostColumn;
    /// Checked at compile time, because a table name is program text and the
    /// only honest moment to reject one is while the program is being built.
    const TABLE: Ident = Ident::must("posts");
    const PRIMARY_KEY: Self::Column = PostColumn::Id;
}

// ---------------------------------------------------------------------------
// A subject, and the rules about it
// ---------------------------------------------------------------------------

/// Whoever is asking. `brocanth-policy` never looks inside it.
///
/// An anonymous visitor is this application's own `None`, turned into a
/// decision at the point a rule reads it.
#[derive(Clone, Debug, Default)]
pub struct Visitor {
    /// The signed-in account, when there is one.
    pub account_id: Option<i64>,
    /// Whether this one may see everything.
    pub admin: bool,
}

impl Visitor {
    /// Nobody. What a request with no session carries.
    pub fn anonymous() -> Self {
        Self::default()
    }
}

/// The rules for `posts`.
///
/// Only `list` is required, because list authorisation is the one that cannot
/// be done per object: a paginator that filters after reading lies about its
/// totals. Everything else defaults to deny, which is what makes shipping this
/// file as it stands safe rather than lucky.
#[derive(Clone, Copy, Debug, Default)]
pub struct PostPolicy;

impl Policy for PostPolicy {
    type Entity = Post;
    type Subject = Visitor;

    fn list(&self, who: &Visitor) -> EntityPred<Post> {
        if who.admin {
            // `Pred::is_always` reports this, so a test can assert which
            // subjects see everything.
            return Pred::Always;
        }
        Pred::any([
            Pred::eq(PostColumn::Status, "published"),
            // `or_deny` is the word that says what an anonymous visitor's
            // missing id means: they own nothing.
            Pred::eq(PostColumn::AuthorId, Bound::or_deny(who.account_id)),
        ])
    }

    fn create(&self, who: &Visitor) -> EntityPred<Post> {
        Pred::eq(PostColumn::AuthorId, Bound::or_deny(who.account_id))
    }

    fn update(&self, who: &Visitor) -> EntityPred<Post> {
        if who.admin {
            return Pred::Always;
        }
        Pred::eq(PostColumn::AuthorId, Bound::or_deny(who.account_id))
    }

    fn writable(&self, who: &Visitor) -> FieldMask<PostColumn> {
        if who.admin {
            FieldMask::except([PostColumn::Id])
        } else {
            // The author edits the body, never the status. The mask is the
            // other half of the rule: `update` says which rows, this says
            // which columns.
            FieldMask::only([PostColumn::AuthorId])
        }
    }
}

// ---------------------------------------------------------------------------

/// What this plugin is built from.
///
/// Nothing deserialises it: `main` builds it from a literal, and the
/// compiler checks that literal. `Plugin::Config` carries no bounds the
/// core imposes, so there is no deserialiser here to link.
#[derive(Debug)]
pub struct Settings {
    /// `postgres` or `sqlite`. A string here and an enum below: a config file
    /// is text, and the translation is checked at startup where a typo can
    /// still be answered.
    pub dialect: String,
}

impl Default for Settings {
    fn default() -> Self {
        Self {
            dialect: "postgres".to_string(),
        }
    }
}

/// What this plugin puts on the context.
///
/// The dialect, and only the dialect. Every compile entry point still takes
/// one as an argument -- the call site says which it meant -- and this is
/// where the call site gets it, so one database is named once.
///
/// The policies are not here. `PostPolicy` is a unit struct and a value, not
/// a service: a handler writes `PostPolicy.list(&who)` and has nothing to
/// look up. What a handler cannot work out on its own is which dialect this
/// deployment is, so that is what the context carries.
pub struct Rules {
    /// Which dialect this application's conditions are compiled for.
    pub dialect: Dialect,
}

impl ContextItem for Rules {}

pub struct Policies {
    settings: Settings,
    /// Read at `configure` and not in `new`, because turning a string into a
    /// dialect can fail and `Plugin::new` cannot.
    dialect: Option<Dialect>,
}

impl Plugin for Policies {
    type Config = Settings;
    type Provides = ProvidesContext<Rules>;
    const NAME: &'static str = "policy";

    fn new(config: Settings) -> Self {
        Self {
            settings: config,
            dialect: None,
        }
    }

    fn configure(&mut self, _lc: &mut Lifecycle<'_>) -> Result<(), PluginError> {
        let dialect = match self.settings.dialect.as_str() {
            "postgres" => Dialect::Postgres,
            "sqlite" => Dialect::Sqlite,
            other => {
                return Err(PluginError::new(format!(
                    "[policy] dialect is {other:?}; write postgres or sqlite"
                )))
            }
        };

        // Compile one rule at startup. It costs a string and it answers the
        // only question this file can get wrong without anybody noticing: a
        // column whose name is not a bare identifier fails the query rather
        // than escaping into it, and finding that out here is better than
        // finding it out on the first list request.
        let condition = PostPolicy
            .list(&Visitor::anonymous())
            .condition::<Post>(dialect)
            .map_err(|e| PluginError::new(format!("[policy] posts: {e}")))?;
        let sql = condition.sql();
        crate::plugins::say!(info, "[policy] posts, anonymous: {sql}"; sql);

        self.dialect = Some(dialect);
        Ok(())
    }

    fn provide(&self, cx: &mut ContextBuilder) {
        // `configure` runs before the server is handed the application, so
        // the dialect is here by the time any request builds a context.
        if let Some(dialect) = self.dialect {
            cx.insert(Rules { dialect });
        }
    }
}

# Working in My Blog

A brief for a coding agent. Written by `brocanth`, so the lists
below are this project's and not the framework's in general.

## Check your work

Run these after every change, in this order. They are ordered by how
fast they fail.

```
cargo test
cargo clippy --all-targets -- -D warnings
cargo fmt --check
```

The route table is `pages::router()` and the test beside it asserts the
shape of the directory. Read it rather than guessing what covers a
request.

## What this binary contains

- `core`  the router, the context, the plugin seams
- `rsx`  `.rsx` templates compiled into the binary; without it a page builds a String
- `db`  one Postgres pool and sea-query statements, read as ctx.db(); without it no SQL driver
- `policy`  typed predicates over an entity's columns; no ORM, no driver, no core
- `pagination`  offset and keyset pages, with links that keep the query; no dependency
- `filters`  typed filters and a sort from the query string, ANDed with the policy's rule in the WHERE
- `search`  Postgres full-text and trigram search: a site index, its migration, a page at /search

**Absent:** `logging`, `observability`, `observability-logs`, `observability-health`, `observability-metrics`, `observability-tracing`, `analytics`, `beacon`, `config`, `sessions`, `model`, `cache`, `cache-memory`, `cache-redis`, `jobs`, `jobs-memory`, `flags`, `audit`, `mail`, `auth`, `auth-oauth`, `auth-screens`, `auth-lockout`, `superuser`, `roles`, `storage`, `storage-routes`, `input`, `client`, `middleware`, `middleware-csrf`, `middleware-security-headers`, `middleware-rate-limit`, `middleware-conditional-get`, `middleware-compression`, `flash`, `gtag`, `openapi`, `validation`, `resource`, `admin`, `admin-model`, `admin-actions`, `webhooks`, `webhooks-in`, `webhooks-out`, `realtime`, `realtime-sse`, `realtime-websocket`, `commands`, `jt`, `rsx-tailwind`, `rsx-htmx`, `sanitize`, `static`, `robots`, `sitemap`, `i18n`, `i18n-routing`, `meta`, `errors`, `docker`, `docker-kubernetes`, `ci`, `ci-github` — not compiled, not linked, absent from `Cargo.lock`. Do not write code that uses them; add the dependency and name the plugin in `main` first.

That is the framework's central rule: a capability nobody asked for is
not compiled and not linked. If you need one, add it to `Cargo.toml`
*and* name its plugin in `src/main.rs`. Nothing registers itself.

## Adding a route

A route is a file. `src/pages/about.rs` is `/about`;
`src/pages/users/[id].rs` is `/users/{id}`. **Write the file. That is
the whole of it** -- no `mod` line, no row to register, because there is
no list. `brocanth_pages::pages!` in `src/pages/mod.rs` reads this
directory while the crate compiles and writes both.

The verb comes off the handler's name: `handle` is GET, then
`handle_head`, `handle_post`, `handle_put`, `handle_patch`,
`handle_delete`, `handle_options`. A `pub fn handle_*` outside that
list is a build error naming the list, rather than a handler nothing
reaches; a page with no handler is a build error too. A private
`fn handle_*` is a helper and is left alone.

A handler may be `pub async fn` when it waits on something, such as a
database: the table awaits it and the wait holds no thread. A page that
waits on nothing stays a plain `fn`. An async handler is dropped at an
`.await` when the client goes away, so do not leave a write half done
across one.

**Adding a file does not rebuild on its own** -- nothing referenced it
yet. Run `touch src/pages/mod.rs` after creating a page or a layout.
Editing or deleting one is noticed.

A handler returns a value and never writes a response itself. That is
what makes a half-written body on an error impossible rather than
unlikely.

## Views

The markup for a route is a `.rsx` file **next to the page that renders it**:
`src/pages/index.rsx` beside `src/pages/index.rs`. There is no second tree and
nothing to register -- `rsx::load_dir!("src/pages")` in `src/pages/mod.rs`
compiles them all into the binary. One file is one module, under `views`, so
`index.rsx` is `views::index` and `users/[id].rsx` is `views::users::_id_`.

### Layouts

`layout.rsx` wraps every page in its directory **and every directory below**,
and a page never names it: which layout wraps a page is a property of where
the page is. That is the Crystal page_router's rule for `layout.cr`.

They stack root to leaf, and **the chain is not written down anywhere**.
`html!` is a macro: it knows which file it was written in, walks up to
`src/`, and wraps the page in every `layout.rsx` it passes. So
`src/pages/users/layout.rsx` wraps the pages under `users/` inside the root
layout, and deleting that file removes exactly that layer with no other edit.

Every page uses the same import and the same call, whatever directory it is
in. Do not add a chain of your own; if a page is wrapped wrongly, the
directory tree is what to change.

A layout takes `children: Element`, and may also take `layout: &str`, which
`html!` fills with the URL prefix that layout covers. Those two are the only
parameters it fills, because it cannot know what a third one would mean.
A value every page shares comes from a `const` -- the root layout reads
`pages::TITLE` that way. A value belonging to one request belongs on the page
component, which the handler builds and which already has the context.

For a fragment that is not a document -- an htmx swap, a partial -- call
`render` instead of `html!`: the same thing with no shell.

**Adding a `layout.rsx` where there was none does not rebuild anything**,
because nothing referenced the file yet. Run `touch src/pages/mod.rs` after
creating one. Editing or deleting an existing layout is noticed.

A component is a function returning `Element`. Its parameters are its contract
with the handler, and a caller fills them as struct fields, so a field the
component does not declare is a compile error rather than a blank in the page.
A parameter of type `Element` receives children, which is all a layout is.

Reach a template from a handler with `html!`, and with nothing else:

```rust
pub fn handle(_ctx: &mut Ctx) -> Result<impl IntoResponse, Error> {
    html!(ctx, views::index::Index { title: TITLE })
}
```

It returns the handler's own `Result`, so there is no `Ok(..)` to write and a
template that bails propagates with `?`. For another status or a header,
compose with the other return shapes: `Ok((html!(ctx, tpl)?, 404))`.

Adding a page with markup is two files in one directory -- `<name>.rs` and
`<name>.rsx` -- and nothing else anywhere. Neither needs a line in
`src/pages/mod.rs`, because the directory is the registry.

Not every response is a page. A redirect is `Response::empty(303)` with a
header and no template anywhere near it.

An interpolated value is escaped on the way out. Templates are read at
**compile** time, so an edit needs a rebuild, and a template that does not
parse is a build failure on its own line rather than a 500.

## The database

`src/plugins/db/mod.rs` opens one Postgres pool at startup from `[db] url`.
A page reads it with `ctx.db()`, a `&brocanth_db::Db`. Build a statement
with `brocanth_db::sea_query`. Do not format values into SQL text: sea-query
sends each value as a bound parameter.

```rust
use brocanth_db::sea_query::{Expr, ExprTrait, Order, Query};
use brocanth_db::Exec;

use crate::pages::{Ctx, Error, IntoResponse};

pub async fn handle(ctx: &mut Ctx) -> Result<impl IntoResponse, Error> {
    let q = Query::select()
        .columns(["id", "title"])
        .from("posts")
        .and_where(Expr::col("published").eq(true))
        .order_by("id", Order::Desc)
        .limit(10)
        .to_owned();
    let posts: Vec<(i64, String)> = ctx.db().all(&q).await?;
    Ok(posts
        .iter()
        .map(|(id, title)| format!("{id} {title}\n"))
        .collect::<String>())
}
```

`ctx.db().json(&q).await?` returns each row as a `serde_json::Value`,
encoded by Postgres. Writes that belong together go in one transaction.
`Ok` commits, and `Err` rolls back:

```rust
ctx.db()
    .transact(async |tx| {
        tx.execute(&debit).await?;
        tx.execute(&credit).await?;
        Ok(())
    })
    .await?;
```

A `brocanth_db::DbError` goes through `?` as a 500. Its message goes to the
log, never to the client.

## Pagination

A list reads its page with `ctx.page_params()?` and builds a
`brocanth_pagination::Page` from the count. Draw it with `<PageNav pager={&pager} />` from
`src/pages/parts/pager.rsx`. The links come
from `ctx.links()`, which keeps the filters and the sort.

- Put the rule, the filters and the search in the `WHERE` of the
  count and of the page. Never drop rows after the page is read: page
  two then overlaps page one, and the total is wrong.
- Order by a unique key last. Without it, two pages can share a row.
- A long list or a feed uses a keyset page (`after=<cursor>`), not an
  offset: it reads one index seek, not every row before it.
- Build the keyset `WHERE` with `brocanth_pagination::Keyset`:
  `rule.and(keyset.after(params.after.as_ref())?)`, compiled once, so
  a forged cursor only moves the window inside the rule.

## Filters

A list declares its filters once with `brocanth_filters::Filters`,
over the entity's column enum, and reads them with
`ctx.filters(&declared)?`.

- Compile with `applied.condition::<E>(policy.list(&who), dialect)`.
  It ANDs the filters with the rule, so a filter narrows what the
  rule admits and never widens it.
- Give the count and the page the same condition. Never filter rows
  after a paginated load: the total is then wrong and pages overlap.
- Order with `applied.order_by::<E>(key)`: the primary key comes last,
  so two pages never share a row.
- Do not parse a filter from the query string by hand. A name that is
  not declared names no column, and that is the guarantee.

- Write page links with `ctx.links()`: each keeps the filters and
  the sort.

## Search

`/search` ranks the rows of `search_documents`, the site index.
`migrations/20260101_000000_search_documents.sql` creates the table; `migrate up` applies it.

- Write a row with `brocanth_search::documents::upsert(ctx.db(), kind,
  key, title, body, url)` when a thing changes, in the same
  transaction or job, and `documents::remove` when it goes.
- Every row is public. Index only what any visitor may read, or put
  the rule in `src/pages/search.rs`: it goes into the `WHERE`, so a
  hidden row is never ranked, counted or shown.
- Never filter hits after the page is read. The total and the pages
  then disagree.
- To search one table instead, declare a `brocanth_search::Index`
  over its entity and write `index.migration()` into a migration.

## Decisions that belong to the person

The modules in this project are the person's choice. Do not add or
remove one on your own. Before you run `brocanth add`, show the person
the question: `brocanth add --questions` lists each module with its
default, what it needs and what it recommends. Give your
recommendation, and wait for the answer.

Not chosen, and recommended by a module that is here. Ask about each:

- `observability-health`, for `db`: its readiness check, `Db::ping`, is read only by /readyz; without it a deploy sends traffic to an instance whose database is down

## House style

- A comment says **why**, next to the decision, and what was chosen
  against. Not what the line does.
- A test name is a sentence describing the guarantee it protects.
- Prefer a type that cannot be built wrong over a check that can be
  forgotten.
- State a limit where someone would look for it rather than leaving it
  to be discovered.
